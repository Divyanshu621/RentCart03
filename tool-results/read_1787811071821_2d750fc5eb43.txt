   143→export default function PaymentCheckoutModal({
   144→  open,
   145→  onClose,
   146→  rentalId,
   147→  rentalData,
   148→  onSuccess,
   149→}: PaymentCheckoutModalProps) {
   150→  const isMobile = useIsMobile();
   151→  const queryClient = useQueryClient();
   152→  const [enabledMethods, setEnabledMethods] = useState<Record<string, boolean> | null>(null);
   153→  const [selectedMethod, setSelectedMethod] = useState('razorpay');
   154→  const [upiId, setUpiId] = useState('');
   155→  const [cardNumber, setCardNumber] = useState('');
   156→  const [cardExpiry, setCardExpiry] = useState('');
   157→  const [cardCvv, setCardCvv] = useState('');
   158→  const [cardName, setCardName] = useState('');
   159→  const [selectedBank, setSelectedBank] = useState('');
   160→  const [selectedWallet, setSelectedWallet] = useState('');
   161→  const [processing, setProcessing] = useState(false);
   162→  const [step, setStep] = useState<'checkout' | 'processing' | 'success'>('checkout');
   163→  const [txnId, setTxnId] = useState('');
   164→  const [summaryExpanded, setSummaryExpanded] = useState(false);
   165→
   166→  // Fetch enabled payment methods from backend
   167→  useEffect(() => {
   168→    if (!open) return;
   169→    let cancelled = false;
   170→    api.getPaymentSettings().then((res) => {
   171→      if (cancelled) return;
   172→      setEnabledMethods(res.enabledMethods);
   173→      // Auto-select first enabled method if current selection is disabled
   174→      if (!res.enabledMethods[selectedMethod]) {
   175→        const firstEnabled = Object.entries(res.enabledMethods).find(([, v]) => v)?.[0];
   176→        if (firstEnabled) setSelectedMethod(firstEnabled);
   177→      }
   178→    }).catch(() => {
   179→      // Fallback: enable all if settings fetch fails
   180→      if (!cancelled) {
   181→        setEnabledMethods({
   182→          razorpay: true, upi: true, card: true, netbanking: true, wallet: true, cash: true,
   183→        });
   184→      }
   185→    });
   186→    return () => { cancelled = true; };
   187→  }, [open]);
   188→
   189→  // Filter payment methods based on backend settings
   190→  const paymentMethods = useMemo(() => {
   191→    if (!enabledMethods) return allPaymentMethods; // show all while loading
   192→    return allPaymentMethods.filter((m) => enabledMethods[m.id]);
   193→  }, [enabledMethods]);
   194→
   195→  const dateRange = useMemo(() => {
   196→    if (!rentalData) return '';
   197→    return `${formatDate(rentalData.startDate)} - ${formatDate(rentalData.endDate)}, ${new Date(rentalData.endDate).getFullYear()} (${rentalData.rentalDays} days)`;
   198→  }, [rentalData]);
   199→
   200→  const formatCardNumber = (value: string) => {
   201→    const v = value.replace(/\D/g, '').slice(0, 16);
   202→    return v.replace(/(.{4})/g, '$1 ').trim();
   203→  };
   204→
   205→  const formatExpiry = (value: string) => {
   206→    const v = value.replace(/\D/g, '').slice(0, 4);
   207→    if (v.length >= 3) return v.slice(0, 2) + '/' + v.slice(2);
   208→    return v;
   209→  };
   210→
   211→  const validateUPI = (id: string) => {
   212→    return id.includes('@') && id.length >= 3;
   213→  };
   214→
   215→  const validateCard = () => {
   216→    const num = cardNumber.replace(/\s/g, '');
   217→    return num.length === 16 && cardExpiry.length === 5 && cardCvv.length === 3 && cardName.trim().length > 0;
   218→  };

   220→  const getPaymentButtonLabel = () => {
   221→    const amount = `₹${rentalData ? formatCurrency(rentalData.totalAmount) : '0'}`;
   222→    switch (selectedMethod) {
   223→      case 'razorpay': return `Pay ${amount} with Razorpay`;
   224→      case 'upi': return `Pay ${amount} via UPI`;
   225→      case 'card': return `Pay ${amount} via Card`;
   226→      case 'netbanking': return `Pay ${amount} via Net Banking`;
   227→      case 'wallet': return `Pay ${amount} via Wallet`;
   228→      case 'cash': return 'Confirm Cash on Pickup';
   229→      default: return `Pay ${amount}`;
   230→    }
   231→  };
   232→
   233→  const isPayDisabled = () => {
   234→    if (processing) return true;
   235→    switch (selectedMethod) {
   236→      case 'upi': return !validateUPI(upiId);
   237→      case 'card': return !validateCard();
   238→      case 'netbanking': return !selectedBank;
   239→      case 'wallet': return !selectedWallet;
   240→      default: return false;
   241→    }
   242→  };
   243→
   244→  const resetState = useCallback(() => {
   245→    setSelectedMethod('razorpay');
   246→    setUpiId('');
   247→    setCardNumber('');
   248→    setCardExpiry('');
   249→    setCardCvv('');
   250→    setCardName('');
   251→    setSelectedBank('');
   252→    setSelectedWallet('');
   253→    setProcessing(false);
   254→    setStep('checkout');
   255→    setTxnId('');
   256→    setSummaryExpanded(false);
   257→  }, []);
   258→
   259→  const handlePaymentSuccess = useCallback((transactionId: string) => {
   260→    setTxnId(transactionId);
   261→    setStep('success');

   263→    queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   264→    queryClient.invalidateQueries({ queryKey: ['my-rentals'] });
   265→  }, [queryClient]);
   266→
   267→  const handlePaymentError = useCallback(() => {

   269→    setProcessing(false);
   270→    setStep('checkout');
   271→  }, []);
   272→
   273→  const handlePayment = async () => {
   274→    if (!rentalData) return;
   275→
   276→    setProcessing(true);
   277→    setStep('processing');
   278→
   279→    try {
   280→      if (selectedMethod === 'cash') {
   281→        await api.verifyPayment({
   282→          rentalId,
   283→          paymentMethod: 'CASH_ON_PICKUP',
   284→        });
   285→        const txId = `CASH-${Date.now()}-${Math.random().toString(36).substring(2, 8)}`;
   286→        handlePaymentSuccess(txId);
   287→        return;
   288→      }
   289→
   290→      // Create order
   291→      const orderData = await api.createPaymentOrder(rentalId, selectedMethod.toUpperCase()) as Record<string, unknown>;
   292→
   293→      if (orderData.method === 'razorpay' && typeof window !== 'undefined' && window.Razorpay) {
   294→        // Real Razorpay mode
   295→        const rzpOptions: Record<string, unknown> = {
   296→          key: orderData.key,
   297→          amount: Math.round((orderData.amount as number) * 100),
   298→          currency: orderData.currency,
   299→          name: 'RentCart',
   300→          description: `Rental: ${rentalData.productTitle || 'Item'}`,
   301→          order_id: orderData.orderId,
   302→          prefill: {
   303→            name: (orderData.customer as Record<string, string>)?.name || '',
   304→            email: (orderData.customer as Record<string, string>)?.email || '',
   305→            contact: (orderData.customer as Record<string, string>)?.contact || '',
   306→          },
   307→          theme: { color: '#059669' },
   308→          modal: !isMobile,
   309→        };
   310→
   311→        const rzp = new window.Razorpay(rzpOptions);
   312→
   313→        rzp.on('payment.failed', () => {
   314→          handlePaymentError();
   315→        });
   316→
   317→        rzp.open();
   318→        // Don't reset processing here — the handler callbacks will do it
   319→        return;
   320→      }
   321→
   322→      // Simulated/demo mode — add a short delay to mimic real payment
   323→      await new Promise((resolve) => setTimeout(resolve, 2000));
   324→      await api.verifyPayment({
   325→        rentalId,
   326→        paymentMethod: selectedMethod.toUpperCase(),
   327→      });
   328→      const txId = (orderData.orderId as string) || `TXN-${Date.now()}`;
   329→      handlePaymentSuccess(txId);
   330→    } catch {
   331→      handlePaymentError();
   332→    }
   333→  };
   334→
   335→  const handleClose = () => {
   336→    if (processing) return;
   337→    if (step === 'success') {
   338→      onSuccess?.();
   339→    }
   340→    resetState();
   341→    onClose();
   342→  };
   343→
   344→  const handleSuccess = () => {
   345→    onSuccess?.();
   346→    resetState();
   347→    onClose();
   348→  };
   349→
   350→  const renderOrderSummary = () => (
   351→    <div className="rounded-xl border border-gray-200 bg-gray-50/50 overflow-hidden">
   352→      {/* Compact header — always visible on mobile */}
   353→      <button
   354→        type="button"
   355→        onClick={() => setSummaryExpanded(!summaryExpanded)}
   356→        className="w-full flex items-center justify-between p-3 sm:p-4 hover:bg-gray-100/50 transition-colors"
   357→      >
   358→        <div className="flex items-center gap-2.5 min-w-0">
   359→          <h3 className="text-sm font-semibold text-gray-900">Order Summary</h3>
   360→          <Badge variant="secondary" className="text-[10px] font-medium bg-emerald-100 text-emerald-700 shrink-0">
   361→            {rentalData?.rentalDays} days
   362→          </Badge>
   363→        </div>
   364→        <div className="flex items-center gap-2 shrink-0">
   365→          <span className="text-sm font-bold text-gray-900">₹{formatCurrency(rentalData?.totalAmount ?? 0)}</span>
   366→          {isMobile && (
   367→            summaryExpanded ? <ChevronUp className="size-4 text-gray-400" /> : <ChevronDown className="size-4 text-gray-400" />
   368→          )}
   369→        </div>
   370→      </button>
   371→
   372→      {/* Expandable details — always shown on desktop, toggleable on mobile */}
   373→      <AnimatePresence>
   374→        {(!isMobile || summaryExpanded) && (
   375→          <motion.div
   376→            initial={isMobile ? { height: 0, opacity: 0 } : false}
   377→            animate={{ height: 'auto', opacity: 1 }}
   378→            exit={{ height: 0, opacity: 0 }}
   379→            transition={{ duration: 0.2 }}
   380→            className="overflow-hidden"
   381→          >
   382→            <Separator className="bg-gray-200" />
   383→            <div className="p-3 sm:p-4 space-y-2">
   384→              <p className="text-sm font-medium text-gray-800 truncate">
   385→                {rentalData?.productTitle || 'Rental Item'}
   386→              </p>
   387→              <p className="text-xs text-gray-500">{dateRange}</p>
   388→
   389→              <Separator className="my-2.5 bg-gray-200" />
   390→
   391→              <div className="space-y-1.5 text-xs">
   392→                <div className="flex items-center justify-between">
   393→                  <span className="text-gray-500">Rental Amount</span>
   394→                  <span className="font-medium text-gray-700">
   395→                    ₹{rentalData?.dailyRate.toLocaleString('en-IN')} × {rentalData?.rentalDays} days
   396→                  </span>
   397→                </div>
   398→                <div className="flex items-center justify-between">
   399→                  <span className="text-gray-500">Platform Fee (10%)</span>
   400→                  <span className="font-medium text-gray-700">₹{formatCurrency(rentalData?.platformFee ?? 0)}</span>
   401→                </div>
   402→                <div className="flex items-center justify-between">
   403→                  <span className="text-gray-500">GST (18%)</span>
   404→                  <span className="font-medium text-gray-700">₹{formatCurrency(rentalData?.tax ?? 0)}</span>
   405→                </div>
   406→                {rentalData && rentalData.deliveryFee > 0 && (
   407→                  <div className="flex items-center justify-between">
   408→                    <span className="text-gray-500">Delivery Fee</span>
   409→                    <span className="font-medium text-gray-700">₹{formatCurrency(rentalData.deliveryFee)}</span>
   410→                  </div>
   411→                )}
   412→                <div className="flex items-center justify-between">
   413→                  <span className="text-gray-500">Security Deposit</span>
   414→                  <div className="flex items-center gap-1.5">
   415→                    <span className="font-medium text-gray-700">₹{formatCurrency(rentalData?.securityDeposit ?? 0)}</span>
   416→                    <span className="text-[10px] text-emerald-600 font-medium">Refundable</span>
   417→                  </div>
   418→                </div>
   419→                {(rentalData?.discount ?? 0) > 0 && (
   420→                  <div className="flex items-center justify-between">
   421→                    <span className="text-gray-500">Discount</span>
   422→                    <span className="font-medium text-emerald-600">-₹{formatCurrency(rentalData?.discount ?? 0)}</span>
   423→                  </div>
   424→                )}
   425→              </div>
   426→
   427→              <Separator className="my-2.5 bg-gray-200" />
   428→
   429→              <div className="flex items-center justify-between">
   430→                <span className="text-sm font-bold text-gray-900">Total Payable</span>
   431→                <span className="text-base font-bold text-gray-900">₹{formatCurrency(rentalData?.totalAmount ?? 0)}</span>
   432→              </div>
   433→            </div>
   434→          </motion.div>
   435→        )}
   436→      </AnimatePresence>
   437→    </div>
   438→  );
   439→

   441→    <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   442→      {paymentMethods.map((method) => {
   443→        const isSelected = selectedMethod === method.id;
   444→        return (
   445→          <motion.button
   446→            key={method.id}
   447→            type="button"
   448→            whileTap={{ scale: 0.97 }}
   449→            onClick={() => {
   450→              setSelectedMethod(method.id);
   451→              setSelectedBank('');
   452→              setSelectedWallet('');
   453→            }}
   454→            className={`
   455→              relative flex items-center gap-3 rounded-xl border-2 p-3 cursor-pointer transition-all text-left w-full min-h-[56px]
   456→              ${isSelected
   457→                ? 'border-emerald-500 bg-emerald-50 shadow-sm'
   458→                : 'border-gray-200 hover:border-gray-300 bg-white'
   459→              }
   460→            `}
   461→          >
   462→            <div
   463→              className={`
   464→                flex items-center justify-center size-9 rounded-lg shrink-0
   465→                ${isSelected ? 'bg-emerald-100 text-emerald-700' : 'bg-gray-100 text-gray-500'}
   466→              `}
   467→            >
   468→              {method.icon}
   469→            </div>
   470→            <div className="min-w-0 flex-1">
   471→              <div className="flex items-center gap-2">
   472→                <span className={`text-sm font-semibold truncate ${isSelected ? 'text-emerald-800' : 'text-gray-800'}`}>
   473→                  {method.label}
   474→                </span>
   475→                {method.tag && method.tagColor && (
   476→                  <Badge
   477→                    className={`text-[10px] px-1.5 py-0 font-semibold leading-tight border-0 shrink-0 ${method.tagColor}`}
   478→                  >
   479→                    {method.tag}
   480→                  </Badge>
   481→                )}
   482→              </div>
   483→              <p className="text-[11px] text-gray-500 mt-0.5 truncate">{method.desc}</p>
   484→            </div>
   485→            {isSelected && (
   486→              <div className="flex items-center justify-center size-5 rounded-full bg-emerald-500 text-white shrink-0">
   487→                <CheckCircle2 className="size-3.5" />
   488→              </div>
   489→            )}
   490→          </motion.button>
   491→        );
   492→      })}
   493→    </div>
   494→  );
   495→
   496→  const renderUPIForm = () => (
   497→    <motion.div
   498→      key="upi-form"
   499→      initial={{ opacity: 0, height: 0 }}
   500→      animate={{ opacity: 1, height: 'auto' }}
   501→      exit={{ opacity: 0, height: 0 }}
   502→      transition={{ duration: 0.2 }}
   503→      className="overflow-hidden"
   504→    >
   505→      <div className="rounded-xl border border-gray-200 bg-white p-3 sm:p-4 space-y-3">
   506→        <div>
   507→          <Label className="text-xs font-semibold text-gray-700 mb-2 block">
   508→            Quick Pay with UPI App
   509→          </Label>
   510→          <div className="grid grid-cols-4 gap-2">
   511→            {upiApps.map((app) => (

   513→                key={app.name}
   514→                type="button"
   515→                whileTap={{ scale: 0.95 }}
   516→                onClick={() => {
   517→                  toast.info(`Opening ${app.name}...`, { description: 'Complete the payment in the UPI app' });
   518→                }}
   519→                className={`flex flex-col items-center gap-1.5 rounded-lg p-2.5 transition-all hover:shadow-sm min-h-[56px] ${app.color}`}
   520→              >
