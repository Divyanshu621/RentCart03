   394→   394→    register,
   395→   395→    handleSubmit,
   396→   396→    control,
   397→   397→    setValue,
   398→   398→    formState: { errors, isSubmitting },
   399→   399→  } = useForm<RegisterFormData>({
   400→   400→    resolver: zodResolver(registerSchema),
   401→   401→    defaultValues: {
   402→   402→      name: '',
   403→   403→      email: '',
   404→   404→      phone: '',
   405→   405→      password: '',
   406→   406→      stateId: '',
   407→   407→      cityId: '',
   408→   408→      pinCode: '',
   409→   409→      address: '',
   410→   410→      agreeTerms: false as unknown as true,

   413→   413→
   414→   414→  const selectedStateId = useWatch({ control, name: 'stateId' });
   415→   415→  const selectedState = states.find((s) => s.id === selectedStateId);
   416→   416→  const cities = selectedState?.cities?.filter((c) => c.isActive) || [];
   417→   417→

   420→   420→  const onSubmit = async (data: RegisterFormData) => {
   421→   421→    setServerError('');

   423→   423→      const res = await api.register({
   424→   424→        name: data.name,
   425→   425→        email: data.email,
   426→   426→        phone: data.phone || undefined,
   427→   427→        password: data.password,
