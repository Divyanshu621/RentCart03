   588→                      </button>
   589→                    </DropdownMenuTrigger>
   590→                    <UserMenuContent
   591→                      user={user}
   592→                      isAdmin={isAdmin}
   593→                      onNavigate={(view) => navigate(view)}
   594→                      onLogout={handleLogout}
   595→                    />
   596→                  </DropdownMenu>
   597→                </>
   598→              ) : (
   599→                <div className="flex items-center gap-2">
   600→                  <Button
   601→                    variant="ghost"
   602→                    size="sm"
   603→                    onClick={() => { setAuthModalView('login'); setAuthModalOpen(true); }}
   604→                    className="text-[#64748b] hover:text-[#0f172a] hover:bg-gray-100"
   605→                  >
   606→                    Login
   607→                  </Button>
   608→                  <Button
   609→                    size="sm"
   610→                    onClick={() => { setAuthModalView('register'); setAuthModalOpen(true); }}
   611→                    className="bg-orange-500 hover:bg-orange-600 text-white"
   612→                  >
   613→                    Sign Up
   614→                  </Button>
   615→                </div>
   616→              )}
   617→
   618→              {/* Mobile menu hamburger */}
   619→              <button
   620→                onClick={() => setShowMobile(true)}
   621→                className="p-2 rounded-lg text-[#64748b] hover:text-[#0f172a] hover:bg-gray-100 md:hidden"
   622→              >
   623→                <Menu className="w-5 h-5" />
   624→              </button>
   625→            </div>
   626→          </div>
   627→        </div>
   628→      </header>
   629→
   630→      {/* Category pills bar - only on marketplace */}
   631→      {isMarketplace && displayCategories.length > 0 && (
   632→        <div className="bg-gray-50 border-b border-[#e2e8f0]">
   633→          <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
   634→            <div
   635→              className="flex gap-2 overflow-x-auto py-2"
   636→              style={{ scrollbarWidth: 'none', msOverflowStyle: 'none' }}
   637→            >
   638→              <button
   639→                onClick={() => navigate('marketplace')}
   640→                className="shrink-0 px-4 py-1.5 rounded-full text-xs font-medium transition-colors bg-white text-[#64748b] border border-[#e2e8f0] hover:border-emerald-300 hover:text-emerald-600"
   641→              >
   642→                All
   643→              </button>
   644→              {displayCategories.map((cat) => (
   645→                <button
   646→                  key={cat.id}
   647→                  onClick={() => handleCategoryPillClick(cat.id)}
   648→                  className="shrink-0 px-4 py-1.5 rounded-full text-xs font-medium transition-colors text-[#64748b] hover:bg-emerald-50 hover:text-emerald-600 border border-transparent"
   649→                >
   650→                  {cat.name}
   651→                </button>
   652→              ))}
   653→            </div>
   654→          </div>
   655→        </div>
   656→      )}
   657→
   658→      {/* Mobile Location Picker Sheet */}
   659→      <Sheet open={showLocationMobile} onOpenChange={setShowLocationMobile}>
   660→        <SheetContent side="bottom" className="rounded-t-2xl">
   661→          <SheetHeader className="p-4 pb-2">
   662→            <SheetTitle className="flex items-center gap-2 text-base">
   663→              <MapPin className="w-4 h-4 text-[#059669]" />
   664→              Select Location
   665→            </SheetTitle>
   666→          </SheetHeader>
   667→          <div className="px-4 pb-6 space-y-4">
   668→            <div className="space-y-1.5">
   669→              <label className="text-xs font-medium text-[#64748b]">State</label>
   670→              <Select value={selectedState?.id ?? ''} onValueChange={(val) => { handleStateChange(val); }}>
   671→                <SelectTrigger className="h-10 w-full">
   672→                  <SelectValue placeholder="Select a state" />
   673→                </SelectTrigger>
   674→                <SelectContent>
   675→                  {states.map((s) => (
   676→                    <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
   677→                  ))}
   678→                </SelectContent>
   679→              </Select>
   680→            </div>
   681→
   682→            {selectedState && cities.length > 0 && (
   683→              <div className="space-y-1.5">
   684→                <label className="text-xs font-medium text-[#64748b]">City</label>
   685→                <Select value={selectedCity?.id ?? ''} onValueChange={(val) => { handleCityChange(val); }}>
   686→                  <SelectTrigger className="h-10 w-full">
   687→                    <SelectValue placeholder="Select a city" />
   688→                  </SelectTrigger>
   689→                  <SelectContent>
   690→                    {cities.map((c) => (
   691→                      <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   692→                    ))}
   693→                  </SelectContent>
   694→                </Select>
   695→              </div>
   696→            )}
   697→
   698→            {selectedCity && areas.length > 0 && (
   699→              <div className="space-y-1.5">
   700→                <label className="text-xs font-medium text-[#64748b]">Area</label>
   701→                <Select value={selectedArea?.id ?? ''} onValueChange={(val) => { handleAreaChange(val); setShowLocationMobile(false); }}>
   702→                  <SelectTrigger className="h-10 w-full">
   703→                    <SelectValue placeholder="Select an area" />
   704→                  </SelectTrigger>
   705→                  <SelectContent>
   706→                    {areas.map((a) => (
   707→                      <SelectItem key={a.id} value={a.id}>{a.name}</SelectItem>
   708→                    ))}
   709→                  </SelectContent>
   710→                </Select>
   711→              </div>
   712→            )}
   713→
   714→            {(selectedState || selectedCity || selectedArea) && (
   715→              <Button
   716→                variant="ghost"
   717→                size="sm"
   718→                className="w-full text-sm text-[#64748b]"
   719→                onClick={() => {
   720→                  setSelectedState(null);
   721→                  setSelectedCity(null);
   722→                  setSelectedArea(null);
   723→                  navigate('marketplace');
   724→                  setShowLocationMobile(false);
   725→                }}
   726→              >
   727→                Clear location selection
   728→              </Button>
   729→            )}
   730→          </div>
   731→        </SheetContent>
   732→      </Sheet>
   733→
   734→      {/* Mobile Side Sheet */}
   735→      <MobileSideSheet
   736→        open={showMobile}
   737→        onOpenChange={setShowMobile}
   738→        user={user}
   739→        isAdmin={isAdmin}
   740→        onNavigate={(view) => { navigate(view); setShowMobile(false); }}
   741→        onAuth={(view) => { setAuthModalView(view); setAuthModalOpen(true); setShowMobile(false); }}
   742→        onLogout={() => { handleLogout(); setShowMobile(false); }}
   743→      />
   744→    </>
   745→  );
   746→}

   749→function NotificationPanel({
