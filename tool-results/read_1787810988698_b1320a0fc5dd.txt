    37→    37→function AnimatedCounter({ target, prefix = '', suffix = '', decimals = 0 }: {
    38→    38→  target: number; prefix?: string; suffix?: string; decimals?: number;
    39→    39→}) {
    40→    40→  const [count, setCount] = useState(0);
    41→    41→  const ref = useRef<HTMLSpanElement>(null);
    42→    42→  const started = useRef(false);
    43→    43→
    44→    44→  useEffect(() => {
    45→    45→    const el = ref.current;
    46→    46→    if (!el) return;
    47→    47→    const observer = new IntersectionObserver(
    48→    48→      ([entry]) => {
    49→    49→        if (entry.isIntersecting && !started.current) {
    50→    50→          started.current = true;
    51→    51→          const duration = 1200;
    52→    52→          const startTime = performance.now();
    53→    53→          const animate = (now: number) => {
    54→    54→            const elapsed = now - startTime;
    55→    55→            const progress = Math.min(elapsed / duration, 1);
    56→    56→            // ease-out cubic
    57→    57→            const eased = 1 - Math.pow(1 - progress, 3);
    58→    58→            setCount(eased * target);
    59→    59→            if (progress < 1) requestAnimationFrame(animate);
    60→    60→          };
    61→    61→          requestAnimationFrame(animate);
    62→    62→        }
    63→    63→      },
    64→    64→      { threshold: 0.1 }
    65→    65→    );
    66→    66→    observer.observe(el);
    67→    67→    return () => observer.disconnect();
    68→    68→  }, [target]);
    69→    69→
    70→    70→  const formatted = decimals > 0
    71→    71→    ? count.toLocaleString('en-IN', { minimumFractionDigits: decimals, maximumFractionDigits: decimals })
    72→    72→    : Math.round(count).toLocaleString('en-IN');
    73→    73→
    74→    74→  return <span ref={ref}>{prefix}{formatted}{suffix}</span>;
    75→    75→}
