    34→// ─── Zod Schemas ────────────────────────────────────────────
    35→const loginSchema = z.object({

    37→  password: z.string().min(1, 'Password is required'),

    42→  email: z.string().min(1, 'Email is required').email('Enter a valid email'),

    47→  pinCode: z.string().optional(),
    48→  address: z.string().optional(),
    49→  agreeTerms: z.literal(true, { errorMap: () => ({ message: 'You must agree to the terms' }) }),
    50→});

    55→// ─── AuthModal Component ────────────────────────────────────
    56→export default function AuthModal() {
    57→  const {
    58→    authModalOpen,
    59→    setAuthModalOpen,
    60→    authModalView,
    61→    setAuthModalView,
    62→    setUser,
    63→    navigate,
    64→    returnUrl,
    65→    setReturnUrl,
    66→  } = useAppStore();
    67→
    68→  const [showPassword, setShowPassword] = useState(false);
    69→  const [states, setStates] = useState<State[]>([]);
    70→  const [loadingStates, setLoadingStates] = useState(true);
    71→

    73→  useEffect(() => {

    85→  const handleClose = useCallback(() => {

    87→  }, [setAuthModalOpen]);
    88→
    89→  const toggleView = useCallback(() => {
    90→    setAuthModalView(authModalView === 'login' ? 'register' : 'login');
    91→  }, [authModalView, setAuthModalView]);
    92→
    93→  const handleAuthSuccess = useCallback(async (userData: Record<string, unknown>) => {
    94→    try {
    95→      const meData = await api.me();
    96→      setUser(meData as unknown as UserType);
    97→    } catch {
    98→      setUser(userData as unknown as UserType);
    99→    }
   100→    toast.success(authModalView === 'login' ? 'Welcome back!' : 'Account created successfully!');
   101→    setAuthModalOpen(false);
   102→    setReturnUrl(null);

   113→              <Sparkles className="w-4 h-4 text-white" />

   115→            <span className="text-white font-bold text-lg">RentLoop</span>
   116→          </div>
   117→          <div className="flex items-center gap-1 bg-white/10 rounded-lg p-0.5">
   118→            <button
   119→              onClick={() => setAuthModalView('login')}
   120→              className={`px-3 py-1.5 text-sm font-medium rounded-md transition-all ${
   121→                authModalView === 'login'
   122→                  ? 'bg-white text-[#0f172a] shadow-sm'
   123→                  : 'text-white/70 hover:text-white'
   124→              }`}
   125→            >
   126→              Login
   127→            </button>
   128→            <button
   129→              onClick={() => setAuthModalView('register')}
   130→              className={`px-3 py-1.5 text-sm font-medium rounded-md transition-all ${
   131→                authModalView === 'register'
   132→                  ? 'bg-white text-[#0f172a] shadow-sm'
   133→                  : 'text-white/70 hover:text-white'
   134→              }`}
   135→            >
   136→              Sign Up
   137→            </button>
   138→          </div>
   139→        </div>
   140→
   141→        <div className="p-6">
   142→          {authModalView === 'login' ? (
   143→            <LoginForm
   144→              onSuccess={handleAuthSuccess}
   145→              onToggleView={toggleView}
   146→              showPassword={showPassword}
   147→              onTogglePassword={() => setShowPassword(!showPassword)}
   148→            />
   149→          ) : (
   150→            <RegisterForm
   151→              onSuccess={handleAuthSuccess}
   152→              onToggleView={toggleView}

   165→// ─── Login Form ─────────────────────────────────────────────
   166→function LoginForm({
   167→  onSuccess,
   168→  onToggleView,
   169→  showPassword,
   170→  onTogglePassword,
   171→}: {
   172→  onSuccess: (data: Record<string, unknown>) => void;
   173→  onToggleView: () => void;
   174→  showPassword: boolean;
   175→  onTogglePassword: () => void;
   176→}) {
   177→  const {
   178→    register,
   179→    handleSubmit,
   180→    formState: { errors, isSubmitting },
   181→    setError,
   182→  } = useForm<LoginFormData>({
   183→    resolver: zodResolver(loginSchema),
   184→  });
   185→
   186→  const [serverError, setServerError] = useState('');
   187→  const [isGoogleLoading, setIsGoogleLoading] = useState(false);
   188→  const [showGoogleDialog, setShowGoogleDialog] = useState(false);
   189→  const [googleEmail, setGoogleEmail] = useState('');
   190→  const [googleName, setGoogleName] = useState('');
   191→
   192→  const handleGoogleLogin = async () => {
   193→    if (!googleEmail.trim() || !googleEmail.includes('@')) {
   194→      toast.error('Please enter a valid email address');
   195→      return;
   196→    }
   197→    setIsGoogleLoading(true);
   198→    try {
   199→      const name = googleName.trim() || googleEmail.split('@')[0];
   200→      const result = await api.googleAuth({ email: googleEmail.trim(), name });
   201→      setShowGoogleDialog(false);
   202→      onSuccess(result.user);
   203→    } catch (err) {
   204→      const message = err instanceof Error ? err.message : 'Google sign-in failed';
   205→      toast.error(message);
   206→    } finally {
   207→      setIsGoogleLoading(false);
   208→    }
   209→  };
   210→
   211→  const onSubmit = async (data: LoginFormData) => {
   212→    setServerError('');
   213→    try {
   214→      const res = await api.login({ email: data.email, password: data.password });
   215→      onSuccess(res.user);
   216→    } catch (err) {
   217→      const message = err instanceof Error ? err.message : 'Login failed';
   218→      setServerError(message);
   219→      toast.error(message);
   220→    }
   221→  };
   222→
   223→  return (
   224→    <div>
   225→      <DialogHeader className="mb-6 text-left">
   226→        <DialogTitle className="text-xl font-bold text-[#0f172a]">
   227→        </DialogTitle>
   228→        <DialogDescription className="text-muted-foreground">
   229→        </DialogDescription>
   230→      </DialogHeader>

   232→      <h2 className="text-xl font-bold text-[#0f172a] mb-1">Welcome back</h2>
   233→      <p className="text-sm text-muted-foreground mb-6">
   234→        Sign in to your RentLoop account
   235→      </p>
   236→
   237→      {serverError && (
   238→        <div className="mb-4 p-3 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm">
   239→          {serverError}
   240→        </div>
   241→      )}
   242→
   243→      {/* Google sign-in divider + button */}
   244→      <div className="relative my-4">
   245→        <div className="absolute inset-0 flex items-center">
   246→          <div className="w-full border-t border-slate-200" />
   247→        </div>
   248→        <div className="relative flex justify-center text-xs">
   249→          <span className="px-3 bg-white text-muted-foreground uppercase tracking-wider">
   250→            or
   251→          </span>
   252→        </div>
   253→      </div>
   254→
   255→      <Button
   256→        type="button"
   257→        variant="outline"
   258→        onClick={() => { setGoogleEmail(''); setGoogleName(''); setShowGoogleDialog(true); }}
   259→        className="w-full h-11 border-slate-200 hover:bg-slate-50 font-medium text-sm"
   260→      >
   261→        <svg className="w-5 h-5" viewBox="0 0 24 24">
   262→          <path d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92a5.06 5.06 0 01-2.2 3.32v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.1z" fill="#4285F4" />
   263→          <path d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z" fill="#34A853" />
   264→          <path d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z" fill="#FBBC05" />
   265→          <path d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z" fill="#EA4335" />
   266→        </svg>
   267→        Continue with Google
   268→      </Button>
   269→
   270→      {/* Google email dialog */}
   271→      {showGoogleDialog && (
   272→        <div className="fixed inset-0 z-[200] bg-black/50 flex items-center justify-center p-4" onClick={() => !isGoogleLoading && setShowGoogleDialog(false)}>
   273→          <div className="bg-white rounded-2xl shadow-2xl w-full max-w-sm p-6" onClick={(e) => e.stopPropagation()}>
   274→            <div className="flex items-center justify-between mb-5">
   275→              <div className="flex items-center gap-3">
   276→                <svg className="w-7 h-7" viewBox="0 0 24 24">
   277→                  <path d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92a5.06 5.06 0 01-2.2 3.32v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.1z" fill="#4285F4" />
   278→                  <path d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z" fill="#34A853" />
   279→                  <path d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z" fill="#FBBC05" />
   280→                  <path d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z" fill="#EA4335" />
   281→                </svg>
   282→                <div>
   283→                  <h3 className="font-bold text-[#0f172a]">Sign in with Google</h3>
   284→                  <p className="text-xs text-muted-foreground">Enter your Gmail to continue</p>
   285→                </div>
   286→              </div>
   287→              <button onClick={() => setShowGoogleDialog(false)} disabled={isGoogleLoading} className="p-1.5 rounded-lg hover:bg-gray-100 text-gray-400">
   288→                <X className="w-4 h-4" />
   289→              </button>
   290→            </div>
   291→            <div className="space-y-4">
   292→              <div className="space-y-1.5">
   293→                <Label className="text-sm font-medium text-[#0f172a]">Gmail Address</Label>
   294→                <Input type="email" placeholder="you@gmail.com" value={googleEmail} onChange={(e) => setGoogleEmail(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleGoogleLogin()} disabled={isGoogleLoading} className="h-11" autoFocus />
   295→              </div>
   296→              <div className="space-y-1.5">
   297→                <Label className="text-sm font-medium text-[#0f172a]">Display Name <span className="text-muted-foreground font-normal">(optional)</span></Label>
   298→                <Input type="text" placeholder="Your name" value={googleName} onChange={(e) => setGoogleName(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleGoogleLogin()} disabled={isGoogleLoading} className="h-11" />
   299→              </div>
   300→              <Button onClick={handleGoogleLogin} disabled={isGoogleLoading || !googleEmail.trim()} className="w-full h-11 bg-emerald-600 hover:bg-emerald-700 text-white font-medium">
   301→                {isGoogleLoading ? (<><Loader2 className="w-4 h-4 animate-spin" /> Signing in...</>) : (<>Continue <ArrowRight className="w-4 h-4" /></>)}

   311→            Email
   312→          </Label>
   313→          <div className="relative">
   314→            <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   315→            <Input
   316→              id="login-email"
   317→              type="email"
   318→              placeholder="you@example.com"
   319→              className="pl-10 h-11"
   320→              {...register('email')}
   321→            />
   322→          </div>
   323→          {errors.email && (
   324→            <p className="text-sm text-red-500">{errors.email.message}</p>
   325→          )}
   326→        </div>
   327→
   328→        <div className="space-y-2">
   329→          <div className="flex items-center justify-between">
   330→            <Label htmlFor="login-password" className="text-sm font-medium text-[#0f172a]">
   331→              Password
   332→            </Label>
   333→            <button
   334→              type="button"
   335→              onClick={() => toast.info('Password reset link sent to your email!')}
   336→              className="text-sm text-emerald-600 hover:text-emerald-700 font-medium"
   337→            >
   338→              Forgot Password?
   339→            </button>

   395→// ─── Register Form ──────────────────────────────────────────
   396→function RegisterForm({
   397→  onSuccess,
   398→  onToggleView,
   399→  showPassword,
   400→  onTogglePassword,
   401→  states,
   402→  loadingStates,
   403→}: {
   404→  onSuccess: (data: Record<string, unknown>) => void;
   405→  onToggleView: () => void;
   406→  showPassword: boolean;
   407→  onTogglePassword: () => void;
   408→  states: State[];
   409→  loadingStates: boolean;
   410→}) {
   411→  const [serverError, setServerError] = useState('');
   412→
   413→  const {
   414→    register,
   415→    handleSubmit,
   416→    control,
   417→    watch,
   418→    setValue,
   419→    formState: { errors, isSubmitting },
   420→  } = useForm<RegisterFormData>({
   421→    resolver: zodResolver(registerSchema),
   422→    defaultValues: {
   423→      name: '',
   424→      email: '',
   425→      phone: '',
   426→      password: '',
   427→      stateId: '',
   428→      cityId: '',
   429→      pinCode: '',
   430→      address: '',
   431→      agreeTerms: false as unknown as true,
   432→    },
   433→  });
   434→
   435→  const selectedStateId = watch('stateId');
   436→  const selectedState = states.find((s) => s.id === selectedStateId);
   437→  const cities = selectedState?.cities?.filter((c) => c.isActive) || [];
   438→
   439→  // Reset city when state changes
   440→  useEffect(() => {
   441→    setValue('cityId', '');
   442→  }, [selectedStateId, setValue]);
   443→
   444→  const onSubmit = async (data: RegisterFormData) => {
   445→    setServerError('');
   446→    try {
   447→      const res = await api.register({
   448→        name: data.name,
   449→        email: data.email,
   450→        phone: data.phone || undefined,
   451→        password: data.password,
   452→        stateId: data.stateId,
   453→        cityId: data.cityId,
   454→        pinCode: data.pinCode || undefined,
   455→        address: data.address || undefined,
   456→      });
   457→      onSuccess(res.user);
   458→    } catch (err) {
   459→      const message = err instanceof Error ? err.message : 'Registration failed';
   460→      setServerError(message);
   461→      toast.error(message);
   462→    }
   463→  };
   464→
   465→  return (
   466→    <div>
   467→      <h2 className="text-xl font-bold text-[#0f172a] mb-1">Create your account</h2>
   468→      <p className="text-sm text-muted-foreground mb-5">
   469→        Join RentLoop and start renting today
   470→      </p>
   471→
   472→      {serverError && (
   473→        <div className="mb-4 p-3 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm">
   474→          {serverError}
   475→        </div>
   476→      )}
   477→
   478→      <form onSubmit={handleSubmit(onSubmit)} className="space-y-3.5 max-h-[55vh] overflow-y-auto pr-1 custom-scrollbar">
   479→        {/* Full Name */}
   480→        <div className="space-y-1.5">
   481→          <Label htmlFor="reg-name" className="text-sm font-medium text-[#0f172a]">
   482→            Full Name <span className="text-red-500">*</span>
   483→          </Label>
   484→          <div className="relative">
   485→            <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   486→            <Input
   487→              id="reg-name"
   488→              placeholder="John Doe"
   489→              className="pl-10 h-10"
   490→              {...register('name')}
   491→            />
   492→          </div>
   493→          {errors.name && (
   494→            <p className="text-xs text-red-500">{errors.name.message}</p>
   495→          )}
   496→        </div>
   497→
   498→        {/* Email */}
   499→        <div className="space-y-1.5">
   500→          <Label htmlFor="reg-email" className="text-sm font-medium text-[#0f172a]">
   501→            Email <span className="text-red-500">*</span>
   502→          </Label>
   503→          <div className="relative">
   504→            <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   505→            <Input
   506→              id="reg-email"
   507→              type="email"
   508→              placeholder="you@example.com"
   509→              className="pl-10 h-10"
   510→              {...register('email')}
   511→            />
   512→          </div>
   513→          {errors.email && (
   514→            <p className="text-xs text-red-500">{errors.email.message}</p>
   515→          )}
