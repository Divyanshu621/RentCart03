   118→   118→// ─── Countdown Timer ───────────────────────────────────────
   119→   119→function CountdownTimer({ endDate }: { endDate: string }) {
   120→   120→  const [timeLeft, setTimeLeft] = useState('');
   121→   121→
   122→   122→  useEffect(() => {
   123→   123→    const update = () => {
   124→   124→      const diff = new Date(endDate).getTime() - Date.now();
   125→   125→      if (diff <= 0) {
   126→   126→        setTimeLeft('Overdue');
   127→   127→        return;
   128→   128→      }
   129→   129→      const days = Math.floor(diff / (1000 * 60 * 60 * 24));
   130→   130→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
   131→   131→      const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
   132→   132→      const secs = Math.floor((diff % (1000 * 60)) / 1000);
   133→   133→      if (days > 0) setTimeLeft(`${days}d ${hours}h ${mins}m`);
   134→   134→      else if (hours > 0) setTimeLeft(`${hours}h ${mins}m ${secs}s`);
   135→   135→      else setTimeLeft(`${mins}m ${secs}s`);
   136→   136→    };
   137→   137→    update();
   138→   138→    const id = setInterval(update, 1000);
   139→   139→    return () => clearInterval(id);
   140→   140→  }, [endDate]);
   141→   141→
   142→   142→  return (
   143→   143→    <span className={`text-xs font-mono font-medium ${timeLeft === 'Overdue' ? 'text-red-600' : 'text-emerald-600'}`}>
   144→   144→      {timeLeft}
   145→   145→    </span>
   146→   146→  );
   147→   147→}

   377→   377→  const [selectedRental, setSelectedRental] = useState<Rental | null>(null);
   378→   378→  const [paymentModalOpen, setPaymentModalOpen] = useState(false);
   379→   379→  const [paymentRental, setPaymentRental] = useState<Rental | null>(null);
   380→   380→  const [cancelDialogOpen, setCancelDialogOpen] = useState(false);
   381→   381→  const [cancelRental, setCancelRental] = useState<Rental | null>(null);
   382→   382→  const [cancelSuccess, setCancelSuccess] = useState(false);
   383→   383→
   384→   384→  // Fetch rentals as customer
   385→   385→  const { data: customerRentals = [], isLoading: loadingCustomer } = useQuery({
   386→   386→    queryKey: ['rentals', 'customer'],
   387→   387→    queryFn: async () => {
   388→   388→      const data = await api.getRentals({ role: 'customer' });
   389→   389→      return data as unknown as Rental[];
   390→   390→    },
   391→   391→    enabled: !!user,
   392→   392→  });
   393→   393→
   394→   394→  // Fetch rentals as owner
   395→   395→  const { data: ownerRentals = [], isLoading: loadingOwner } = useQuery({
   396→   396→    queryKey: ['rentals', 'owner'],
   397→   397→    queryFn: async () => {
   398→   398→      const data = await api.getRentals({ role: 'owner' });
   399→   399→      return data as unknown as Rental[];
   400→   400→    },
   401→   401→    enabled: !!user,
   402→   402→  });
   403→   403→
   404→   404→  // Merge and deduplicate
   405→   405→  const allRentals = (() => {
   406→   406→    const map = new Map<string, Rental>();
   407→   407→    for (const r of [...customerRentals, ...ownerRentals]) {
   408→   408→      map.set(r.id, r as Rental);
   409→   409→    }
   410→   410→    return Array.from(map.values()).sort(
   411→   411→      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
   412→   412→    );
   413→   413→  })();
   414→   414→
   415→   415→  const isLoading = loadingCustomer || loadingOwner;
   416→   416→
   417→   417→  // Filter by tab
   418→   418→  const filteredRentals = activeTab === 'all'
   419→   419→    ? allRentals
   420→   420→    : allRentals.filter((r) => statusToTab[r.status] === activeTab);
   421→   421→
   422→   422→  // Mutations
   423→   423→  const payMutation = useMutation({
   424→   424→    mutationFn: (id: string) => api.payRental(id),
   425→   425→    onSuccess: () => { toast.success('Payment successful!'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   426→   426→    onError: (e: Error) => toast.error(e.message),
   427→   427→  });
   428→   428→
   429→   429→  const cancelMutation = useMutation({
   430→   430→    mutationFn: ({ id, reason }: { id: string; reason?: string }) => api.cancelRental(id, reason),
   431→   431→    onSuccess: () => {
   432→   432→      setCancelSuccess(true);
   433→   433→      queryClient.invalidateQueries({ queryKey: ['rentals'] });
   434→   434→    },
   435→   435→    onError: (e: Error) => toast.error(e.message),
   436→   436→  });
   437→   437→
   438→   438→  const acceptMutation = useMutation({
   439→   439→    mutationFn: (id: string) => api.acceptRental(id),
   440→   440→    onSuccess: () => { toast.success('Rental accepted'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   441→   441→    onError: (e: Error) => toast.error(e.message),
   442→   442→  });
   443→   443→
   444→   444→  const rejectMutation = useMutation({
   445→   445→    mutationFn: (id: string) => api.rejectRental(id),
   446→   446→    onSuccess: () => { toast.success('Rental rejected'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   447→   447→    onError: (e: Error) => toast.error(e.message),
   448→   448→  });
   449→   449→
   450→   450→  const returnMutation = useMutation({
   451→   451→    mutationFn: (id: string) => api.returnRental(id),
   452→   452→    onSuccess: () => { toast.success('Return initiated'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   453→   453→    onError: (e: Error) => toast.error(e.message),
   454→   454→  });
   455→   455→
   456→   456→  const handleAction = useCallback((rental: Rental, action: string) => {
   457→   457→    switch (action) {
   458→   458→      case 'pay':
   459→   459→        setPaymentRental(rental);
   460→   460→        setPaymentModalOpen(true);
   461→   461→        break;
   462→   462→      case 'cancel':
   463→   463→        setCancelRental(rental);
   464→   464→        setCancelDialogOpen(true);
   465→   465→        break;
   466→   466→      case 'reject':
   467→   467→        rejectMutation.mutate(rental.id);
   468→   468→        break;
   469→   469→      case 'accept':
   470→   470→        acceptMutation.mutate(rental.id);
   471→   471→        break;
   472→   472→      case 'return':
   473→   473→        returnMutation.mutate(rental.id);
   474→   474→        break;
   475→   475→      case 'extend':
   476→   476→        setSelectedRental(rental);
   477→   477→        setDetailOpen(true);
   478→   478→        break;
   479→   479→      case 'confirmReturn':
   480→   480→        returnMutation.mutate(rental.id);
   481→   481→        break;
   482→   482→      case 'contact':
   483→   483→        toast.info('Message feature coming soon!');
   484→   484→        break;
   485→   485→      case 'rentAgain':
   486→   486→        navigate('product', { productId: rental.productId });
   487→   487→        break;
   488→   488→    }
   489→   489→  }, [payMutation, cancelMutation, acceptMutation, rejectMutation, returnMutation, navigate]);
   490→   490→
   491→   491→  const handleSelect = (rental: Rental) => {
   492→   492→    setSelectedRental(rental);
   493→   493→    setDetailOpen(true);
   494→   494→  };
   495→   495→
   496→   496→  // Tab counts
   497→   497→  const tabCounts: Record<TabKey, number> = {
   498→   498→    all: allRentals.length,
   499→   499→    upcoming: allRentals.filter((r) => statusToTab[r.status] === 'upcoming').length,
   500→   500→    active: allRentals.filter((r) => statusToTab[r.status] === 'active').length,
   501→   501→    returning: allRentals.filter((r) => statusToTab[r.status] === 'returning').length,
   502→   502→    completed: allRentals.filter((r) => statusToTab[r.status] === 'completed').length,
   503→   503→    cancelled: allRentals.filter((r) => statusToTab[r.status] === 'cancelled').length,
   504→   504→    overdue: allRentals.filter((r) => statusToTab[r.status] === 'overdue').length,
   505→   505→  };
   506→   506→
   507→   507→  return (
   508→   508→    <div className="min-h-screen bg-slate-50 pb-20 sm:pb-0">
   509→   509→      {/* Header */}
   510→   510→      <div className="bg-white border-b border-slate-200 sticky top-0 z-30">
   511→   511→        <div className="max-w-5xl mx-auto px-4 sm:px-6">
   512→   512→          <div className="flex items-center gap-3 h-16">
   513→   513→            <Button variant="ghost" size="sm" className="text-slate-600 hover:text-slate-900" onClick={() => navigate('dashboard')}>
   514→   514→              <ArrowLeft className="h-4 w-4 mr-1" />Back
   515→   515→            </Button>
   516→   516→            <div className="h-6 w-px bg-slate-200" />
   517→   517→            <h1 className="text-lg font-bold text-slate-900">My Rentals</h1>
   518→   518→            <span className="text-sm text-slate-400 ml-1">({allRentals.length})</span>
   519→   519→          </div>
   520→   520→        </div>
   521→   521→      </div>
   522→   522→
   523→   523→      <div className="max-w-5xl mx-auto px-4 sm:px-6 py-4 sm:py-6">
   524→   524→        {/* Tabs */}
   525→   525→        <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as TabKey)} className="mb-6">
   526→   526→          <TabsList className="bg-slate-100 p-1 h-auto flex-nowrap gap-1 overflow-x-auto scrollbar-none">
   527→   527→            {(Object.keys(tabCounts) as TabKey[]).map((tab) => (
   528→   528→              <TabsTrigger
   529→   529→                key={tab}
   530→   530→                value={tab}
   531→   531→                className="text-xs sm:text-sm px-3 sm:px-4 py-1.5 whitespace-nowrap data-[state=active]:bg-white data-[state=active]:text-slate-900 data-[state=active]:shadow-sm capitalize"
   532→   532→              >
   533→   533→                {tab}
   534→   534→                {tabCounts[tab] > 0 && (
   535→   535→                  <span className="ml-1.5 text-[10px] bg-slate-200 text-slate-600 rounded-full px-1.5 py-0.5 data-[state=active]:bg-emerald-100 data-[state=active]:text-emerald-700">
   536→   536→                    {tabCounts[tab]}
   537→   537→                  </span>
   538→   538→                )}
   539→   539→              </TabsTrigger>
   540→   540→            ))}
   541→   541→          </TabsList>
   542→   542→        </Tabs>
   543→   543→
   544→   544→        {/* Content */}
   545→   545→        {isLoading ? (
   546→   546→          <div className="space-y-3">
   547→   547→            {Array.from({ length: 4 }).map((_, i) => <RentalCardSkeleton key={i} />)}
   548→   548→          </div>
   549→   549→        ) : filteredRentals.length === 0 ? (
   550→   550→          <EmptyState tab={activeTab} />
   551→   551→        ) : (
   552→   552→          <div className="space-y-3 max-h-[calc(100vh-220px)] overflow-y-auto pr-1 pb-16 sm:pb-0">
   553→   553→            <AnimatePresence mode="popLayout">
   554→   554→              {filteredRentals.map((rental) => (
   555→   555→                <RentalCard
   556→   556→                  key={rental.id}
   557→   557→                  rental={rental as Rental}
   558→   558→                  onSelect={handleSelect}
   559→   559→                  onAction={handleAction}
   560→   560→                />
   561→   561→              ))}
   562→   562→            </AnimatePresence>
   563→   563→          </div>
   564→   564→        )}
   565→   565→      </div>
   566→   566→
   567→   567→      {/* Detail Dialog */}
   568→   568→      <RentalDetailDialog
   569→   569→        rentalId={selectedRental?.id ?? null}
   570→   570→        open={detailOpen}
   571→   571→        onClose={() => { setDetailOpen(false); setSelectedRental(null); }}
   572→   572→      />
   573→   573→      {/* Cancel Dialog */}
   574→   574→      <CancelRentalDialog
   575→   575→        open={cancelDialogOpen}
   576→   576→        onClose={() => { setCancelDialogOpen(false); setCancelRental(null); setCancelSuccess(false); }}
   577→   577→        onConfirm={(reason) => cancelMutation.mutate({ id: cancelRental!.id, reason })}
   578→   578→        rentalStatus={cancelRental?.status as RentalStatus}
   579→   579→        productName={cancelRental?.product?.title || ''}
   580→   580→        totalAmount={cancelRental?.totalAmount || 0}
   581→   581→        isPending={cancelMutation.isPending}
   582→   582→        isSuccess={cancelSuccess}
   583→   583→      />
   584→   584→      {/* Payment Modal */}
   585→   585→      <PaymentCheckoutModal
   586→   586→        open={paymentModalOpen}
   587→   587→        onClose={() => { setPaymentModalOpen(false); setPaymentRental(null); }}
   588→   588→        rentalId={paymentRental?.id || ''}
   589→   589→        rentalData={paymentRental ? {
   590→   590→          totalAmount: paymentRental.totalAmount,
   591→   591→          rentalAmount: paymentRental.rentalAmount,
   592→   592→          platformFee: paymentRental.platformFee,
   593→   593→          tax: paymentRental.tax,
   594→   594→          deliveryFee: paymentRental.deliveryFee,
   595→   595→          discount: paymentRental.discount,
   596→   596→          securityDeposit: paymentRental.securityDeposit,
   597→   597→          rentalDays: paymentRental.rentalDays,
   598→   598→          dailyRate: paymentRental.dailyRate,
   599→   599→          startDate: paymentRental.startDate,
   600→   600→          endDate: paymentRental.endDate,
   601→   601→          productTitle: paymentRental.product?.title,
   602→   602→        } : null}
   603→   603→        onSuccess={() => {
   604→   604→          setPaymentModalOpen(false);
   605→   605→          setPaymentRental(null);
   606→   606→          queryClient.invalidateQueries({ queryKey: ['rentals'] });
   607→   607→          toast.success('Payment successful! Rental request sent to owner.');
   608→   608→        }}
   609→   609→      />
   610→   610→    </div>
   611→   611→  );
   612→   612→}
