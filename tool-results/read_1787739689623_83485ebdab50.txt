    69→// ─── AuthModal Component ────────────────────────────────────
    70→export default function AuthModal() {
    71→  const {
    72→    authModalOpen,
    73→    setAuthModalOpen,
    74→    authModalView,
    75→    setAuthModalView,
    76→    setUser,
    77→    navigate,
    78→    returnUrl,
    79→    setReturnUrl,
    80→  } = useAppStore();
    81→
    82→  const [states, setStates] = useState<State[]>([]);
    83→  const [loadingStates, setLoadingStates] = useState(true);
    84→
    85→  useEffect(() => {
    86→    api.getStates()
    87→      .then((data) => {
    88→        const parsed = data as unknown as State[];
    89→        setStates(parsed.filter((s) => s.isActive));
    90→      })
    91→      .catch(() => toast.error('Failed to load states'))
    92→      .finally(() => setLoadingStates(false));
    93→  }, []);
    94→
    95→  const handleClose = useCallback(() => setAuthModalOpen(false), [setAuthModalOpen]);
    96→
    97→  const toggleView = useCallback(() => {
    98→    setAuthModalView(authModalView === 'login' ? 'register' : 'login');
    99→  }, [authModalView, setAuthModalView]);
   100→
   101→  const handleAuthSuccess = useCallback(async (userData: Record<string, unknown>) => {
   102→    try {
   103→      const meData = await api.me();
   104→      const userObj = (meData as Record<string, unknown>).user as unknown as UserType;
   105→      setUser(userObj);
   106→      toast.success(authModalView === 'login' ? 'Welcome back!' : 'Account created successfully!');
   107→      setAuthModalOpen(false);
   108→      setReturnUrl(null);
   109→      if (userObj.role === 'OWNER' && userObj.kycStatus !== 'VERIFIED') {
   110→        navigate('seller-kyc');
   111→      } else {
   112→        navigate(returnUrl ? (returnUrl as 'marketplace' | 'dashboard') : 'marketplace');
   113→      }
   114→    } catch {
   115→      setUser(userData as unknown as UserType);
   116→    }
   117→  }, [authModalView, returnUrl, navigate, setAuthModalOpen, setReturnUrl, setUser]);
   118→
   119→  return (
   120→    <Dialog open={authModalOpen} onOpenChange={(open) => { if (!open) handleClose(); }}>
   121→      <DialogContent className="sm:max-w-[440px] max-h-[92vh] p-0 overflow-hidden gap-0 rounded-3xl border-gray-200 flex flex-col">
   122→        <div className="p-6 sm:p-7 overflow-y-auto flex-1">
   123→          {authModalView === 'login' ? (
   124→            <LoginForm
   125→              onSuccess={handleAuthSuccess}
   126→              onToggleView={toggleView}
   127→            />
   128→          ) : (
   129→            <RegisterForm
   130→              onSuccess={handleAuthSuccess}
   131→              onToggleView={toggleView}
   132→              states={states}
   133→              loadingStates={loadingStates}
   134→            />
   135→          )}
   136→        </div>
   137→      </DialogContent>
   138→    </Dialog>
   139→  );
   140→}

   142→// ─── Login Form ─────────────────────────────────────────────
   143→function LoginForm({

   145→  onToggleView,

   147→  onSuccess: (data: Record<string, unknown>) => void;
   148→  onToggleView: () => void;

   150→  const [showPassword, setShowPassword] = useState(false);
   151→  const [serverError, setServerError] = useState('');
   152→  const [isGoogleLoading, setIsGoogleLoading] = useState(false);
   153→  const [showGoogleDialog, setShowGoogleDialog] = useState(false);
   154→  const [googleEmail, setGoogleEmail] = useState('');
   155→  const [googleName, setGoogleName] = useState('');
   156→  const [googleConfigured, setGoogleConfigured] = useState(false);
   157→  const [checkedConfig, setCheckedConfig] = useState(false);
   158→
   159→  const {
   160→    register,
   161→    handleSubmit,
   162→    formState: { errors, isSubmitting },
   163→  } = useForm<LoginFormData>({ resolver: zodResolver(loginSchema) });

   165→  useEffect(() => {
   166→    api.getGoogleConfig()
   167→      .then((c) => setGoogleConfigured(c.configured))
   168→      .catch(() => setGoogleConfigured(false))
   169→      .finally(() => setCheckedConfig(true));
   170→  }, []);
   171→
   172→  const handleGoogleClick = async () => {
   173→    if (googleConfigured) {
   174→      setIsGoogleLoading(true);
   175→      window.location.href = '/api/auth/google';
   176→    } else {
   177→      setGoogleEmail('');
   178→      setGoogleName('');
   179→      setShowGoogleDialog(true);
   180→    }
   181→  };
   182→
   183→  const handleDemoGoogleLogin = async () => {
   184→    if (!googleEmail.trim() || !googleEmail.includes('@')) {
   185→      toast.error('Please enter a valid email address');
   186→      return;
   187→    }
   188→    const emailLower = googleEmail.trim().toLowerCase();
   189→    if (!emailLower.endsWith('@gmail.com') && !emailLower.endsWith('@googlemail.com')) {
   190→      toast.error('Please enter a valid Gmail address (e.g., you@gmail.com)');
   191→      return;
   192→    }
   193→    setIsGoogleLoading(true);
   194→    try {
   195→      const name = googleName.trim() || emailLower.split('@')[0].replace(/[._]/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   196→      const result = await api.googleAuth({ email: emailLower, name });
   197→      setShowGoogleDialog(false);
   198→      onSuccess(result.user);
   199→    } catch (err) {
   200→      toast.error(err instanceof Error ? err.message : 'Google sign-in failed');
   201→    } finally {
   202→      setIsGoogleLoading(false);
   203→    }
   204→  };
   205→
   206→  const onSubmit = async (data: LoginFormData) => {
   207→    setServerError('');
   208→    try {
   209→      const res = await api.login({ email: data.email, password: data.password });
   210→      onSuccess(res.user);
   211→    } catch (err) {
   212→      const message = err instanceof Error ? err.message : 'Login failed';
   213→      setServerError(message);
   214→      toast.error(message);
   215→    }
   216→  };
   217→
   218→  return (
   219→    <>
   220→      <div>
   221→        {/* Heading */}
   222→        <h2 className="text-[24px] font-semibold text-[#0f172a] leading-tight mb-5">
   223→          Log in to discover more<br />rental deals just for you
   224→        </h2>
   225→
   226→        {serverError && (
   227→          <div className="mb-3 p-3 rounded-xl bg-red-50 border border-red-200 text-red-700 text-sm">
   228→            {serverError}
   229→          </div>
   230→        )}
   231→
   232→        <form onSubmit={handleSubmit(onSubmit)} className="space-y-3">
   233→          {/* Email */}
   234→          <div>
   235→            <Input
   236→              type="email"
   237→              placeholder="Email"
   238→              className={errors.email ? inputErrorClass : inputClass}
   239→              {...register('email')}
   240→            />
   241→            {errors.email && (
   242→              <p className="text-sm text-red-600 mt-1 ml-1">{errors.email.message}</p>
   243→            )}
   244→          </div>
   245→
   246→          {/* Password */}
   247→          <div>
   248→            <div className="relative">
   249→              <Input
   250→                type={showPassword ? 'text' : 'password'}
   251→                placeholder="Password"
   252→                className={`pr-12 ${errors.password ? inputErrorClass : inputClass}`}
   253→                {...register('password')}
   254→              />
   255→              <button
   256→                type="button"
   257→                onClick={() => setShowPassword(!showPassword)}
   258→                className="absolute right-4 top-1/2 -translate-y-1/2 text-gray-500 hover:text-gray-800 transition-colors p-1"
   259→                tabIndex={-1}
   260→              >
   261→                {showPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
   262→              </button>
   263→            </div>
   264→            {errors.password && (
   265→              <p className="text-sm text-red-600 mt-1 ml-1">{errors.password.message}</p>
   266→            )}
   267→          </div>
   268→
   269→          {/* Forgot password - left aligned */}
   270→          <div className="pt-0.5">
   271→            <button
   272→              type="button"
   273→              onClick={() => toast.info('Password reset link sent to your email!')}
   274→              className="text-[15px] text-[#0074e8] hover:text-[#0060b8] font-medium transition-colors"
   275→            >
   276→              Forgot password?
   277→            </button>
   278→          </div>
   279→
   280→          {/* Log in - Red pill button */}
   281→          <Button
   282→            type="submit"
   283→            disabled={isSubmitting}
   284→            className="w-full h-[52px] rounded-full bg-[#e60023] hover:bg-[#cc001f] active:bg-[#b3001b] text-white font-semibold text-[16px] transition-colors shadow-none"
   285→          >
   286→            {isSubmitting ? (
   287→              <>
   288→                <Loader2 className="w-5 h-5 animate-spin" />
   289→                Logging in...
   290→              </>
   291→            ) : (
   292→              'Log in'
   293→            )}
   294→          </Button>
   295→        </form>
   296→
   297→        {/* OR divider */}
   298→        <div className="relative my-5 flex items-center">
   299→          <div className="flex-1 border-t border-gray-300" />
   300→          <span className="px-4 text-sm font-medium text-[#0f172a]">OR</span>
   301→          <div className="flex-1 border-t border-gray-300" />
   302→        </div>
   303→
   304→        {/* Continue with Google */}
   305→        <button
   306→          type="button"
   307→          onClick={handleGoogleClick}
   308→          disabled={isGoogleLoading || !checkedConfig}
   309→          className="w-full h-[52px] rounded-2xl border border-gray-300 bg-white hover:bg-gray-50 active:bg-gray-100 flex items-center justify-center gap-2.5 text-[15px] font-medium text-[#0f172a] transition-colors disabled:opacity-60"
   310→        >
   311→          {isGoogleLoading ? (
   312→            <Loader2 className="w-5 h-5 animate-spin text-gray-500" />
   313→          ) : (
   314→            <GoogleIcon />
   315→          )}
   316→          {isGoogleLoading ? 'Signing in...' : 'Continue with Google'}
   317→        </button>
   318→
   319→        {/* Sign up link */}
   320→        <p className="mt-5 text-center text-sm text-[#0f172a]">
   321→          Not on RentCart yet?{' '}
   322→          <button
   323→            onClick={onToggleView}
   324→            className="text-[#0074e8] hover:text-[#0060b8] font-semibold transition-colors"
   325→          >
   326→            Sign up
   327→          </button>
   328→        </p>
   329→
   330→        {/* Terms */}
   331→        <p className="mt-3 text-center text-xs text-gray-500 leading-relaxed">
   332→          By continuing, you agree to RentCart&apos;s{' '}
   333→          <span className="underline cursor-pointer">Terms of Service</span>
   334→          {' '}and acknowledge our{' '}
   335→          <span className="underline cursor-pointer">Privacy Policy</span>.
   336→        </p>
   337→      </div>
   338→
   339→      {/* Google Dialog (Demo) */}
   340→      {showGoogleDialog && (
   341→        <div className="fixed inset-0 z-[200] bg-black/50 flex items-center justify-center p-4" onClick={() => !isGoogleLoading && setShowGoogleDialog(false)}>
   342→          <div className="bg-white rounded-3xl shadow-2xl w-full max-w-sm p-7" onClick={(e) => e.stopPropagation()}>
   343→            <div className="flex items-center justify-between mb-6">
   344→              <div className="flex items-center gap-3">
   345→                <GoogleIcon className="w-7 h-7" />
   346→                <div>
   347→                  <h3 className="font-bold text-[#0f172a] text-[17px]">Sign in with Google</h3>
   348→                  <p className="text-xs text-gray-500 mt-0.5">Enter your Gmail to continue</p>
   349→                </div>
   350→              </div>
   351→              <button onClick={() => setShowGoogleDialog(false)} disabled={isGoogleLoading} className="p-1.5 rounded-full hover:bg-gray-100 text-gray-400 hover:text-gray-600 transition-colors">
   352→                <X className="w-5 h-5" />
   353→              </button>
   354→            </div>

   356→              <div>
   357→                <label className="text-sm font-medium text-[#0f172a] mb-1.5 block">Gmail Address</label>

   366→              </div>
   367→              <Button onClick={handleDemoGoogleLogin} disabled={isGoogleLoading || !googleEmail.trim()} className="w-full h-[48px] rounded-full bg-[#e60023] hover:bg-[#cc001f] active:bg-[#b3001b] text-white font-semibold text-[15px] transition-colors shadow-none">
   368→                {isGoogleLoading ? (<><Loader2 className="w-5 h-5 animate-spin" /> Signing in...</>) : 'Continue'}
   369→              </Button>
   370→            </div>
   371→          </div>
   372→        </div>
   373→      )}
   374→    </>
   375→  );
   376→}

   378→// ─── Register Form ──────────────────────────────────────────
   379→function RegisterForm({
   380→  onSuccess,
   381→  onToggleView,
   382→  states,
   383→  loadingStates,
   384→}: {
   385→  onSuccess: (data: Record<string, unknown>) => void;
   386→  onToggleView: () => void;
   387→  states: State[];
   388→  loadingStates: boolean;
   389→}) {
   390→  const [showPassword, setShowPassword] = useState(false);
   391→  const [serverError, setServerError] = useState('');
   392→
   393→  const {
   394→    register,
   395→    handleSubmit,
   396→    control,
   397→    setValue,
   398→    formState: { errors, isSubmitting },
   399→  } = useForm<RegisterFormData>({
   400→    resolver: zodResolver(registerSchema),
   401→    defaultValues: {
   402→      name: '',
   403→      email: '',
   404→      phone: '',
   405→      password: '',
   406→      stateId: '',
   407→      cityId: '',
   408→      pinCode: '',
   409→      address: '',
   410→      agreeTerms: false as unknown as true,
   411→    },
   412→  });
   413→
   414→  const selectedStateId = useWatch({ control, name: 'stateId' });
   415→  const selectedState = states.find((s) => s.id === selectedStateId);
   416→  const cities = selectedState?.cities?.filter((c) => c.isActive) || [];
   417→
   418→  useEffect(() => { setValue('cityId', ''); }, [selectedStateId, setValue]);
   419→
   420→  const onSubmit = async (data: RegisterFormData) => {
   421→    setServerError('');
   422→    try {
   423→      const res = await api.register({
   424→        name: data.name,
   425→        email: data.email,
   426→        phone: data.phone || undefined,
   427→        password: data.password,
   428→        stateId: data.stateId,
   429→        cityId: data.cityId,
   430→        pinCode: data.pinCode || undefined,
   431→        address: data.address || undefined,
   432→      });
   433→      onSuccess(res.user);
   434→    } catch (err) {
   435→      const message = err instanceof Error ? err.message : 'Registration failed';
   436→      setServerError(message);
   437→      toast.error(message);
   438→    }
   439→  };
   440→
   441→  return (
   442→    <>
   443→      <div>
   444→        {/* Heading */}
   445→        <h2 className="text-[24px] font-semibold text-[#0f172a] leading-tight mb-1">
   446→          Welcome to RentCart
   447→        </h2>
   448→        <p className="text-sm text-gray-500 mb-5">
   449→          Create an account to start renting and saving
   450→        </p>
   451→
   452→        {serverError && (
   453→          <div className="mb-3 p-3 rounded-xl bg-red-50 border border-red-200 text-red-700 text-sm">
   454→            {serverError}
   455→          </div>
   456→        )}
   457→
   458→        <form onSubmit={handleSubmit(onSubmit)} className="space-y-2.5">
   459→          {/* Full Name */}
   460→          <div>
   461→            <Input
   462→              placeholder="Full Name"
   463→              className={errors.name ? inputErrorClass : inputClass}
   464→              {...register('name')}
   465→            />
   466→            {errors.name && <p className="text-sm text-red-600 mt-1 ml-1">{errors.name.message}</p>}
   467→          </div>
   468→
   469→          {/* Email */}
   470→          <div>
   471→            <Input
   472→              type="email"
   473→              placeholder="Email"
   474→              className={errors.email ? inputErrorClass : inputClass}
   475→              {...register('email')}
   476→            />
   477→            {errors.email && <p className="text-sm text-red-600 mt-1 ml-1">{errors.email.message}</p>}
   478→          </div>
   479→
   480→          {/* Phone */}
   481→          <div>
   482→            <div className="relative">
   483→              <Phone className="absolute left-4 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 pointer-events-none" />
   484→              <Input
   485→                type="tel"
   486→                placeholder="Phone Number"
   487→                className={`pl-10 ${errors.phone ? inputErrorClass : inputClass}`}
   488→                {...register('phone')}
   489→              />
   490→            </div>
   491→            {errors.phone && <p className="text-sm text-red-600 mt-1 ml-1">{errors.phone.message}</p>}
   492→          </div>
   493→
   494→          {/* Password */}
   495→          <div>
   496→            <div className="relative">
   497→              <Input
   498→                type={showPassword ? 'text' : 'password'}
   499→                placeholder="Password (min. 6 characters)"
   500→                className={`pr-12 ${errors.password ? inputErrorClass : inputClass}`}
   501→                {...register('password')}
   502→              />
   503→              <button
   504→                type="button"
   505→                onClick={() => setShowPassword(!showPassword)}
   506→                className="absolute right-4 top-1/2 -translate-y-1/2 text-gray-500 hover:text-gray-800 transition-colors p-1"
   507→                tabIndex={-1}
   508→              >
   509→                {showPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
   510→              </button>
   511→            </div>
   512→            {errors.password && <p className="text-sm text-red-600 mt-1 ml-1">{errors.password.message}</p>}
   513→          </div>
   514→
   515→          {/* State & City */}
   516→          <div className="grid grid-cols-2 gap-3">
   517→            <div>
   518→              <Controller
   519→                name="stateId"
