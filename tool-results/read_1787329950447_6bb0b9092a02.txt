   145→  onClose,

   147→  rentalData,
   148→  onSuccess,

   153→  const [upiId, setUpiId] = useState('');
   154→  const [cardNumber, setCardNumber] = useState('');
   155→  const [cardExpiry, setCardExpiry] = useState('');
   156→  const [cardCvv, setCardCvv] = useState('');
   157→  const [cardName, setCardName] = useState('');
   158→  const [selectedBank, setSelectedBank] = useState('');
   159→  const [selectedWallet, setSelectedWallet] = useState('');
   160→  const [processing, setProcessing] = useState(false);
   161→  const [step, setStep] = useState<'checkout' | 'processing' | 'success'>('checkout');
   162→  const [txnId, setTxnId] = useState('');
   163→  const [orderSummaryExpanded, setOrderSummaryExpanded] = useState(false);

   165→  const dateRange = useMemo(() => {

   175→  const formatExpiry = (value: string) => {
   176→    const v = value.replace(/\D/g, '').slice(0, 4);
   177→    if (v.length >= 3) return v.slice(0, 2) + '/' + v.slice(2);
   178→    return v;
   179→  };

   182→    return id.includes('@') && id.length >= 3;
   183→  };

   186→    const num = cardNumber.replace(/\s/g, '');

   195→      case 'card': return `Pay ${amount} via Card`;
   196→      case 'netbanking': return `Pay ${amount} via Net Banking`;
   197→      case 'wallet': return `Pay ${amount} via Wallet`;
   198→      case 'cash': return 'Confirm Cash on Pickup';

   210→      default: return false;
   211→    }
   212→  };

   218→    for (let i = 0; i < 12; i++) {
   219→      result += chars.charAt(Math.floor(Math.random() * chars.length));
   220→    }
   221→    return result;
   222→  };

   233→          paymentMethod: 'CASH_ON_PICKUP',

   238→        queryClient.invalidateQueries({ queryKey: ['dashboard'] });

   241→        setProcessing(false);
   242→        setStep('checkout');
   243→      }

   274→              paymentMethod: selectedMethod.toUpperCase(),

   282→              setProcessing(false);
   283→              setStep('checkout');
   284→            });

   287→          theme: { color: '#059669' },

   290→        const rzp = new window.Razorpay(options);

   292→          toast.error('Payment failed. Please try again.');
   293→          setProcessing(false);
   294→          setStep('checkout');

   302→          rentalId,
   303→          paymentMethod: selectedMethod.toUpperCase(),
   304→        });
   305→        setTxnId(generateTxnId());
   306→        setStep('success');
   307→        queryClient.invalidateQueries({ queryKey: ['rentals'] });
   308→        queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   309→      }
   310→    } catch (error) {
   311→      toast.error('Payment failed. Please try again.');
   312→      setProcessing(false);
   313→      setStep('checkout');
   314→    }
