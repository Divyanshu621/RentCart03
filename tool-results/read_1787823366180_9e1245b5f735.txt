   118→// ─── Countdown Timer ───────────────────────────────────────
   119→function CountdownTimer({ endDate }: { endDate: string }) {
   120→  const [timeLeft, setTimeLeft] = useState('');
   121→
   122→  useEffect(() => {
   123→    const update = () => {
   124→      const diff = new Date(endDate).getTime() - Date.now();
   125→      if (diff <= 0) {
   126→        setTimeLeft('Overdue');
   127→        return;
   128→      }
   129→      const days = Math.floor(diff / (1000 * 60 * 60 * 24));
   130→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
   131→      const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
   132→      const secs = Math.floor((diff % (1000 * 60)) / 1000);
   133→      if (days > 0) setTimeLeft(`${days}d ${hours}h ${mins}m`);
   134→      else if (hours > 0) setTimeLeft(`${hours}h ${mins}m ${secs}s`);
   135→      else setTimeLeft(`${mins}m ${secs}s`);
   136→    };
   137→    update();
   138→    const id = setInterval(update, 1000);
   139→    return () => clearInterval(id);
   140→  }, [endDate]);
   141→
   142→  return (
   143→    <span className={`text-xs font-mono font-medium ${timeLeft === 'Overdue' ? 'text-red-600' : 'text-emerald-600'}`}>
   144→      {timeLeft}
   145→    </span>
   146→  );
   147→}

   410→    return Array.from(map.values()).sort(
   411→      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
   412→    );

   434→    },
   435→    onError: (e: Error) => toast.error(e.message),
   436→  });
   437→
   438→  const acceptMutation = useMutation({
   439→    mutationFn: (id: string) => api.acceptRental(id),
   440→    onSuccess: () => { toast.success('Rental accepted'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   441→    onError: (e: Error) => toast.error(e.message),
   442→  });
   443→
   444→  const rejectMutation = useMutation({
   445→    mutationFn: (id: string) => api.rejectRental(id),
   446→    onSuccess: () => { toast.success('Rental rejected'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   447→    onError: (e: Error) => toast.error(e.message),
   448→  });
   449→
   450→  const returnMutation = useMutation({
   451→    mutationFn: (id: string) => api.returnRental(id),
   452→    onSuccess: () => { toast.success('Return initiated'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   453→    onError: (e: Error) => toast.error(e.message),
   454→  });
   455→
   456→  const handleAction = useCallback((rental: Rental, action: string) => {
   457→    switch (action) {
   458→      case 'pay':
   459→        setPaymentRental(rental);
   460→        setPaymentModalOpen(true);
   461→        break;
   462→      case 'cancel':
   463→        setCancelRental(rental);
   464→        setCancelDialogOpen(true);
   465→        break;
   466→      case 'reject':
   467→        rejectMutation.mutate(rental.id);
   468→        break;
   469→      case 'accept':
   470→        acceptMutation.mutate(rental.id);
   471→        break;
   472→      case 'return':
   473→        returnMutation.mutate(rental.id);
   474→        break;
   475→      case 'extend':
   476→        setSelectedRental(rental);
   477→        setDetailOpen(true);
   478→        break;
   479→      case 'confirmReturn':
   480→        returnMutation.mutate(rental.id);
   481→        break;
   482→      case 'contact':
   483→        toast.info('Message feature coming soon!');
   484→        break;
   485→      case 'rentAgain':
   486→        navigate('product', { productId: rental.productId });
   487→        break;
   488→    }
   489→  }, [payMutation, cancelMutation, acceptMutation, rejectMutation, returnMutation, navigate]);
   490→
   491→  const handleSelect = (rental: Rental) => {
   492→    setSelectedRental(rental);
   493→    setDetailOpen(true);
   494→  };
   495→
   496→  // Tab counts
   497→  const tabCounts: Record<TabKey, number> = {
   498→    all: allRentals.length,
   499→    upcoming: allRentals.filter((r) => statusToTab[r.status] === 'upcoming').length,
   500→    active: allRentals.filter((r) => statusToTab[r.status] === 'active').length,
   501→    returning: allRentals.filter((r) => statusToTab[r.status] === 'returning').length,
   502→    completed: allRentals.filter((r) => statusToTab[r.status] === 'completed').length,
   503→    cancelled: allRentals.filter((r) => statusToTab[r.status] === 'cancelled').length,
   504→    overdue: allRentals.filter((r) => statusToTab[r.status] === 'overdue').length,
   505→  };
   506→
   507→  return (
   508→    <div className="min-h-screen bg-slate-50 pb-20 sm:pb-0">
   509→      {/* Header */}
   510→      <div className="bg-white border-b border-slate-200 sticky top-0 z-30">
   511→        <div className="max-w-5xl mx-auto px-4 sm:px-6">
   512→          <div className="flex items-center gap-3 h-16">
   513→            <Button variant="ghost" size="sm" className="text-slate-600 hover:text-slate-900" onClick={() => navigate('dashboard')}>
   514→              <ArrowLeft className="h-4 w-4 mr-1" />Back
   515→            </Button>
   516→            <div className="h-6 w-px bg-slate-200" />
   517→            <h1 className="text-lg font-bold text-slate-900">My Rentals</h1>
   518→            <span className="text-sm text-slate-400 ml-1">({allRentals.length})</span>
   519→          </div>
   520→        </div>
   521→      </div>
   522→
   523→      <div className="max-w-5xl mx-auto px-4 sm:px-6 py-4 sm:py-6">
   524→        {/* Tabs */}
   525→        <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as TabKey)} className="mb-6">
   526→          <TabsList className="bg-slate-100 p-1 h-auto flex-nowrap gap-1 overflow-x-auto scrollbar-none">
   527→            {(Object.keys(tabCounts) as TabKey[]).map((tab) => (
   528→              <TabsTrigger
   529→                key={tab}
   530→                value={tab}
   531→                className="text-xs sm:text-sm px-3 sm:px-4 py-1.5 whitespace-nowrap data-[state=active]:bg-white data-[state=active]:text-slate-900 data-[state=active]:shadow-sm capitalize"
   532→              >
   533→                {tab}
   534→                {tabCounts[tab] > 0 && (
   535→                  <span className="ml-1.5 text-[10px] bg-slate-200 text-slate-600 rounded-full px-1.5 py-0.5 data-[state=active]:bg-emerald-100 data-[state=active]:text-emerald-700">
   536→                    {tabCounts[tab]}
   537→                  </span>
   538→                )}
   539→              </TabsTrigger>
   540→            ))}
   541→          </TabsList>
   542→        </Tabs>
   543→
   544→        {/* Content */}
   545→        {isLoading ? (
   546→          <div className="space-y-3">
   547→            {Array.from({ length: 4 }).map((_, i) => <RentalCardSkeleton key={i} />)}
   548→          </div>
   549→        ) : filteredRentals.length === 0 ? (
   550→          <EmptyState tab={activeTab} />
   551→        ) : (
   552→          <div className="space-y-3 max-h-[calc(100vh-220px)] overflow-y-auto pr-1 pb-16 sm:pb-0">
   553→            <AnimatePresence mode="popLayout">
   554→              {filteredRentals.map((rental) => (
   555→                <RentalCard
   556→                  key={rental.id}
   557→                  rental={rental as Rental}
   558→                  onSelect={handleSelect}
   559→                  onAction={handleAction}
   560→                />
   561→              ))}
   562→            </AnimatePresence>
   563→          </div>
   564→        )}
   565→      </div>
   566→
   567→      {/* Detail Dialog */}
   568→      <RentalDetailDialog
   569→        rentalId={selectedRental?.id ?? null}
   570→        open={detailOpen}
   571→        onClose={() => { setDetailOpen(false); setSelectedRental(null); }}
   572→      />
   573→      {/* Cancel Dialog */}
   574→      <CancelRentalDialog
   575→        open={cancelDialogOpen}
   576→        onClose={() => { setCancelDialogOpen(false); setCancelRental(null); setCancelSuccess(false); }}
   577→        onConfirm={(reason) => cancelMutation.mutate({ id: cancelRental!.id, reason })}
   578→        rentalStatus={cancelRental?.status as RentalStatus}
   579→        productName={cancelRental?.product?.title || ''}
   580→        totalAmount={cancelRental?.totalAmount || 0}
   581→        isPending={cancelMutation.isPending}
   582→        isSuccess={cancelSuccess}
   583→      />
   584→      {/* Payment Modal */}
   585→      <PaymentCheckoutModal
   586→        open={paymentModalOpen}
   587→        onClose={() => { setPaymentModalOpen(false); setPaymentRental(null); }}
   588→        rentalId={paymentRental?.id || ''}
   589→        rentalData={paymentRental ? {
   590→          totalAmount: paymentRental.totalAmount,
   591→          rentalAmount: paymentRental.rentalAmount,
   592→          platformFee: paymentRental.platformFee,
   593→          tax: paymentRental.tax,
   594→          deliveryFee: paymentRental.deliveryFee,
   595→          discount: paymentRental.discount,
   596→          securityDeposit: paymentRental.securityDeposit,
   597→          rentalDays: paymentRental.rentalDays,
   598→          dailyRate: paymentRental.dailyRate,
   599→          startDate: paymentRental.startDate,
   600→          endDate: paymentRental.endDate,
   601→          productTitle: paymentRental.product?.title,
   602→        } : null}
   603→        onSuccess={() => {
   604→          setPaymentModalOpen(false);
   605→          setPaymentRental(null);
   606→          queryClient.invalidateQueries({ queryKey: ['rentals'] });
   607→          toast.success('Payment successful! Rental request sent to owner.');
   608→        }}
   609→      />
   610→    </div>
   611→  );
   612→}
