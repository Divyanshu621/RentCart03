   342→    status: string, actualReturnDate?: Date,

   355→        customerId: data.customer.id,

   359→        endDate: data.endDate,
   360→        actualReturnDate: data.actualReturnDate ?? null,
   361→        rentalDays,
   362→        dailyRate,
   363→        rentalAmount,
   364→        securityDeposit,
   365→        platformFee,
   366→        tax,
   367→        totalAmount,
   368→        status: data.status,
   369→        cancellationReason: data.cancellationReason ?? null,
   370→      },
   371→    })
