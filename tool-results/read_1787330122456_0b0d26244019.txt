   306→        setStep('success');
   307→        queryClient.invalidateQueries({ queryKey: ['rentals'] });
   308→        queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   309→      }

   312→      setProcessing(false);
   313→      setStep('checkout');
   314→    }
