    35→// ─── Animated Counter ──────────────────────────────────────
    36→function AnimatedCounter({ target, prefix = '', suffix = '', decimals = 0 }: {
    37→  target: number; prefix?: string; suffix?: string; decimals?: number;
    38→}) {
    39→  const [count, setCount] = useState(0);
    40→  const ref = useRef<HTMLSpanElement>(null);
    41→  const started = useRef(false);
    42→
    43→  useEffect(() => {
    44→    const el = ref.current;
    45→    if (!el) return;
    46→    const observer = new IntersectionObserver(
    47→      ([entry]) => {
    48→        if (entry.isIntersecting && !started.current) {
    49→          started.current = true;
    50→          const duration = 1200;
    51→          const startTime = performance.now();
    52→          const animate = (now: number) => {
    53→            const elapsed = now - startTime;
    54→            const progress = Math.min(elapsed / duration, 1);
    55→            // ease-out cubic
    56→            const eased = 1 - Math.pow(1 - progress, 3);
    57→            setCount(eased * target);
    58→            if (progress < 1) requestAnimationFrame(animate);
    59→          };
    60→          requestAnimationFrame(animate);
    61→        }
    62→      },
    63→      { threshold: 0.1 }
    64→    );
    65→    observer.observe(el);
    66→    return () => observer.disconnect();
    67→  }, [target]);
    68→
    69→  const formatted = decimals > 0
    70→    ? count.toLocaleString('en-IN', { minimumFractionDigits: decimals, maximumFractionDigits: decimals })
    71→    : Math.round(count).toLocaleString('en-IN');
    72→
    73→  return <span ref={ref}>{prefix}{formatted}{suffix}</span>;
    74→}
