    44→// ─── Document Upload Component ───────────────────────────────
    45→function DocumentUpload({
    46→  label,
    47→  description,
    48→  value,
    49→  onChange,
    50→  required,
    51→  disabled,
    52→}: {
    53→  label: string;
    54→  description: string;
    55→  value?: string;
    56→  onChange: (base64: string) => void;
    57→  required?: boolean;
    58→  disabled?: boolean;
    59→}) {
    60→  const inputRef = useRef<HTMLInputElement>(null);
    61→  const [localPreview, setLocalPreview] = useState<string | null>(null);
    62→  const [dragOver, setDragOver] = useState(false);
    63→  const [showPreview, setShowPreview] = useState(false);
    64→
    65→  // Use local upload state, fallback to parent value
    66→  const preview = localPreview || (value || null);
    67→
    68→  const handleFile = useCallback((file: File) => {
    69→    if (!file.type.startsWith('image/')) {
    70→      toast.error('Please upload an image file (JPG, PNG)');
    71→      return;
    72→    }
    73→    if (file.size > 5 * 1024 * 1024) {
    74→      toast.error('File size must be less than 5MB');
    75→      return;
    76→    }
    77→    const reader = new FileReader();
    78→    reader.onload = (e) => {
    79→      const result = e.target?.result as string;
    80→      setLocalPreview(result);
    81→      onChange(result);
    82→    };
    83→    reader.readAsDataURL(file);
    84→  }, [onChange]);
    85→
    86→  const handleDrop = useCallback((e: React.DragEvent) => {
    87→    e.preventDefault();
    88→    setDragOver(false);
    89→    const file = e.dataTransfer.files[0];
    90→    if (file) handleFile(file);
    91→  }, [handleFile]);
    92→
    93→  if (preview) {
    94→    return (
    95→      <div className="relative">
    96→        <div
    97→          className="relative rounded-xl overflow-hidden border-2 border-emerald-200 bg-emerald-50/50 cursor-pointer group"
    98→          onClick={() => !disabled && setShowPreview(!showPreview)}
    99→        >
   100→          <div className="flex items-center gap-3 p-3">
   101→            <div className="w-16 h-16 rounded-lg overflow-hidden bg-gray-100 shrink-0">
   102→              <img src={preview} alt={label} className="w-full h-full object-cover" />
   103→            </div>
   104→            <div className="flex-1 min-w-0">
   105→              <p className="text-sm font-medium text-emerald-800">{label}</p>
   106→              <p className="text-xs text-emerald-600 mt-0.5">Document uploaded successfully</p>
   107→            </div>
   108→            <CheckCircle2 className="w-5 h-5 text-emerald-500 shrink-0" />
   109→          </div>
   110→        </div>
   111→        {!disabled && (
   112→          <button
   113→            onClick={(e) => {
   114→              e.stopPropagation();
   115→              setLocalPreview(null);
   116→              onChange('');
   117→            }}
   118→            className="absolute -top-2 -right-2 w-6 h-6 bg-red-500 text-white rounded-full flex items-center justify-center shadow-sm hover:bg-red-600 transition-colors"
   119→          >
   120→            <X className="w-3 h-3" />
   121→          </button>
   122→        )}
   123→        {showPreview && !disabled && (
   124→          <div
   125→            className="fixed inset-0 z-[100] bg-black/80 flex items-center justify-center p-4"
   126→            onClick={() => setShowPreview(false)}
   127→          >
   128→            <div className="relative max-w-lg w-full" onClick={(e) => e.stopPropagation()}>
   129→              <img src={preview} alt={label} className="w-full rounded-xl" />
   130→              <button
   131→                onClick={() => setShowPreview(false)}
   132→                className="absolute -top-3 -right-3 w-8 h-8 bg-white rounded-full flex items-center justify-center shadow-lg"
   133→              >
   134→                <X className="w-4 h-4" />
   135→              </button>
   136→            </div>
   137→          </div>
   138→        )}
   139→      </div>
   140→    );
   141→  }
   142→
   143→  return (
   144→    <div
   145→      className={`relative border-2 border-dashed rounded-xl p-4 text-center transition-all ${
   146→        dragOver ? 'border-emerald-500 bg-emerald-50' : disabled ? 'border-gray-200 bg-gray-50' : 'border-gray-300 bg-gray-50/50 hover:border-emerald-400 hover:bg-emerald-50/30'
   147→      } ${required && !value ? 'border-amber-300' : ''}`}
   148→      onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
   149→      onDragLeave={() => setDragOver(false)}
   150→      onDrop={handleDrop}
   151→      onClick={() => !disabled && inputRef.current?.click()}
   152→    >
   153→      <input
   154→        ref={inputRef}
   155→        type="file"
   156→        accept="image/*"
   157→        capture="environment"
   158→        className="hidden"
   159→        onChange={(e) => {
   160→          const file = e.target.files?.[0];
   161→          if (file) handleFile(file);
   162→          e.target.value = '';
   163→        }}
   164→        disabled={disabled}
   165→      />
   166→      <div className="flex flex-col items-center gap-2">
   167→        <div className={`w-10 h-10 rounded-full flex items-center justify-center ${required && !value ? 'bg-amber-100' : 'bg-gray-100'}`}>
   168→          <Camera className={`w-5 h-5 ${required && !value ? 'text-amber-600' : 'text-gray-400'}`} />
   169→        </div>
   170→        <div>
   171→          <p className="text-sm font-medium text-[#0f172a]">{label} {required && <span className="text-red-500">*</span>}</p>
   172→          <p className="text-xs text-muted-foreground mt-0.5">{description}</p>
   173→        </div>
   174→        <div className="flex items-center gap-1.5 text-xs text-muted-foreground mt-1">
   175→          <ImageIcon className="w-3.5 h-3.5" />
   176→          <span>JPG, PNG up to 5MB</span>
   177→        </div>
   178→      </div>
   179→    </div>
   180→  );
   181→}

   248→// ─── Steps Indicator ─────────────────────────────────────────
   249→function StepsIndicator({ currentStep }: { currentStep: number }) {
   250→  const steps = ['Identity', 'Bank Details', 'Business Info'];
   251→  return (
   252→    <div className="flex items-center gap-1 sm:gap-2">
   253→      {steps.map((label, i) => (
   254→        <div key={label} className="flex items-center gap-1 sm:gap-2">
   255→          <div className="flex items-center gap-1.5">
   256→            <div
   257→              className={`w-6 h-6 sm:w-7 sm:h-7 rounded-full flex items-center justify-center text-xs font-bold transition-all ${
   258→                i < currentStep
   259→                  ? 'bg-emerald-500 text-white'
   260→                  : i === currentStep
   261→                  ? 'bg-emerald-100 text-emerald-700 ring-2 ring-emerald-500'
   262→                  : 'bg-gray-100 text-gray-400'
   263→              }`}
   264→            >
   265→              {i < currentStep ? <CheckCircle2 className="w-3.5 h-3.5" /> : i + 1}
   266→            </div>
   267→            <span className={`text-xs sm:text-sm font-medium hidden sm:block ${i <= currentStep ? 'text-[#0f172a]' : 'text-gray-400'}`}>
   268→              {label}
   269→            </span>
   270→          </div>
   271→          {i < steps.length - 1 && (
   272→            <ChevronRight className={`w-4 h-4 ${i < currentStep ? 'text-emerald-400' : 'text-gray-300'}`} />
   273→          )}
   274→        </div>
   275→      ))}
   276→    </div>
   277→  );
   278→}

   280→// ─── Main Component ──────────────────────────────────────────
   281→export default function SellerKycPage() {
   282→  const { user, navigate, setUser } = useAppStore();
   283→  const [kyc, setKyc] = useState<SellerKycType | null>(null);
   284→  const [loading, setLoading] = useState(true);
   285→  const [submitting, setSubmitting] = useState(false);
   286→  const [currentStep, setCurrentStep] = useState(0);
   287→
   288→  // Form state
   289→  const [aadhaarNumber, setAadhaarNumber] = useState('');
   290→  const [panNumber, setPanNumber] = useState('');
   291→  const [gstNumber, setGstNumber] = useState('');
   292→  const [aadhaarFront, setAadhaarFront] = useState('');
   293→  const [aadhaarBack, setAadhaarBack] = useState('');
   294→  const [panCard, setPanCard] = useState('');
   295→  const [bankAccountNo, setBankAccountNo] = useState('');
   296→  const [bankIfsc, setBankIfsc] = useState('');
   297→  const [bankName, setBankName] = useState('');
   298→  const [bankHolderName, setBankHolderName] = useState('');
   299→  const [passbook, setPassbook] = useState('');
   300→  const [businessName, setBusinessName] = useState('');
   301→  const [businessType, setBusinessType] = useState<BusinessType | ''>('');
   302→  const [businessAddress, setBusinessAddress] = useState('');
   303→
   304→  const isLocked = kyc?.status === 'SUBMITTED' || kyc?.status === 'UNDER_REVIEW' || kyc?.status === 'VERIFIED';
   305→  const canEdit = kyc?.status === 'DRAFT' || kyc?.status === 'REJECTED';
   306→
   307→  // Fetch KYC data
   308→  useEffect(() => {
   309→    const fetchKyc = async () => {
   310→      try {
   311→        const data = await api.getKycStatus();
   312→        if (data.kyc) {
   313→          const k = data.kyc as unknown as SellerKycType;
   314→          setKyc(k);
   315→          setAadhaarNumber(k.aadhaarNumber || '');
   316→          setPanNumber(k.panNumber || '');
   317→          setGstNumber(k.gstNumber || '');
   318→          setAadhaarFront(k.aadhaarFrontUrl || '');
   319→          setAadhaarBack(k.aadhaarBackUrl || '');
   320→          setPanCard(k.panCardUrl || '');
   321→          setBankAccountNo(k.bankAccountNo || '');
   322→          setBankIfsc(k.bankIfsc || '');
   323→          setBankName(k.bankName || '');
   324→          setBankHolderName(k.bankHolderName || '');
   325→          setPassbook(k.passbookUrl || '');
   326→          setBusinessName(k.businessName || '');
   327→          setBusinessType(k.businessType || '');
   328→          setBusinessAddress(k.businessAddress || '');
   329→
   330→          // Set step based on progress
   331→          if (k.aadhaarNumber && k.panNumber && k.aadhaarFrontUrl) setCurrentStep(1);
   332→          if (k.bankAccountNo && k.bankIfsc) setCurrentStep(2);
   333→        }
   334→      } catch {
   335→        toast.error('Failed to load KYC status');
   336→      } finally {
   337→        setLoading(false);
   338→      }
   339→    };
   340→    fetchKyc();
   341→  }, []);
   342→
   343→  // Calculate completion progress
   344→  const getProgress = () => {
   345→    let filled = 0;
   346→    const total = 8;
   347→    if (aadhaarNumber) filled++;
   348→    if (panNumber) filled++;
   349→    if (aadhaarFront) filled++;
   350→    if (aadhaarBack) filled++;
   351→    if (panCard) filled++;
   352→    if (bankAccountNo) filled++;
   353→    if (bankIfsc) filled++;
   354→    if (bankHolderName) filled++;
   355→    return Math.round((filled / total) * 100);
   356→  };
   357→
   358→  const saveDraft = useCallback(async () => {
   359→    if (!canEdit) return;
   360→    try {
   361→      await api.saveKycDraft({
   362→        aadhaarNumber, panNumber, gstNumber,
   363→        aadhaarFrontUrl: aadhaarFront, aadhaarBackUrl: aadhaarBack, panCardUrl: panCard,
   364→        bankAccountNo, bankIfsc, bankName, bankHolderName, passbookUrl: passbook,
   365→        businessName, businessType, businessAddress,
   366→      });
   367→    } catch {
   368→      // Silent fail for draft save
   369→    }
   370→  }, [canEdit, aadhaarNumber, panNumber, gstNumber, aadhaarFront, aadhaarBack, panCard, bankAccountNo, bankIfsc, bankName, bankHolderName, passbook, businessName, businessType, businessAddress]);
   371→
   372→  // Client-side validation before submitting
   373→  const validateForm = (): string | null => {
   374→    const digits = aadhaarNumber.replace(/\D/g, '');
   375→    if (!digits || digits.length !== 12) return 'Please enter a valid 12-digit Aadhaar number';
   376→    if (!panNumber || !/^[A-Z]{5}[0-9]{4}[A-Z]$/.test(panNumber)) return 'Please enter a valid PAN number (e.g., ABCDE1234F)';
   377→    if (!aadhaarFront) return 'Please upload your Aadhaar card (front)';
   378→    if (!aadhaarBack) return 'Please upload your Aadhaar card (back)';
   379→    if (!panCard) return 'Please upload your PAN card';
   380→    if (!bankHolderName || !bankHolderName.trim()) return 'Please enter the account holder name';
   381→    if (!bankAccountNo || bankAccountNo.length < 8) return 'Please enter a valid bank account number';
   382→    if (!bankIfsc || !/^[A-Z]{4}0[A-Z0-9]{6}$/.test(bankIfsc)) return 'Please enter a valid IFSC code (e.g., SBIN0001234)';
   383→    return null;
   384→  };
   385→
   386→  const scrollToStep = (stepIndex: number) => {
   387→    setCurrentStep(stepIndex);
   388→    const cardEls = document.querySelectorAll('[data-kyc-step]');
   389→    if (cardEls[stepIndex]) {
   390→      cardEls[stepIndex].scrollIntoView({ behavior: 'smooth', block: 'center' });
   391→    }
   392→  };
   393→
   394→  const hasRequiredFields = !!(aadhaarNumber.replace(/\D/g, '').length === 12 && panNumber && aadhaarFront && aadhaarBack && panCard && bankHolderName?.trim() && bankAccountNo?.length >= 8 && bankIfsc);
   395→
   396→  const handleSubmit = async () => {
   397→    // Validate and scroll to first invalid step
   398→    const error = validateForm();
   399→    if (error) {
   400→      // Determine which step has the error
   401→      const digits = aadhaarNumber.replace(/\D/g, '');
   402→      const panOk = panNumber && /^[A-Z]{5}[0-9]{4}[A-Z]$/.test(panNumber);
   403→      const docsOk = aadhaarFront && aadhaarBack && panCard;
   404→      const bankOk = bankHolderName?.trim() && bankAccountNo?.length >= 8 && bankIfsc && /^[A-Z]{4}0[A-Z0-9]{6}$/.test(bankIfsc);
   405→
   406→      if (!digits || !panOk || !docsOk) scrollToStep(0);
   407→      else if (!bankOk) scrollToStep(1);
   408→
   409→      toast.error(error);
   410→      return;
   411→    }
   412→
   413→    setSubmitting(true);
   414→    try {
   415→      await api.submitKyc({
   416→        aadhaarNumber: digits.replace(/(\d{4})(?=\d)/g, '$1 '), panNumber, gstNumber,
   417→        aadhaarFrontUrl: aadhaarFront, aadhaarBackUrl: aadhaarBack, panCardUrl: panCard,
   418→        bankAccountNo, bankIfsc, bankName, bankHolderName, passbookUrl: passbook,
   419→        businessName, businessType: businessType || undefined, businessAddress,
   420→      });
   421→      toast.success('KYC documents submitted successfully!');
   422→      // Refresh user data
   423→      try {
   424→        const meData = await api.me();
   425→        setUser((meData as Record<string, unknown>).user as unknown as UserType);
   426→      } catch {}
   427→      // Refresh KYC
   428→      const data = await api.getKycStatus();
   429→      if (data.kyc) setKyc(data.kyc as unknown as SellerKycType);
   430→    } catch (err) {
   431→      const msg = err instanceof Error ? err.message : 'Submission failed';
   432→      toast.error(msg);
   433→    } finally {
   434→      setSubmitting(false);
   435→    }
   436→  };
   437→
   438→  const formatAadhaar = (val: string) => {
   439→    const digits = val.replace(/\D/g, '').slice(0, 12);
   440→    return digits.replace(/(\d{4})(?=\d)/g, '$1 ');
   441→  };
   442→
   443→  const formatPan = (val: string) => {
   444→    return val.toUpperCase().slice(0, 10).replace(/[^A-Z0-9]/g, '');
   445→  };
   446→
   447→  if (loading) {
   448→    return (
   449→      <div className="flex items-center justify-center min-h-[60vh]">
   450→        <Loader2 className="w-8 h-8 animate-spin text-emerald-500" />
   451→      </div>
   452→    );
   453→  }
   454→
   455→  return (
   456→    <div className="max-w-3xl mx-auto px-4 sm:px-6 py-4 sm:py-6 pb-24 md:pb-6">
   457→      {/* Header */}
   458→      <div className="flex items-center gap-3 mb-4 sm:mb-6">
   459→        <button
   460→          onClick={() => navigate('dashboard')}
   461→          className="w-9 h-9 rounded-lg bg-white border border-gray-200 flex items-center justify-center hover:bg-gray-50 transition-colors"
   462→        >
   463→          <ArrowLeft className="w-4 h-4" />
   464→        </button>
   465→        <div>
   466→          <h1 className="text-xl sm:text-2xl font-bold text-[#0f172a]">Seller Verification</h1>
   467→          <p className="text-xs sm:text-sm text-muted-foreground">Complete KYC to start listing items</p>
   468→        </div>
   469→      </div>
   470→
   471→      {/* Status Banner */}
   472→      {kyc && <StatusBanner status={kyc.status} rejectionReason={kyc.rejectionReason || undefined} />}
   473→
   474→      {/* Progress */}
   475→      {canEdit && (
   476→        <div className="mt-4">
   477→          <div className="flex items-center justify-between mb-2">
   478→            <span className="text-xs font-medium text-muted-foreground">Completion Progress</span>
   479→            <span className="text-xs font-bold text-emerald-600">{getProgress()}%</span>
   480→          </div>
   481→          <Progress value={getProgress()} className="h-2" />
   482→        </div>
   483→      )}
   484→
   485→      {/* Steps */}
   486→      {canEdit && (
   487→        <div className="mt-4 p-3 bg-white rounded-xl border border-gray-200">
   488→          <StepsIndicator currentStep={currentStep} />
   489→        </div>
   490→      )}
   491→
   492→      {/* Step Content */}
   493→      <div className="mt-4 sm:mt-6 space-y-4 sm:space-y-6">
   494→        {/* Step 1: Identity Documents */}
   495→        {(canEdit || isLocked) && (
   496→          <Card data-kyc-step="0" className="border-gray-200">
   497→            <CardHeader className="pb-3">
   498→              <div className="flex items-center gap-2">
   499→                <div className="w-8 h-8 rounded-lg bg-blue-50 flex items-center justify-center">
   500→                  <Shield className="w-4 h-4 text-blue-600" />
   501→                </div>
