   339→  // Helper to create rental
   340→  async function createRental(data: {
   341→    customer: any, owner: any, product: any,
   342→    startDate: Date, endDate: Date,
   343→    status: string, actualReturnDate?: Date,
   344→    cancellationReason?: string,
   345→  }) {
   346→    const rentalDays = Math.ceil((data.endDate.getTime() - data.startDate.getTime()) / 86400000)
   347→    const dailyRate = data.product.dailyPrice
   348→    const rentalAmount = rentalDays * dailyRate
   349→    const securityDeposit = data.product.securityDeposit
   350→    const platformFee = Math.round(rentalAmount * 0.1 * 100) / 100
   351→    const tax = Math.round(rentalAmount * 0.18 * 100) / 100
   352→    const totalAmount = rentalAmount + platformFee + tax + securityDeposit
   353→
   354→    const rental = await db.rental.create({
   355→      data: {
   356→        customerId: data.customer.id,
   357→        ownerId: data.owner.id,
   358→        productId: data.product.id,
   359→        startDate: data.startDate,
   360→        endDate: data.endDate,
   361→        actualReturnDate: data.actualReturnDate ?? null,
   362→        rentalDays,
   363→        dailyRate,
   364→        rentalAmount,
   365→        securityDeposit,
   366→        platformFee,
   367→        tax,
   368→        totalAmount,
   369→        status: data.status,
   370→        cancellationReason: data.cancellationReason ?? null,
   371→      },
   372→    })
   373→    rentals.push(rental)
   374→    return rental
   375→  }
