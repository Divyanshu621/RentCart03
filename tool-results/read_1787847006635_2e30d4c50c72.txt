   118→// ─── Countdown Timer ───────────────────────────────────────
   119→function CountdownTimer({ endDate }: { endDate: string }) {
   120→  const [timeLeft, setTimeLeft] = useState('');
   121→
   122→  useEffect(() => {
   123→    const update = () => {
   124→      const diff = new Date(endDate).getTime() - Date.now();
   125→      if (diff <= 0) {
   126→        setTimeLeft('Overdue');
   127→        return;
   128→      }
   129→      const days = Math.floor(diff / (1000 * 60 * 60 * 24));
   130→      const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
   131→      const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
   132→      const secs = Math.floor((diff % (1000 * 60)) / 1000);
   133→      if (days > 0) setTimeLeft(`${days}d ${hours}h ${mins}m`);
   134→      else if (hours > 0) setTimeLeft(`${hours}h ${mins}m ${secs}s`);
   135→      else setTimeLeft(`${mins}m ${secs}s`);
   136→    };
   137→    update();
   138→    const id = setInterval(update, 1000);
   139→    return () => clearInterval(id);
   140→  }, [endDate]);
   141→
   142→  return (
   143→    <span className={`text-xs font-mono font-medium ${timeLeft === 'Overdue' ? 'text-red-600' : 'text-emerald-600'}`}>
   144→      {timeLeft}
   145→    </span>
   146→  );
   147→}

   389→    queryKey: ['rentals', 'customer'],
   390→    queryFn: async () => {
   391→      const data = await api.getRentals({ role: 'customer' });

   399→    queryKey: ['rentals', 'owner'],
   400→    queryFn: async () => {
   401→      const data = await api.getRentals({ role: 'owner' });

   404→    enabled: !!user,
   405→  });

   407→  // Merge and deduplicate
   408→  const allRentals = (() => {
   409→    const map = new Map<string, Rental>();
   410→    for (const r of [...customerRentals, ...ownerRentals]) {
   411→      map.set(r.id, r as Rental);
   412→    }
   413→    return Array.from(map.values()).sort(
   414→      (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime(),
   415→    );
   416→  })();
   417→
   418→  const isLoading = loadingCustomer || loadingOwner;
   419→
   420→  // Filter by tab
   421→  const filteredRentals = activeTab === 'all'
   422→    ? allRentals
   423→    : allRentals.filter((r) => statusToTab[r.status] === activeTab);
   424→
   425→  // Mutations
   426→  const payMutation = useMutation({
   427→    mutationFn: (id: string) => api.payRental(id),
   428→    onSuccess: () => { toast.success('Payment successful!'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   429→    onError: (e: Error) => toast.error(e.message),
   430→  });
   431→
   432→  const cancelMutation = useMutation({
   433→    mutationFn: ({ id, reason }: { id: string; reason?: string }) => api.cancelRental(id, reason),
   434→    onSuccess: () => {
   435→      setCancelSuccess(true);
   436→      queryClient.invalidateQueries({ queryKey: ['rentals'] });
   437→    },
   438→    onError: (e: Error) => toast.error(e.message),
   439→  });
   440→
   441→  const acceptMutation = useMutation({
   442→    mutationFn: (id: string) => api.acceptRental(id),
   443→    onSuccess: () => { toast.success('Rental accepted'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   444→    onError: (e: Error) => toast.error(e.message),
   445→  });
   446→
   447→  const rejectMutation = useMutation({
   448→    mutationFn: (id: string) => api.rejectRental(id),
   449→    onSuccess: () => { toast.success('Rental rejected'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   450→    onError: (e: Error) => toast.error(e.message),
   451→  });
   452→
   453→  const returnMutation = useMutation({
   454→    mutationFn: (id: string) => api.returnRental(id),
   455→    onSuccess: () => { toast.success('Return initiated'); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   456→    onError: (e: Error) => toast.error(e.message),
   457→  });
   458→
   459→  const handleAction = useCallback((rental: Rental, action: string) => {
   460→    switch (action) {
   461→      case 'pay':
   462→        setPaymentRental(rental);
   463→        setPaymentModalOpen(true);
   464→        break;
   465→      case 'cancel':
   466→        setCancelRental(rental);
   467→        setCancelDialogOpen(true);
   468→        break;
   469→      case 'reject':
   470→        rejectMutation.mutate(rental.id);
   471→        break;
   472→      case 'accept':
   473→        acceptMutation.mutate(rental.id);
   474→        break;
   475→      case 'return':
   476→        returnMutation.mutate(rental.id);
   477→        break;
   478→      case 'extend':
   479→        setSelectedRental(rental);
   480→        setDetailOpen(true);
   481→        break;
   482→      case 'confirmReturn':
   483→        returnMutation.mutate(rental.id);
   484→        break;
   485→      case 'contact':
   486→        navigate('messages');
   487→        break;
   488→      case 'rentAgain':
   489→        navigate('product', { productId: rental.productId });
   490→        break;
   491→    }
   492→  }, [payMutation, cancelMutation, acceptMutation, rejectMutation, returnMutation, navigate]);
   493→
   494→  const handleSelect = (rental: Rental) => {
   495→    setSelectedRental(rental);
   496→    setDetailOpen(true);
   497→  };
   498→
   499→  // Tab counts
   500→  const tabCounts: Record<TabKey, number> = {
   501→    all: allRentals.length,
   502→    upcoming: allRentals.filter((r) => statusToTab[r.status] === 'upcoming').length,
   503→    active: allRentals.filter((r) => statusToTab[r.status] === 'active').length,
   504→    returning: allRentals.filter((r) => statusToTab[r.status] === 'returning').length,
   505→    completed: allRentals.filter((r) => statusToTab[r.status] === 'completed').length,
   506→    cancelled: allRentals.filter((r) => statusToTab[r.status] === 'cancelled').length,
   507→    overdue: allRentals.filter((r) => statusToTab[r.status] === 'overdue').length,
   508→  };
   509→
   510→  return (
   511→    <div className="min-h-screen bg-slate-50 pb-20 sm:pb-0">
   512→      {/* Header */}
   513→      <div className="bg-white border-b border-slate-200 sticky top-0 z-30">
   514→        <div className="max-w-5xl mx-auto px-4 sm:px-6">
   515→          <div className="flex items-center gap-3 h-16">
   516→            <Button variant="ghost" size="sm" className="text-slate-600 hover:text-slate-900" onClick={() => navigate('dashboard')}>
   517→              <ArrowLeft className="h-4 w-4 mr-1" />Back
   518→            </Button>
   519→            <div className="h-6 w-px bg-slate-200" />
   520→            <h1 className="text-lg font-bold text-slate-900">My Rentals</h1>
   521→            <span className="text-sm text-slate-400 ml-1">({allRentals.length})</span>
   522→          </div>
   523→        </div>
   524→      </div>
   525→
   526→      <div className="max-w-5xl mx-auto px-4 sm:px-6 py-4 sm:py-6">
   527→        {/* Tabs */}
   528→        <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as TabKey)} className="mb-6">
   529→          <TabsList className="bg-slate-100 p-1 h-auto flex-nowrap gap-1 overflow-x-auto scrollbar-none">
   530→            {(Object.keys(tabCounts) as TabKey[]).map((tab) => (
   531→              <TabsTrigger
   532→                key={tab}
   533→                value={tab}
   534→                className="text-xs sm:text-sm px-3 sm:px-4 py-1.5 whitespace-nowrap data-[state=active]:bg-white data-[state=active]:text-slate-900 data-[state=active]:shadow-sm capitalize"
   535→              >
   536→                {tab}
   537→                {tabCounts[tab] > 0 && (
   538→                  <span className="ml-1.5 text-[10px] bg-slate-200 text-slate-600 rounded-full px-1.5 py-0.5 data-[state=active]:bg-emerald-100 data-[state=active]:text-emerald-700">
   539→                    {tabCounts[tab]}
   540→                  </span>
   541→                )}
   542→              </TabsTrigger>
   543→            ))}
   544→          </TabsList>
   545→        </Tabs>
   546→
   547→        {/* Content */}
   548→        {isLoading ? (
   549→          <div className="space-y-3">
   550→            {Array.from({ length: 4 }).map((_, i) => <RentalCardSkeleton key={i} />)}
   551→          </div>
   552→        ) : filteredRentals.length === 0 ? (
   553→          <EmptyState tab={activeTab} />
   554→        ) : (
   555→          <div className="space-y-3 max-h-[calc(100vh-220px)] overflow-y-auto pr-1 pb-16 sm:pb-0">
   556→            <AnimatePresence mode="popLayout">
   557→              {filteredRentals.map((rental) => (
   558→                <RentalCard
   559→                  key={rental.id}
   560→                  rental={rental as Rental}
   561→                  onSelect={handleSelect}
   562→                  onAction={handleAction}
   563→                  isAccepting={acceptMutation.isPending}
   564→                />
   565→              ))}
   566→            </AnimatePresence>
   567→          </div>
   568→        )}
   569→      </div>
   570→
   571→      {/* Detail Dialog */}
   572→      <RentalDetailDialog
   573→        rentalId={selectedRental?.id ?? null}
   574→        open={detailOpen}
   575→        onClose={() => { setDetailOpen(false); setSelectedRental(null); }}
   576→      />
   577→      {/* Cancel Dialog */}
   578→      <CancelRentalDialog
   579→        open={cancelDialogOpen}
   580→        onClose={() => { setCancelDialogOpen(false); setCancelRental(null); setCancelSuccess(false); }}
   581→        onConfirm={(reason) => cancelMutation.mutate({ id: cancelRental!.id, reason })}
   582→        rentalStatus={cancelRental?.status as RentalStatus}
   583→        productName={cancelRental?.product?.title || ''}
   584→        totalAmount={cancelRental?.totalAmount || 0}
   585→        isPending={cancelMutation.isPending}
   586→        isSuccess={cancelSuccess}
   587→      />
   588→      {/* Payment Modal */}
   589→      <PaymentCheckoutModal
   590→        open={paymentModalOpen}
   591→        onClose={() => { setPaymentModalOpen(false); setPaymentRental(null); }}
   592→        rentalId={paymentRental?.id || ''}
   593→        rentalData={paymentRental ? {
   594→          totalAmount: paymentRental.totalAmount,
   595→          rentalAmount: paymentRental.rentalAmount,
   596→          platformFee: paymentRental.platformFee,
   597→          tax: paymentRental.tax,
   598→          deliveryFee: paymentRental.deliveryFee,
   599→          discount: paymentRental.discount,
   600→          securityDeposit: paymentRental.securityDeposit,
   601→          rentalDays: paymentRental.rentalDays,
   602→          dailyRate: paymentRental.dailyRate,
   603→          startDate: paymentRental.startDate,
   604→          endDate: paymentRental.endDate,
   605→          productTitle: paymentRental.product?.title,
   606→        } : null}
   607→        onSuccess={() => {
   608→          setPaymentModalOpen(false);
   609→          setPaymentRental(null);
   610→          queryClient.invalidateQueries({ queryKey: ['rentals'] });
   611→          toast.success('Payment successful! Rental request sent to owner.');
   612→        }}
   613→      />
   614→    </div>
   615→  );
   616→}
