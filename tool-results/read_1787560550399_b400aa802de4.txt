    39→// ─── Zod Schema ────────────────────────────────────────────
    40→const productSchema = z.object({
    41→  title: z.string().min(3, 'Title must be at least 3 characters').max(100),
    42→  categoryId: z.string().min(1, 'Category is required'),
    43→  description: z.string().min(10, 'Description must be at least 10 characters').max(2000),
    44→  condition: z.enum(['NEW', 'LIKE_NEW', 'GOOD', 'FAIR', 'DAMAGED']),
    45→  brand: z.string().max(100).optional().or(z.literal('')),
    46→  model: z.string().max(100).optional().or(z.literal('')),
    47→  purchaseYear: z.string().optional().or(z.literal('')),
    48→  dailyPrice: z.string().min(1, 'Daily price is required'),
    49→  weeklyPrice: z.string().optional().or(z.literal('')),
    50→  securityDeposit: z.string().min(1, 'Security deposit is required'),
    51→  minRentalDays: z.string().min(1, 'Min rental days is required'),
    52→  maxRentalDays: z.string().min(1, 'Max rental days is required'),
    53→  stateId: z.string().min(1, 'State is required'),
    54→  cityId: z.string().min(1, 'City is required'),
    55→  pickupAddress: z.string().min(5, 'Pickup address must be at least 5 characters').max(500),
    56→  deliveryAvailable: z.boolean(),
    57→  deliveryFee: z.string().optional().or(z.literal('')),
    58→  rentalRules: z.string().max(2000).optional().or(z.literal('')),
    59→  cancellationPolicy: z.string().max(2000).optional().or(z.literal('')),
    60→  ownerNotes: z.string().max(2000).optional().or(z.literal('')),
    61→});
    62→
    63→type ProductFormData = z.infer<typeof productSchema>;

    95→// ─── Main Page ─────────────────────────────────────────────
    96→export default function ListItemPage() {
    97→  const navigate = useAppStore((s) => s.navigate);
    98→  const viewData = useAppStore((s) => s.viewData);
    99→  const queryClient = useQueryClient();
   100→
   101→  const editingProduct = viewData?.product as Product | undefined;
   102→  const isEditing = !!editingProduct;
   103→
   104→  // ─── Image State ───────────────────────────────
   105→  const [images, setImages] = useState<ImageItem[]>(() => {
   106→    if (editingProduct?.images && editingProduct.images.length > 0) {
   107→      return editingProduct.images.map((img: { url: string; id?: string }) => ({
   108→        url: img.url,
   109→        serverUrl: img.url,
   110→      }));
   111→    }
   112→    return [];
   113→  });
   114→  const [isDragOver, setIsDragOver] = useState(false);
   115→  const fileInputRef = useRef<HTMLInputElement>(null);
   116→
   117→  // Load existing images when editing
   118→
   119→  const MAX_IMAGES = 5;
   120→  const MAX_SIZE = 5 * 1024 * 1024; // 5MB
   121→  const ALLOWED_TYPES = ['image/png', 'image/jpeg', 'image/webp', 'image/jpg'];
   122→
   123→  const processFiles = useCallback(async (files: FileList | File[]) => {
   124→    const fileArray = Array.from(files);
   125→    const remaining = MAX_IMAGES - images.length;
   126→    if (remaining <= 0) {
   127→      toast.error(`Maximum ${MAX_IMAGES} images allowed`);
   128→      return;
   129→    }
   130→    const toProcess = fileArray.slice(0, remaining);
   131→    const validFiles: File[] = [];
   132→
   133→    for (const f of toProcess) {
   134→      if (!ALLOWED_TYPES.includes(f.type)) {
   135→        toast.error(`"${f.name}" is not a supported format. Use PNG, JPG, or WEBP.`);
   136→        continue;
   137→      }
   138→      if (f.size > MAX_SIZE) {
   139→        toast.error(`"${f.name}" exceeds 5MB limit`);
   140→        continue;
   141→      }
   142→      validFiles.push(f);
   143→    }
   144→
   145→    if (validFiles.length === 0) return;
   146→
   147→    // Add items with uploading state
   148→    const newItems: ImageItem[] = validFiles.map(f => ({
   149→      file: f,
   150→      url: URL.createObjectURL(f),
   151→      uploading: true,
   152→      progress: 0,
   153→    }));
   154→    setImages(prev => [...prev, ...newItems]);
   155→
   156→    // Upload each file
   157→    for (let i = 0; i < validFiles.length; i++) {
   158→      const itemIdx = images.length + i;
   159→      try {
   160→        // Simulate progress
   161→        const progressInterval = setInterval(() => {
   162→          setImages(prev => {
   163→            const updated = [...prev];
   164→            if (updated[itemIdx]) {
   165→              updated[itemIdx] = {
   166→                ...updated[itemIdx],
   167→                progress: Math.min((updated[itemIdx].progress || 0) + Math.random() * 30, 90),
   168→              };
   169→            }
   170→            return updated;
   171→          });
   172→        }, 200);
   173→
   174→        const result = await api.uploadImages([validFiles[i]]);
   175→
   176→        clearInterval(progressInterval);
   177→
   178→        setImages(prev => {
   179→          const updated = [...prev];
   180→          if (updated[itemIdx] && result.urls && result.urls[0]) {
   181→            updated[itemIdx] = {
   182→              ...updated[itemIdx],
   183→              serverUrl: result.urls[0],
   184→              uploading: false,
   185→              progress: 100,
   186→            };
   187→          }
   188→          return updated;
   189→        });
   190→      } catch {
   191→        setImages(prev => prev.filter((_, idx) => idx !== itemIdx));
   192→        toast.error(`Failed to upload "${validFiles[i].name}"`);
   193→      }
   194→    }
   195→  }, [images.length]);
   196→
   197→  const handleDragOver = useCallback((e: React.DragEvent) => {
   198→    e.preventDefault();
   199→    e.stopPropagation();
   200→    setIsDragOver(true);
   201→  }, []);
   202→
   203→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   204→    e.preventDefault();
   205→    e.stopPropagation();
   206→    setIsDragOver(false);
   207→  }, []);
   208→
   209→  const handleDrop = useCallback((e: React.DragEvent) => {
   210→    e.preventDefault();
   211→    e.stopPropagation();
   212→    setIsDragOver(false);
   213→    if (e.dataTransfer.files.length > 0) {
   214→      processFiles(e.dataTransfer.files);
   215→    }
   216→  }, [processFiles]);
   217→
   218→  const removeImage = useCallback((index: number) => {
   219→    setImages(prev => {
   220→      const item = prev[index];
   221→      // Revoke object URL if it's a local preview
   222→      if (item && !item.serverUrl && item.url.startsWith('blob:')) {
   223→        URL.revokeObjectURL(item.url);
   224→      }
   225→      return prev.filter((_, i) => i !== index);
   226→    });
   227→  }, []);
   228→
   229→  // Fetch categories & states
   230→  const { data: categories = [], isLoading: loadingCategories } = useQuery({
   231→    queryKey: ['categories'],
   232→    queryFn: async () => { const d = await api.getCategories(); return d as unknown as Category[]; },
   233→  });
   234→
   235→  const { data: states = [], isLoading: loadingStates } = useQuery({
   236→    queryKey: ['states'],
   237→    queryFn: async () => { const d = await api.getStates(); return d as unknown as State[]; },
   238→  });

   240→  // Form
   241→  const form = useForm<ProductFormData>({
   242→    resolver: zodResolver(productSchema),

   245→      categoryId: '',
   246→      description: '',
   247→      condition: 'GOOD',
   248→      brand: '',
   249→      model: '',
   250→      purchaseYear: '',
   251→      dailyPrice: '',
   252→      weeklyPrice: '',
   253→      securityDeposit: '',
   254→      minRentalDays: '1',
   255→      maxRentalDays: '30',
   256→      stateId: '',
   257→      cityId: '',
   258→      pickupAddress: '',
   259→      deliveryAvailable: false,
   260→      deliveryFee: '',
   261→      rentalRules: '',
   262→      cancellationPolicy: '',
   263→      ownerNotes: '',
   264→    },
   265→  });
   266→
   267→  const { register, handleSubmit, control, setValue, formState: { errors } } = form;
   268→  const selectedStateId = useWatch({ control: form.control, name: 'stateId' });
   269→  const deliveryAvailable = useWatch({ control: form.control, name: 'deliveryAvailable' });
   270→
   271→  // Get cities for selected state
   272→  const selectedState = states.find((s: State) => s.id === selectedStateId);
   273→  const cities = (selectedState as any)?.cities ?? [];
   274→
   275→  // Populate form when editing
   276→  useEffect(() => {
   277→    if (editingProduct) {
   278→      setValue('title', editingProduct.title);
   279→      setValue('categoryId', editingProduct.categoryId);
   280→      setValue('description', editingProduct.description || '');
   281→      setValue('condition', editingProduct.condition);
   282→      setValue('brand', editingProduct.brand || '');
   283→      setValue('model', editingProduct.model || '');
   284→      setValue('purchaseYear', editingProduct.purchaseYear ? String(editingProduct.purchaseYear) : '');
   285→      setValue('dailyPrice', String(editingProduct.dailyPrice));
   286→      setValue('weeklyPrice', editingProduct.weeklyPrice ? String(editingProduct.weeklyPrice) : '');
   287→      setValue('securityDeposit', String(editingProduct.securityDeposit));
   288→      setValue('minRentalDays', String(editingProduct.minRentalDays));
   289→      setValue('maxRentalDays', String(editingProduct.maxRentalDays));
   290→      setValue('stateId', editingProduct.stateId || '');
   291→      setValue('cityId', editingProduct.cityId || '');
   292→      setValue('pickupAddress', editingProduct.pickupAddress || '');
   293→      setValue('deliveryAvailable', editingProduct.deliveryAvailable);
   294→      setValue('deliveryFee', String(editingProduct.deliveryFee));
   295→      setValue('rentalRules', editingProduct.rentalRules || '');
   296→      setValue('cancellationPolicy', editingProduct.cancellationPolicy || '');
   297→      setValue('ownerNotes', editingProduct.ownerNotes || '');
   298→    }
   299→  }, [editingProduct, setValue]);
   300→
   301→  // Reset city when state changes
   302→  useEffect(() => {
   303→    if (!isEditing || !editingProduct?.cityId) {
   304→      setValue('cityId', '');
   305→    }
   306→  }, [selectedStateId]);
   307→
   308→  // Submit mutation
   309→  const submitMutation = useMutation({
   310→    mutationFn: async (data: ProductFormData) => {
   311→      // Wait for any pending uploads
   312→      const pendingUpload = images.some(img => img.uploading);
   313→      if (pendingUpload) {
   314→        throw new Error('Please wait for images to finish uploading');
   315→      }
   316→
   317→      const payload: Record<string, unknown> = {
   318→        title: data.title,
   319→        categoryId: data.categoryId,
   320→        description: data.description,
   321→        condition: data.condition,
   322→        brand: data.brand || undefined,
   323→        model: data.model || undefined,
   324→        purchaseYear: data.purchaseYear ? parseInt(data.purchaseYear) : undefined,
   325→        dailyPrice: parseFloat(data.dailyPrice),
   326→        weeklyPrice: data.weeklyPrice ? parseFloat(data.weeklyPrice) : undefined,
   327→        securityDeposit: parseFloat(data.securityDeposit),
   328→        minRentalDays: parseInt(data.minRentalDays),
   329→        maxRentalDays: parseInt(data.maxRentalDays),
   330→        stateId: data.stateId,
   331→        cityId: data.cityId,
   332→        pickupAddress: data.pickupAddress,
   333→        deliveryAvailable: data.deliveryAvailable,
   334→        deliveryFee: data.deliveryFee ? parseFloat(data.deliveryFee) : 0,
   335→        rentalRules: data.rentalRules || undefined,
   336→        cancellationPolicy: data.cancellationPolicy || undefined,
   337→        ownerNotes: data.ownerNotes || undefined,
   338→      };
   339→
   340→      // Attach uploaded image URLs for new products
   341→      if (!isEditing) {
   342→        const uploadedUrls = images
   343→          .filter(img => img.serverUrl)
   344→          .map(img => img.serverUrl);
   345→        if (uploadedUrls.length > 0) {
   346→          payload.imageUrls = uploadedUrls;
   347→        }
   348→      }
   349→
   350→      if (isEditing) {
   351→        return api.updateProduct(editingProduct.id, payload);
   352→      }
   353→      return api.createProduct(payload);
   354→    },
   355→    onSuccess: () => {
   356→      toast.success(isEditing ? 'Listing updated successfully!' : 'Listing created successfully!');
   357→      queryClient.invalidateQueries({ queryKey: ['my-products'] });
   358→      queryClient.invalidateQueries({ queryKey: ['products'] });
   359→      navigate('my-listings');
   360→    },
   361→    onError: (e: Error) => toast.error(e.message),
   362→  });
   363→
   364→  const onSubmit = (data: ProductFormData) => {
   365→    submitMutation.mutate(data);
   366→  };
   367→
   368→  const isSubmitting = submitMutation.isPending;
   369→  const isLoadingData = loadingCategories || loadingStates;
   370→  const hasUploadingImages = images.some(img => img.uploading);
   371→
   372→  return (
   373→    <div className="min-h-screen bg-slate-50">
   374→      {/* Header */}
   375→      <div className="bg-white border-b border-slate-200 sticky top-0 z-30">
   376→        <div className="max-w-4xl mx-auto px-4 sm:px-6">
   377→          <div className="flex items-center justify-between h-16">
   378→            <div className="flex items-center gap-3">
   379→              <Button variant="ghost" size="sm" className="text-slate-600 hover:text-slate-900" onClick={() => navigate('my-listings')}>
   380→                <ArrowLeft className="h-4 w-4 mr-1" />Back
   381→              </Button>
   382→              <div className="h-6 w-px bg-slate-200" />
   383→              <h1 className="text-lg font-bold text-slate-900">{isEditing ? 'Edit Listing' : 'List a New Item'}</h1>
   384→            </div>
   385→          </div>
   386→        </div>
   387→      </div>
   388→
   389→      <div className="max-w-4xl mx-auto px-4 sm:px-6 py-6">
   390→        {isLoadingData ? (
   391→          <div className="space-y-6">
   392→            {Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-48 w-full rounded-xl" />)}
   393→          </div>
   394→        ) : (
   395→          <motion.form
   396→            initial={{ opacity: 0, y: 16 }}
   397→            animate={{ opacity: 1, y: 0 }}
   398→            onSubmit={handleSubmit(onSubmit)}
   399→            className="space-y-6"
   400→          >
   401→            {/* Basic Information */}
   402→            <FormSection title="Basic Information">
   403→              <div className="space-y-4">
   404→                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   405→                  <div className="md:col-span-2">
   406→                    <Label htmlFor="title" className="text-sm font-medium text-slate-700">
   407→                      Product Title <span className="text-red-500">*</span>
   408→                    </Label>
   409→                    <Input id="title" placeholder="e.g., Canon EOS R5 Camera" className="mt-1.5" {...register('title')} />
   410→                    {errors.title && <p className="text-xs text-red-500 mt-1">{errors.title.message}</p>}
   411→                  </div>
   412→
   413→                  <div>
   414→                    <Label className="text-sm font-medium text-slate-700">Category <span className="text-red-500">*</span></Label>
   415→                    <Controller
   416→                      name="categoryId"
   417→                      control={control}
   418→                      render={({ field }) => (
   419→                        <Select value={field.value} onValueChange={field.onChange}>
   420→                          <SelectTrigger className="mt-1.5"><SelectValue placeholder="Select category" /></SelectTrigger>
   421→                          <SelectContent>
   422→                            {categories.map((cat) => (
   423→                              <SelectItem key={cat.id} value={cat.id}>{cat.name}</SelectItem>
   424→                            ))}
   425→                          </SelectContent>
   426→                        </Select>
   427→                      )}
   428→                    />
   429→                    {errors.categoryId && <p className="text-xs text-red-500 mt-1">{errors.categoryId.message}</p>}
   430→                  </div>
   431→
   432→                  <div>
   433→                    <Label className="text-sm font-medium text-slate-700">Condition <span className="text-red-500">*</span></Label>
   434→                    <Controller
   435→                      name="condition"
   436→                      control={control}
   437→                      render={({ field }) => (
   438→                        <Select value={field.value} onValueChange={field.onChange}>
   439→                          <SelectTrigger className="mt-1.5"><SelectValue /></SelectTrigger>
   440→                          <SelectContent>
   441→                            {conditionOptions.map((opt) => (
   442→                              <SelectItem key={opt.value} value={opt.value}>{opt.label}</SelectItem>
   443→                            ))}
   444→                          </SelectContent>
   445→                        </Select>
   446→                      )}
   447→                    />
   448→                  </div>
   449→                </div>
   450→
   451→                <div>
   452→                  <Label htmlFor="description" className="text-sm font-medium text-slate-700">
   453→                    Description <span className="text-red-500">*</span>
   454→                  </Label>
   455→                  <Textarea id="description" placeholder="Describe your item in detail..." rows={4} className="mt-1.5 resize-none" {...register('description')} />
   456→                  {errors.description && <p className="text-xs text-red-500 mt-1">{errors.description.message}</p>}
   457→                </div>
   458→
   459→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   460→                  <div>
   461→                    <Label htmlFor="brand" className="text-sm font-medium text-slate-700">Brand</Label>
   462→                    <Input id="brand" placeholder="e.g., Canon" className="mt-1.5" {...register('brand')} />
   463→                  </div>
   464→                  <div>
   465→                    <Label htmlFor="model" className="text-sm font-medium text-slate-700">Model</Label>
   466→                    <Input id="model" placeholder="e.g., EOS R5" className="mt-1.5" {...register('model')} />
   467→                  </div>
   468→                  <div>
   469→                    <Label htmlFor="purchaseYear" className="text-sm font-medium text-slate-700">Purchase Year</Label>
   470→                    <Input id="purchaseYear" placeholder="e.g., 2023" className="mt-1.5" {...register('purchaseYear')} />
   471→                  </div>
   472→                </div>
   473→              </div>
   474→            </FormSection>
   475→
   476→            {/* Pricing */}
   477→            <FormSection title="Pricing">
   478→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   479→                <div>
   480→                  <Label htmlFor="dailyPrice" className="text-sm font-medium text-slate-700">Daily Price (₹) <span className="text-red-500">*</span></Label>
   481→                  <div className="relative mt-1.5">
   482→                    <IndianRupee className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
   483→                    <Input id="dailyPrice" placeholder="0" className="pl-9" {...register('dailyPrice')} />

   493→                </div>
   494→                <div>
