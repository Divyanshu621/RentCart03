    36→    36→function AnimatedCounter({ target, prefix = '', suffix = '', decimals = 0 }: {
    37→    37→  target: number; prefix?: string; suffix?: string; decimals?: number;
    38→    38→}) {
    39→    39→  const [count, setCount] = useState(0);
    40→    40→  const ref = useRef<HTMLSpanElement>(null);
    41→    41→  const started = useRef(false);
    42→    42→
    43→    43→  useEffect(() => {
    44→    44→    const el = ref.current;
    45→    45→    if (!el) return;
    46→    46→    const observer = new IntersectionObserver(
    47→    47→      ([entry]) => {
    48→    48→        if (entry.isIntersecting && !started.current) {
    49→    49→          started.current = true;
    50→    50→          const duration = 1200;
    51→    51→          const startTime = performance.now();
    52→    52→          const animate = (now: number) => {
    53→    53→            const elapsed = now - startTime;
    54→    54→            const progress = Math.min(elapsed / duration, 1);
    55→    55→            // ease-out cubic
    56→    56→            const eased = 1 - Math.pow(1 - progress, 3);
    57→    57→            setCount(eased * target);
    58→    58→            if (progress < 1) requestAnimationFrame(animate);
    59→    59→          };
    60→    60→          requestAnimationFrame(animate);
    61→    61→        }
    62→    62→      },
    63→    63→      { threshold: 0.1 }
    64→    64→    );
    65→    65→    observer.observe(el);
    66→    66→    return () => observer.disconnect();
    67→    67→  }, [target]);
    68→    68→
    69→    69→  const formatted = decimals > 0
    70→    70→    ? count.toLocaleString('en-IN', { minimumFractionDigits: decimals, maximumFractionDigits: decimals })
    71→    71→    : Math.round(count).toLocaleString('en-IN');
    72→    72→
    73→    73→  return <span ref={ref}>{prefix}{formatted}{suffix}</span>;
    74→    74→}
