   748→  async function createRental(data: {
   749→    customer: any, owner: any, product: any,
   750→    startDate: Date, endDate: Date,
   751→    status: string, actualReturnDate?: Date,
   752→    cancellationReason?: string,

   757→    const securityDeposit = data.product.securityDeposit

   759→    const tax = Math.round(rentalAmount * 0.18 * 100) / 100
   760→    const totalAmount = rentalAmount + platformFee + tax + securityDeposit
   761→
   762→    const rental = await db.rental.create({
   763→      data: {
   764→        customerId: data.customer.id,
   765→        ownerId: data.owner.id,
   766→        productId: data.product.id,
   767→        startDate: data.startDate,
   768→        endDate: data.endDate,
   769→        actualReturnDate: data.actualReturnDate ?? null,
   770→        rentalDays,
   771→        dailyRate,
   772→        rentalAmount,
   773→        securityDeposit,
   774→        platformFee,
   775→        tax,
   776→        totalAmount,
   777→        status: data.status,
   778→        cancellationReason: data.cancellationReason ?? null,
   779→      },
   780→    })
   781→    rentals.push(rental)
   782→    return rental
   783→  }
