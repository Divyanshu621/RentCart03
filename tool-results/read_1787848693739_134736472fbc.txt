    49→    49→  securityDeposit: z.string().min(1, 'Security deposit is required'),

   100→   100→  const editingProduct = viewData?.product as Product | undefined;
   101→   101→  const isEditing = !!editingProduct;
   102→   102→
   103→   103→  // ─── Image State ───────────────────────────────
   104→   104→  const [images, setImages] = useState<ImageItem[]>(() => {
   105→   105→    if (editingProduct?.images && editingProduct.images.length > 0) {
   106→   106→      return editingProduct.images.map((img: { url: string; id?: string }) => ({
   107→   107→        url: img.url,

   115→   115→
   116→   116→  // Load existing images when editing
   117→   117→
   118→   118→  const MAX_IMAGES = 5;
   119→   119→  const MAX_SIZE = 5 * 1024 * 1024; // 5MB
   120→   120→  const ALLOWED_TYPES = ['image/png', 'image/jpeg', 'image/webp', 'image/jpg'];
   121→   121→
   122→   122→  const processFiles = useCallback(async (files: FileList | File[]) => {
   123→   123→    const fileArray = Array.from(files);
   124→   124→    const remaining = MAX_IMAGES - images.length;
   125→   125→    if (remaining <= 0) {
   126→   126→      toast.error(`Maximum ${MAX_IMAGES} images allowed`);
   127→   127→      return;
   128→   128→    }
   129→   129→    const toProcess = fileArray.slice(0, remaining);
   130→   130→    const validFiles: File[] = [];
   131→   131→
   132→   132→    for (const f of toProcess) {
   133→   133→      if (!ALLOWED_TYPES.includes(f.type)) {
   134→   134→        toast.error(`"${f.name}" is not a supported format. Use PNG, JPG, or WEBP.`);
   135→   135→        continue;
   136→   136→      }
   137→   137→      if (f.size > MAX_SIZE) {
   138→   138→        toast.error(`"${f.name}" exceeds 5MB limit`);
   139→   139→        continue;
   140→   140→      }
   141→   141→      validFiles.push(f);
   142→   142→    }
   143→   143→
   144→   144→    if (validFiles.length === 0) return;
   145→   145→
   146→   146→    // Add items with uploading state
   147→   147→    const newItems: ImageItem[] = validFiles.map(f => ({
   148→   148→      file: f,
   149→   149→      url: URL.createObjectURL(f),
   150→   150→      uploading: true,
   151→   151→      progress: 0,
   152→   152→    }));
   153→   153→    setImages(prev => [...prev, ...newItems]);
   154→   154→
   155→   155→    // Upload each file
   156→   156→    for (let i = 0; i < validFiles.length; i++) {
   157→   157→      const itemIdx = images.length + i;
   158→   158→      try {
   159→   159→        // Simulate progress
   160→   160→        const progressInterval = setInterval(() => {
   161→   161→          setImages(prev => {
   162→   162→            const updated = [...prev];
   163→   163→            if (updated[itemIdx]) {
   164→   164→              updated[itemIdx] = {
   165→   165→                ...updated[itemIdx],
   166→   166→                progress: Math.min((updated[itemIdx].progress || 0) + Math.random() * 30, 90),
   167→   167→              };
   168→   168→            }
   169→   169→            return updated;
   170→   170→          });
   171→   171→        }, 200);
   172→   172→
   173→   173→        const result = await api.uploadImages([validFiles[i]]);
   174→   174→
   175→   175→        clearInterval(progressInterval);
   176→   176→
   177→   177→        setImages(prev => {
   178→   178→          const updated = [...prev];
   179→   179→          if (updated[itemIdx] && result.url) {
   180→   180→            updated[itemIdx] = {
   181→   181→              ...updated[itemIdx],
   182→   182→              serverUrl: result.url,
   183→   183→              uploading: false,
   184→   184→              progress: 100,
   185→   185→            };
   186→   186→          }
   187→   187→          return updated;
   188→   188→        });
   189→   189→      } catch {
   190→   190→        setImages(prev => prev.filter((_, idx) => idx !== itemIdx));
   191→   191→        toast.error(`Failed to upload "${validFiles[i].name}"`);
   192→   192→      }
   193→   193→    }
   194→   194→  }, [images.length]);
   195→   195→
   196→   196→  const handleDragOver = useCallback((e: React.DragEvent) => {
   197→   197→    e.preventDefault();

   248→   248→      model: '',
   249→   249→      purchaseYear: '',
   250→   250→      dailyPrice: '',
   251→   251→      weeklyPrice: '',
   252→   252→      securityDeposit: '',
   253→   253→      minRentalDays: '1',
   254→   254→      maxRentalDays: '30',
   255→   255→      stateId: '',
   256→   256→      cityId: '',
   257→   257→      pickupAddress: '',
   258→   258→      deliveryAvailable: false,
   259→   259→      deliveryFee: '',
   260→   260→      rentalRules: '',
   261→   261→      cancellationPolicy: '',
   262→   262→      ownerNotes: '',
   263→   263→    },
   264→   264→  });
   265→   265→
   266→   266→  const { register, handleSubmit, control, setValue, formState: { errors } } = form;

   273→   273→
   274→   274→  // Populate form when editing
   275→   275→  useEffect(() => {

   282→   282→      setValue('model', editingProduct.model || '');
   283→   283→      setValue('purchaseYear', editingProduct.purchaseYear ? String(editingProduct.purchaseYear) : '');
   284→   284→      setValue('dailyPrice', String(editingProduct.dailyPrice));
   285→   285→      setValue('weeklyPrice', editingProduct.weeklyPrice ? String(editingProduct.weeklyPrice) : '');
   286→   286→      setValue('securityDeposit', String(editingProduct.securityDeposit));
   287→   287→      setValue('minRentalDays', String(editingProduct.minRentalDays));
   288→   288→      setValue('maxRentalDays', String(editingProduct.maxRentalDays));

   299→   299→

   306→   306→
   307→   307→  // Submit mutation
   308→   308→  const submitMutation = useMutation({

   329→   329→        stateId: data.stateId,

   331→   331→        pickupAddress: data.pickupAddress,
   332→   332→        deliveryAvailable: data.deliveryAvailable,
   333→   333→        deliveryFee: data.deliveryFee ? parseFloat(data.deliveryFee) : 0,

   339→   339→      // Attach uploaded image URLs

   382→   382→          </div>
   383→   383→        </div>
   384→   384→      </div>
   385→   385→
   386→   386→      <div className="max-w-4xl mx-auto px-4 sm:px-6 py-6">
   387→   387→        {isLoadingData ? (

   392→   392→          <motion.form

   395→   395→            onSubmit={handleSubmit(onSubmit)}
   396→   396→            className="space-y-6"
   397→   397→          >
   398→   398→            {/* Basic Information */}
   399→   399→            <FormSection title="Basic Information">
   400→   400→              <div className="space-y-4">
   401→   401→                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   402→   402→                  <div className="md:col-span-2">
