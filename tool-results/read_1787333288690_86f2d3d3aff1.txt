    49→  agreeTerms: z.literal(true, { errorMap: () => ({ message: 'You must agree to the terms' }) }),
    50→});

   165→// ─── Login Form ─────────────────────────────────────────────
   166→function LoginForm({
   167→  onSuccess,
   168→  onToggleView,
   169→  showPassword,
   170→  onTogglePassword,
   171→}: {
   172→  onSuccess: (data: Record<string, unknown>) => void;
   173→  onToggleView: () => void;
   174→  showPassword: boolean;
   175→  onTogglePassword: () => void;
   176→}) {
   177→  const {
   178→    register,
   179→    handleSubmit,
   180→    formState: { errors, isSubmitting },
   181→    setError,
   182→  } = useForm<LoginFormData>({
   183→    resolver: zodResolver(loginSchema),
   184→  });
   185→
   186→  const [serverError, setServerError] = useState('');
   187→  const [isGoogleLoading, setIsGoogleLoading] = useState(false);
   188→  const [showGoogleDialog, setShowGoogleDialog] = useState(false);
   189→  const [googleEmail, setGoogleEmail] = useState('');
   190→  const [googleName, setGoogleName] = useState('');
   191→  const [googleConfigured, setGoogleConfigured] = useState(false);
   192→  const [checkedConfig, setCheckedConfig] = useState(false);
   193→
   194→  // Check if real Google OAuth is configured
   195→  useEffect(() => {
   196→    api.getGoogleConfig()
   197→      .then((config) => setGoogleConfigured(config.configured))
   198→      .catch(() => setGoogleConfigured(false))
   199→      .finally(() => setCheckedConfig(true));
   200→  }, []);
   201→
   202→  const handleGoogleClick = async () => {
   203→    if (googleConfigured) {
   204→      // Real OAuth: redirect to backend which redirects to Google
   205→      setIsGoogleLoading(true);
   206→      window.location.href = '/api/auth/google';
   207→    } else {
   208→      // Demo mode: show the Google-style dialog
   209→      setGoogleEmail('');
   210→      setGoogleName('');
   211→      setShowGoogleDialog(true);
   212→    }
   213→  };
   214→
   215→  const handleDemoGoogleLogin = async () => {
   216→    if (!googleEmail.trim() || !googleEmail.includes('@')) {
   217→      toast.error('Please enter a valid email address');
   218→      return;
   219→    }
   220→    const emailLower = googleEmail.trim().toLowerCase();
   221→    if (!emailLower.endsWith('@gmail.com') && !emailLower.endsWith('@googlemail.com')) {
   222→      toast.error('Please enter a valid Gmail address (e.g., you@gmail.com)');
   223→      return;
   224→    }
   225→    setIsGoogleLoading(true);
   226→    try {
   227→      const name = googleName.trim() || emailLower.split('@')[0].replace(/[._]/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase());
   228→      const result = await api.googleAuth({ email: emailLower, name });
   229→      setShowGoogleDialog(false);
   230→      onSuccess(result.user);
   231→    } catch (err) {
   232→      const message = err instanceof Error ? err.message : 'Google sign-in failed';
   233→      toast.error(message);
   234→    } finally {
   235→      setIsGoogleLoading(false);
   236→    }
   237→  };
   238→
   239→  const onSubmit = async (data: LoginFormData) => {
   240→    setServerError('');
   241→    try {
   242→      const res = await api.login({ email: data.email, password: data.password });
   243→      onSuccess(res.user);
   244→    } catch (err) {
   245→      const message = err instanceof Error ? err.message : 'Login failed';
   246→      setServerError(message);
   247→      toast.error(message);
   248→    }
   249→  };
   250→
   251→  return (
   252→    <div>
   253→      <DialogHeader className="mb-6 text-left">
   254→        <DialogTitle className="text-xl font-bold text-[#0f172a]">
   255→        </DialogTitle>
   256→        <DialogDescription className="text-muted-foreground">
   257→        </DialogDescription>
   258→      </DialogHeader>
   259→
   260→      <h2 className="text-xl font-bold text-[#0f172a] mb-1">Welcome back</h2>
   261→      <p className="text-sm text-muted-foreground mb-6">
   262→        Sign in to your RentCart account
   263→      </p>
   264→
   265→      {serverError && (
   266→        <div className="mb-4 p-3 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm">
   267→          {serverError}
   268→        </div>
   269→      )}
   270→
   271→      {/* Google sign-in divider + button */}
   272→      <div className="relative my-4">
   273→        <div className="absolute inset-0 flex items-center">
   274→          <div className="w-full border-t border-slate-200" />
   275→        </div>
   276→        <div className="relative flex justify-center text-xs">
   277→          <span className="px-3 bg-white text-muted-foreground uppercase tracking-wider">
   278→            or
   279→          </span>
   280→        </div>
   281→      </div>
   282→
   283→      <Button
   284→        type="button"
   285→        variant="outline"
   286→        onClick={handleGoogleClick}
   287→        disabled={isGoogleLoading || !checkedConfig}
   288→        className="w-full h-11 border-slate-200 hover:bg-slate-50 font-medium text-sm"
   289→      >
   290→        {isGoogleLoading ? (
   291→          <Loader2 className="w-4 h-4 animate-spin" />
   292→        ) : (
   293→          <svg className="w-5 h-5" viewBox="0 0 24 24">
   294→            <path d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92a5.06 5.06 0 01-2.2 3.32v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.1z" fill="#4285F4" />
   295→            <path d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z" fill="#34A853" />
   296→            <path d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z" fill="#FBBC05" />
   297→            <path d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z" fill="#EA4335" />
   298→          </svg>
   299→        )}
   300→        {isGoogleLoading ? 'Signing in...' : 'Continue with Google'}
   301→      </Button>
   302→
   303→      {/* Google email dialog */}
   304→      {showGoogleDialog && (
   305→        <div className="fixed inset-0 z-[200] bg-black/50 flex items-center justify-center p-4" onClick={() => !isGoogleLoading && setShowGoogleDialog(false)}>
   306→          <div className="bg-white rounded-2xl shadow-2xl w-full max-w-sm p-6" onClick={(e) => e.stopPropagation()}>
   307→            <div className="flex items-center justify-between mb-5">
   308→              <div className="flex items-center gap-3">
   309→                <svg className="w-7 h-7" viewBox="0 0 24 24">
   310→                  <path d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92a5.06 5.06 0 01-2.2 3.32v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.1z" fill="#4285F4" />
   311→                  <path d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z" fill="#34A853" />
   312→                  <path d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z" fill="#FBBC05" />
   313→                  <path d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z" fill="#EA4335" />
   314→                </svg>
   315→                <div>
   316→                  <h3 className="font-bold text-[#0f172a]">Sign in with Google</h3>
   317→                  <p className="text-xs text-muted-foreground">Enter your Gmail to continue</p>
   318→                </div>
   319→              </div>
   320→              <button onClick={() => setShowGoogleDialog(false)} disabled={isGoogleLoading} className="p-1.5 rounded-lg hover:bg-gray-100 text-gray-400">
   321→                <X className="w-4 h-4" />
   322→              </button>
   323→            </div>
   324→            <div className="space-y-4">
   325→              <div className="space-y-1.5">
   326→                <Label className="text-sm font-medium text-[#0f172a]">Gmail Address</Label>
   327→                <Input type="email" placeholder="you@gmail.com" value={googleEmail} onChange={(e) => setGoogleEmail(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleDemoGoogleLogin()} disabled={isGoogleLoading} className="h-11" autoFocus />
   328→                <p className="text-xs text-muted-foreground">Only Gmail addresses are supported for demo mode</p>
   329→              </div>
   330→              <div className="space-y-1.5">
   331→                <Label className="text-sm font-medium text-[#0f172a]">Display Name <span className="text-muted-foreground font-normal">(optional)</span></Label>
   332→                <Input type="text" placeholder="Your name" value={googleName} onChange={(e) => setGoogleName(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleDemoGoogleLogin()} disabled={isGoogleLoading} className="h-11" />
   333→              </div>
   334→              <Button onClick={handleDemoGoogleLogin} disabled={isGoogleLoading || !googleEmail.trim()} className="w-full h-11 bg-emerald-600 hover:bg-emerald-700 text-white font-medium">
   335→                {isGoogleLoading ? (<><Loader2 className="w-4 h-4 animate-spin" /> Signing in...</>) : (<>Continue <ArrowRight className="w-4 h-4" /></>)}
   336→              </Button>
   337→            </div>
   338→          </div>
   339→        </div>
   340→      )}
   341→
   342→      <form onSubmit={handleSubmit(onSubmit)} className="space-y-4">
   343→        <div className="space-y-2">
   344→          <Label htmlFor="login-email" className="text-sm font-medium text-[#0f172a]">
   345→            Email
   346→          </Label>
   347→          <div className="relative">
   348→            <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   349→            <Input
   350→              id="login-email"
   351→              type="email"
   352→              placeholder="you@example.com"
   353→              className="pl-10 h-11"
   354→              {...register('email')}
   355→            />
   356→          </div>
   357→          {errors.email && (
   358→            <p className="text-sm text-red-500">{errors.email.message}</p>
   359→          )}
   360→        </div>
   361→
   362→        <div className="space-y-2">
   363→          <div className="flex items-center justify-between">
   364→            <Label htmlFor="login-password" className="text-sm font-medium text-[#0f172a]">
   365→              Password
   366→            </Label>
   367→            <button
   368→              type="button"
   369→              onClick={() => toast.info('Password reset link sent to your email!')}
   370→              className="text-sm text-emerald-600 hover:text-emerald-700 font-medium"
   371→            >

   395→        </div>
   396→
   397→        <Button

   421→        >
   422→          Sign Up
   423→        </button>
   424→      </p>
   425→    </div>
   426→  );

   429→// ─── Register Form ──────────────────────────────────────────
   430→function RegisterForm({
   431→  onSuccess,
   432→  onToggleView,
   433→  showPassword,
   434→  onTogglePassword,
   435→  states,
   436→  loadingStates,
   437→}: {
   438→  onSuccess: (data: Record<string, unknown>) => void;
   439→  onToggleView: () => void;
   440→  showPassword: boolean;
   441→  onTogglePassword: () => void;
   442→  states: State[];
   443→  loadingStates: boolean;
   444→}) {
   445→  const [serverError, setServerError] = useState('');
   446→
   447→  const {
   448→    register,
   449→    handleSubmit,
   450→    control,
   451→    setValue,
   452→    formState: { errors, isSubmitting },
   453→  } = useForm<RegisterFormData>({
   454→    resolver: zodResolver(registerSchema),
   455→    defaultValues: {
   456→      name: '',
   457→      email: '',
   458→      phone: '',
   459→      password: '',
   460→      stateId: '',
   461→      cityId: '',
   462→      pinCode: '',
   463→      address: '',
   464→      agreeTerms: false as unknown as true,
   465→    },
   466→  });
   467→
   468→  const selectedStateId = useWatch({ control, name: 'stateId' });
   469→  const selectedState = states.find((s) => s.id === selectedStateId);
   470→  const cities = selectedState?.cities?.filter((c) => c.isActive) || [];
   471→
   472→  // Reset city when state changes
   473→  useEffect(() => {
   474→    setValue('cityId', '');
   475→  }, [selectedStateId, setValue]);
   476→
   477→  const onSubmit = async (data: RegisterFormData) => {
   478→    setServerError('');
   479→    try {
   480→      const res = await api.register({
   481→        name: data.name,
   482→        email: data.email,
   483→        phone: data.phone || undefined,
   484→        password: data.password,
   485→        stateId: data.stateId,
   486→        cityId: data.cityId,
   487→        pinCode: data.pinCode || undefined,
   488→        address: data.address || undefined,
   489→      });
   490→      onSuccess(res.user);
   491→    } catch (err) {
   492→      const message = err instanceof Error ? err.message : 'Registration failed';
   493→      setServerError(message);
   494→      toast.error(message);
   495→    }
   496→  };
   497→
   498→  return (
   499→    <div>
   500→      <h2 className="text-xl font-bold text-[#0f172a] mb-1">Create your account</h2>
   501→      <p className="text-sm text-muted-foreground mb-5">
   502→        Join RentCart and start renting today
   503→      </p>
   504→
   505→      {serverError && (
   506→        <div className="mb-4 p-3 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm">
   507→          {serverError}
   508→        </div>
   509→      )}
   510→
   511→      <form onSubmit={handleSubmit(onSubmit)} className="space-y-3.5 max-h-[55vh] overflow-y-auto pr-1 custom-scrollbar">
