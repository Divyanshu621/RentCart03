    50→    50→  securityDeposit: z.string().min(1, 'Security deposit is required'),

   100→   100→
   101→   101→  const editingProduct = viewData?.product as Product | undefined;
   102→   102→  const isEditing = !!editingProduct;
   103→   103→
   104→   104→  // ─── Image State ───────────────────────────────
   105→   105→  const [images, setImages] = useState<ImageItem[]>(() => {
   106→   106→    if (editingProduct?.images && editingProduct.images.length > 0) {
   107→   107→      return editingProduct.images.map((img: { url: string; id?: string }) => ({
   108→   108→        url: img.url,
   109→   109→        serverUrl: img.url,

   116→   116→

   118→   118→
   119→   119→  const MAX_IMAGES = 5;
   120→   120→  const MAX_SIZE = 5 * 1024 * 1024; // 5MB
   121→   121→  const ALLOWED_TYPES = ['image/png', 'image/jpeg', 'image/webp', 'image/jpg'];
   122→   122→
   123→   123→  const processFiles = useCallback(async (files: FileList | File[]) => {
   124→   124→    const fileArray = Array.from(files);
   125→   125→    const remaining = MAX_IMAGES - images.length;
   126→   126→    if (remaining <= 0) {
   127→   127→      toast.error(`Maximum ${MAX_IMAGES} images allowed`);
   128→   128→      return;
   129→   129→    }
   130→   130→    const toProcess = fileArray.slice(0, remaining);
   131→   131→    const validFiles: File[] = [];
   132→   132→
   133→   133→    for (const f of toProcess) {
   134→   134→      if (!ALLOWED_TYPES.includes(f.type)) {
   135→   135→        toast.error(`"${f.name}" is not a supported format. Use PNG, JPG, or WEBP.`);
   136→   136→        continue;
   137→   137→      }
   138→   138→      if (f.size > MAX_SIZE) {
   139→   139→        toast.error(`"${f.name}" exceeds 5MB limit`);
   140→   140→        continue;
   141→   141→      }
   142→   142→      validFiles.push(f);
   143→   143→    }
   144→   144→
   145→   145→    if (validFiles.length === 0) return;
   146→   146→
   147→   147→    // Add items with uploading state
   148→   148→    const newItems: ImageItem[] = validFiles.map(f => ({
   149→   149→      file: f,
   150→   150→      url: URL.createObjectURL(f),
   151→   151→      uploading: true,
   152→   152→      progress: 0,
   153→   153→    }));
   154→   154→    setImages(prev => [...prev, ...newItems]);
   155→   155→
   156→   156→    // Upload each file
   157→   157→    for (let i = 0; i < validFiles.length; i++) {
   158→   158→      const itemIdx = images.length + i;
   159→   159→      try {
   160→   160→        // Simulate progress
   161→   161→        const progressInterval = setInterval(() => {
   162→   162→          setImages(prev => {
   163→   163→            const updated = [...prev];
   164→   164→            if (updated[itemIdx]) {
   165→   165→              updated[itemIdx] = {
   166→   166→                ...updated[itemIdx],
   167→   167→                progress: Math.min((updated[itemIdx].progress || 0) + Math.random() * 30, 90),
   168→   168→              };
   169→   169→            }
   170→   170→            return updated;
   171→   171→          });
   172→   172→        }, 200);
   173→   173→
   174→   174→        const result = await api.uploadImages([validFiles[i]]);
   175→   175→
   176→   176→        clearInterval(progressInterval);
   177→   177→
   178→   178→        setImages(prev => {
   179→   179→          const updated = [...prev];
   180→   180→          if (updated[itemIdx] && result.urls && result.urls[0]) {
   181→   181→            updated[itemIdx] = {
   182→   182→              ...updated[itemIdx],
   183→   183→              serverUrl: result.urls[0],
   184→   184→              uploading: false,
   185→   185→              progress: 100,
   186→   186→            };
   187→   187→          }
   188→   188→          return updated;
   189→   189→        });
   190→   190→      } catch {
   191→   191→        setImages(prev => prev.filter((_, idx) => idx !== itemIdx));
   192→   192→        toast.error(`Failed to upload "${validFiles[i].name}"`);
   193→   193→      }
   194→   194→    }
   195→   195→  }, [images.length]);
   196→   196→

   246→   246→      description: '',
   247→   247→      condition: 'GOOD',
   248→   248→      brand: '',
   249→   249→      model: '',
   250→   250→      purchaseYear: '',
   251→   251→      dailyPrice: '',
   252→   252→      weeklyPrice: '',
   253→   253→      securityDeposit: '',
   254→   254→      minRentalDays: '1',
   255→   255→      maxRentalDays: '30',
   256→   256→      stateId: '',
   257→   257→      cityId: '',
   258→   258→      pickupAddress: '',
   259→   259→      deliveryAvailable: false,
   260→   260→      deliveryFee: '',
   261→   261→      rentalRules: '',
   262→   262→      cancellationPolicy: '',
   263→   263→      ownerNotes: '',
   264→   264→    },
   265→   265→  });
   266→   266→
   267→   267→  const { register, handleSubmit, control, setValue, formState: { errors } } = form;

   273→   273→  const cities = (selectedState as any)?.cities ?? [];
   274→   274→
   275→   275→  // Populate form when editing
   276→   276→  useEffect(() => {

   282→   282→      setValue('brand', editingProduct.brand || '');
   283→   283→      setValue('model', editingProduct.model || '');
   284→   284→      setValue('purchaseYear', editingProduct.purchaseYear ? String(editingProduct.purchaseYear) : '');
   285→   285→      setValue('dailyPrice', String(editingProduct.dailyPrice));
   286→   286→      setValue('weeklyPrice', editingProduct.weeklyPrice ? String(editingProduct.weeklyPrice) : '');
   287→   287→      setValue('securityDeposit', String(editingProduct.securityDeposit));
   288→   288→      setValue('minRentalDays', String(editingProduct.minRentalDays));
   289→   289→      setValue('maxRentalDays', String(editingProduct.maxRentalDays));

   300→   300→

   307→   307→
   308→   308→  // Submit mutation

   332→   332→        pickupAddress: data.pickupAddress,
   333→   333→        deliveryAvailable: data.deliveryAvailable,
   334→   334→        deliveryFee: data.deliveryFee ? parseFloat(data.deliveryFee) : 0,

   383→   383→          </div>
   384→   384→        </div>
   385→   385→      </div>
   386→   386→

   392→   392→        ) : (
   393→   393→          <motion.form
   394→   394→            initial={{ opacity: 0, y: 16 }}
   395→   395→            animate={{ opacity: 1, y: 0 }}
   396→   396→            onSubmit={handleSubmit(onSubmit)}
   397→   397→            className="space-y-6"
   398→   398→          >
   399→   399→            {/* Basic Information */}
   400→   400→            <FormSection title="Basic Information">
   401→   401→              <div className="space-y-4">
   402→   402→                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   403→   403→                  <div className="md:col-span-2">

   411→   411→                  <div>

   430→   430→                  <div>
