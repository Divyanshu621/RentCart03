    38→// ─── Zod Schema ────────────────────────────────────────────
    39→const productSchema = z.object({
    40→  title: z.string().min(3, 'Title must be at least 3 characters').max(100),
    41→  categoryId: z.string().min(1, 'Category is required'),
    42→  description: z.string().min(10, 'Description must be at least 10 characters').max(2000),
    43→  condition: z.enum(['NEW', 'LIKE_NEW', 'GOOD', 'FAIR', 'DAMAGED']),
    44→  brand: z.string().max(100).optional().or(z.literal('')),
    45→  model: z.string().max(100).optional().or(z.literal('')),
    46→  purchaseYear: z.string().optional().or(z.literal('')),
    47→  dailyPrice: z.string().min(1, 'Daily price is required'),
    48→  weeklyPrice: z.string().optional().or(z.literal('')),
    49→  securityDeposit: z.string().min(1, 'Security deposit is required'),
    50→  minRentalDays: z.string().min(1, 'Min rental days is required'),
    51→  maxRentalDays: z.string().min(1, 'Max rental days is required'),
    52→  stateId: z.string().min(1, 'State is required'),
    53→  cityId: z.string().min(1, 'City is required'),
    54→  pickupAddress: z.string().min(5, 'Pickup address must be at least 5 characters').max(500),
    55→  deliveryAvailable: z.boolean(),
    56→  deliveryFee: z.string().optional().or(z.literal('')),
    57→  rentalRules: z.string().max(2000).optional().or(z.literal('')),
    58→  cancellationPolicy: z.string().max(2000).optional().or(z.literal('')),
    59→  ownerNotes: z.string().max(2000).optional().or(z.literal('')),
    60→});

    95→export default function ListItemPage() {
    96→  const navigate = useAppStore((s) => s.navigate);
    97→  const viewData = useAppStore((s) => s.viewData);
    98→  const queryClient = useQueryClient();
    99→
   100→  const editingProduct = viewData?.product as Product | undefined;
   101→  const isEditing = !!editingProduct;
   102→
   103→  // ─── Image State ───────────────────────────────
   104→  const [images, setImages] = useState<ImageItem[]>(() => {
   105→    if (editingProduct?.images && editingProduct.images.length > 0) {
   106→      return editingProduct.images.map((img: { url: string; id?: string }) => ({
   107→        url: img.url,
   108→        serverUrl: img.url,
   109→      }));
   110→    }
   111→    return [];
   112→  });
   113→  const [isDragOver, setIsDragOver] = useState(false);
   114→  const fileInputRef = useRef<HTMLInputElement>(null);
   115→
   116→  // Load existing images when editing
   117→
   118→  const MAX_IMAGES = 5;
   119→  const MAX_SIZE = 5 * 1024 * 1024; // 5MB
   120→  const ALLOWED_TYPES = ['image/png', 'image/jpeg', 'image/webp', 'image/jpg'];
   121→
   122→  const processFiles = useCallback(async (files: FileList | File[]) => {
   123→    const fileArray = Array.from(files);
   124→    const remaining = MAX_IMAGES - images.length;
   125→    if (remaining <= 0) {
   126→      toast.error(`Maximum ${MAX_IMAGES} images allowed`);
   127→      return;
   128→    }
   129→    const toProcess = fileArray.slice(0, remaining);
   130→    const validFiles: File[] = [];
   131→
   132→    for (const f of toProcess) {
   133→      if (!ALLOWED_TYPES.includes(f.type)) {
   134→        toast.error(`"${f.name}" is not a supported format. Use PNG, JPG, or WEBP.`);
   135→        continue;
   136→      }
   137→      if (f.size > MAX_SIZE) {
   138→        toast.error(`"${f.name}" exceeds 5MB limit`);
   139→        continue;
   140→      }
   141→      validFiles.push(f);
   142→    }
   143→
   144→    if (validFiles.length === 0) return;
   145→
   146→    // Add items with uploading state
   147→    const newItems: ImageItem[] = validFiles.map(f => ({
   148→      file: f,
   149→      url: URL.createObjectURL(f),
   150→      uploading: true,
   151→      progress: 0,
   152→    }));
   153→    setImages(prev => [...prev, ...newItems]);
   154→
   155→    // Upload each file
   156→    for (let i = 0; i < validFiles.length; i++) {
   157→      const itemIdx = images.length + i;
   158→      try {
   159→        // Simulate progress
   160→        const progressInterval = setInterval(() => {
   161→          setImages(prev => {
   162→            const updated = [...prev];
   163→            if (updated[itemIdx]) {
   164→              updated[itemIdx] = {
   165→                ...updated[itemIdx],
   166→                progress: Math.min((updated[itemIdx].progress || 0) + Math.random() * 30, 90),
   167→              };
   168→            }
   169→            return updated;
   170→          });
   171→        }, 200);
   172→
   173→        const result = await api.uploadImages([validFiles[i]]);
   174→
   175→        clearInterval(progressInterval);
   176→
   177→        setImages(prev => {
   178→          const updated = [...prev];
   179→          if (updated[itemIdx] && result.url) {
   180→            updated[itemIdx] = {
   181→              ...updated[itemIdx],
   182→              serverUrl: result.url,
   183→              uploading: false,
   184→              progress: 100,
   185→            };
   186→          }
   187→          return updated;
   188→        });
   189→      } catch {
   190→        setImages(prev => prev.filter((_, idx) => idx !== itemIdx));
   191→        toast.error(`Failed to upload "${validFiles[i].name}"`);
   192→      }
   193→    }
   194→  }, [images.length]);
   195→
   196→  const handleDragOver = useCallback((e: React.DragEvent) => {
   197→    e.preventDefault();
   198→    e.stopPropagation();
   199→    setIsDragOver(true);
   200→  }, []);
   201→
   202→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   203→    e.preventDefault();
   204→    e.stopPropagation();
   205→    setIsDragOver(false);
   206→  }, []);
   207→
   208→  const handleDrop = useCallback((e: React.DragEvent) => {
   209→    e.preventDefault();
   210→    e.stopPropagation();
   211→    setIsDragOver(false);
   212→    if (e.dataTransfer.files.length > 0) {
   213→      processFiles(e.dataTransfer.files);
   214→    }
   215→  }, [processFiles]);
   216→
   217→  const removeImage = useCallback((index: number) => {
   218→    setImages(prev => {
   219→      const item = prev[index];
   220→      // Revoke object URL if it's a local preview
   221→      if (item && !item.serverUrl && item.url.startsWith('blob:')) {

   225→    });

   227→

   269→
   270→  // Get cities for selected state

   273→
   274→  // Populate form when editing
   275→  useEffect(() => {
   276→    if (editingProduct) {
   277→      setValue('title', editingProduct.title);
   278→      setValue('categoryId', editingProduct.categoryId);
   279→      setValue('description', editingProduct.description || '');
   280→      setValue('condition', editingProduct.condition);
   281→      setValue('brand', editingProduct.brand || '');
   282→      setValue('model', editingProduct.model || '');
   283→      setValue('purchaseYear', editingProduct.purchaseYear ? String(editingProduct.purchaseYear) : '');
   284→      setValue('dailyPrice', String(editingProduct.dailyPrice));
   285→      setValue('weeklyPrice', editingProduct.weeklyPrice ? String(editingProduct.weeklyPrice) : '');
   286→      setValue('securityDeposit', String(editingProduct.securityDeposit));
   287→      setValue('minRentalDays', String(editingProduct.minRentalDays));

   290→      setValue('cityId', editingProduct.cityId || '');
   291→      setValue('pickupAddress', editingProduct.pickupAddress || '');

   295→      setValue('cancellationPolicy', editingProduct.cancellationPolicy || '');

   297→    }

   311→      const pendingUpload = images.some(img => img.uploading);
   312→      if (pendingUpload) {
   313→        throw new Error('Please wait for images to finish uploading');
   314→      }
   315→
   316→      const payload: Record<string, unknown> = {
   317→        title: data.title,
   318→        categoryId: data.categoryId,
   319→        description: data.description,
   320→        condition: data.condition,
   321→        brand: data.brand || undefined,
   322→        model: data.model || undefined,
   323→        purchaseYear: data.purchaseYear ? parseInt(data.purchaseYear) : undefined,
   324→        dailyPrice: parseFloat(data.dailyPrice),
   325→        weeklyPrice: data.weeklyPrice ? parseFloat(data.weeklyPrice) : undefined,
   326→        securityDeposit: parseFloat(data.securityDeposit),
   327→        minRentalDays: parseInt(data.minRentalDays),
   328→        maxRentalDays: parseInt(data.maxRentalDays),
   329→        stateId: data.stateId,
   330→        cityId: data.cityId,
   331→        pickupAddress: data.pickupAddress,
   332→        deliveryAvailable: data.deliveryAvailable,
   333→        deliveryFee: data.deliveryFee ? parseFloat(data.deliveryFee) : 0,
   334→        rentalRules: data.rentalRules || undefined,
   335→        cancellationPolicy: data.cancellationPolicy || undefined,
   336→        ownerNotes: data.ownerNotes || undefined,
   337→      };
   338→
   339→      // Attach uploaded image URLs
   340→      const uploadedUrls = images
   341→        .filter(img => img.serverUrl)
   342→        .map(img => img.serverUrl);
   343→      if (uploadedUrls.length > 0) {
   344→        payload.imageUrls = uploadedUrls;
   345→      }
   346→
   347→      if (isEditing) {
   348→        return api.updateProduct(editingProduct.id, payload);

   363→    },
   364→    onError: (e: Error) => toast.error(e.message),
   365→  });

   369→  };

   371→  const isSubmitting = submitMutation.isPending;
   372→  const isLoadingData = loadingCategories || loadingStates;
   373→  const hasUploadingImages = images.some(img => img.uploading);
   374→
   375→  return (
   376→    <div className="min-h-screen bg-slate-50">
   377→      {/* Header */}
   378→      <div className="bg-white border-b border-slate-200 sticky top-0 z-30">
   379→        <div className="max-w-4xl mx-auto px-4 sm:px-6">
   380→          <div className="flex items-center justify-between h-16">
   381→            <div className="flex items-center gap-3">
   382→              <Button variant="ghost" size="sm" className="text-slate-600 hover:text-slate-900" onClick={() => navigate('my-listings')}>
   383→                <ArrowLeft className="h-4 w-4 mr-1" />Back
   384→              </Button>
   385→              <div className="h-6 w-px bg-slate-200" />
   386→              <h1 className="text-lg font-bold text-slate-900">{isEditing ? 'Edit Listing' : 'List a New Item'}</h1>
   387→            </div>
   388→          </div>
   389→        </div>
   390→      </div>
   391→
   392→      <div className="max-w-4xl mx-auto px-4 sm:px-6 py-6">
   393→        {isLoadingData ? (

   397→        ) : (

   399→            noValidate
   400→            initial={{ opacity: 0, y: 16 }}

   403→            className="space-y-6"

   405→            {/* Basic Information */}

   407→              <div className="space-y-4">
   408→                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   409→                  <div className="md:col-span-2">
   410→                    <Label htmlFor="title" className="text-sm font-medium text-slate-700">

   412→                    </Label>
   413→                    <Input id="title" placeholder="e.g., Canon EOS R5 Camera" className="mt-1.5" {...register('title')} />
   414→                    {errors.title && <p className="text-xs text-red-500 mt-1">{errors.title.message}</p>}
   415→                  </div>
   416→
   417→                  <div>
   418→                    <Label className="text-sm font-medium text-slate-700">Category <span className="text-red-500">*</span></Label>
   419→                    <Controller

   422→                      render={({ field }) => (

   424→                          <SelectTrigger className="mt-1.5"><SelectValue placeholder="Select category" /></SelectTrigger>
   425→                          <SelectContent>

   429→                          </SelectContent>
   430→                        </Select>
   431→                      )}
   432→                    />
   433→                    {errors.categoryId && <p className="text-xs text-red-500 mt-1">{errors.categoryId.message}</p>}
   434→                  </div>
   435→
   436→                  <div>
   437→                    <Label className="text-sm font-medium text-slate-700">Condition <span className="text-red-500">*</span></Label>
   438→                    <Controller
   439→                      name="condition"

   451→                    />

   460→                  {errors.description && <p className="text-xs text-red-500 mt-1">{errors.description.message}</p>}
