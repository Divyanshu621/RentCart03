   108→// ─── Props ─────────────────────────────────────────────────
   109→interface RentalDetailDialogProps {
   110→  rentalId: string | null;
   111→  open: boolean;
   112→  onClose: () => void;
   113→}
   114→
   115→export default function RentalDetailDialog({ rentalId, open, onClose }: RentalDetailDialogProps) {
   116→  const user = useAppStore((s) => s.user);
   117→  const navigate = useAppStore((s) => s.navigate);
   118→  const queryClient = useQueryClient();
   119→
   120→  const { data: rental, isLoading } = useQuery({
   121→    queryKey: ['rental', rentalId],
   122→    queryFn: async () => {
   123→      const data = await api.getRental(rentalId!);
   124→      return (data as any)?.rental as Rental;
   125→    },
   126→    enabled: !!rentalId && open,
   127→  });
   128→
   129→  const payMutation = useMutation({
   130→    mutationFn: (id: string) => api.payRental(id),
   131→    onSuccess: () => { toast.success('Payment successful!'); queryClient.invalidateQueries({ queryKey: ['rental', rentalId] }); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   132→    onError: (e: Error) => toast.error(e.message),
   133→  });
   134→
   135→  const [cancelDialogOpen, setCancelDialogOpen] = useState(false);
   136→  const [cancelSuccess, setCancelSuccess] = useState(false);
   137→
   138→  const cancelMutation = useMutation({
   139→    mutationFn: ({ id, reason }: { id: string; reason?: string }) => api.cancelRental(id, reason),
   140→    onSuccess: () => { setCancelSuccess(true); queryClient.invalidateQueries({ queryKey: ['rental', rentalId] }); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   141→    onError: (e: Error) => toast.error(e.message),
   142→  });
   143→
   144→  const acceptMutation = useMutation({
   145→    mutationFn: (id: string) => api.acceptRental(id),
   146→    onSuccess: () => { toast.success('Rental accepted!'); queryClient.invalidateQueries({ queryKey: ['rental', rentalId] }); queryClient.invalidateQueries({ queryKey: ['rentals'] }); onClose(); },
   147→    onError: (e: Error) => toast.error(e.message),
   148→  });
   149→
   150→  const rejectMutation = useMutation({
   151→    mutationFn: (id: string) => api.rejectRental(id),
   152→    onSuccess: () => { toast.success('Rental rejected'); queryClient.invalidateQueries({ queryKey: ['rental', rentalId] }); queryClient.invalidateQueries({ queryKey: ['rentals'] }); onClose(); },
   153→    onError: (e: Error) => toast.error(e.message),
   154→  });
   155→
   156→  const returnMutation = useMutation({
   157→    mutationFn: (id: string) => api.returnRental(id),
   158→    onSuccess: () => { toast.success('Return initiated'); queryClient.invalidateQueries({ queryKey: ['rental', rentalId] }); queryClient.invalidateQueries({ queryKey: ['rentals'] }); },
   159→    onError: (e: Error) => toast.error(e.message),
   160→  });
   161→
   162→  const respondExtMutation = useMutation({
   163→    mutationFn: ({ rentalId, extId, approved }: { rentalId: string; extId: string; approved: boolean }) =>
   164→      api.respondExtension(rentalId, extId, { approved }),
   165→    onSuccess: () => { toast.success('Extension response recorded'); queryClient.invalidateQueries({ queryKey: ['rental', rentalId] }); },
   166→    onError: (e: Error) => toast.error(e.message),
   167→  });
   168→
   169→  const isOwner = user?.id === rental?.ownerId;
   170→  const isCustomer = user?.id === rental?.customerId;
   171→
   172→  const getCurrentStepIndex = () => {
   173→    if (!rental) return -1;
   174→    const idx = statusFlow.indexOf(rental.status);
   175→    if (idx !== -1) return idx;
   176→    if (rental.status === 'CANCELLED') return statusFlow.indexOf('OWNER_PENDING');
   177→    if (rental.status === 'OVERDUE') return statusFlow.indexOf('ACTIVE');
   178→    if (rental.status === 'DISPUTED') return statusFlow.indexOf('ACTIVE');
   179→    if (rental.status === 'OWNER_REJECTED') return statusFlow.indexOf('OWNER_PENDING');
   180→    return -1;
   181→  };
   182→
   183→  const currentStep = getCurrentStepIndex();
   184→
   185→  return (
   186→    <>
   187→    <Dialog open={open} onOpenChange={(v) => { if (!v) onClose(); }}>
   188→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto p-0">
   189→        <DialogHeader className="p-6 pb-0 sticky top-0 bg-white z-10 border-b border-slate-100 rounded-t-lg">
   190→          <div className="flex items-center justify-between pr-6">
   191→            <DialogTitle className="text-lg font-bold text-slate-900">Rental Details</DialogTitle>
   192→            <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-slate-400" onClick={onClose}>
   193→              <X className="h-4 w-4" />
   194→            </Button>
   195→          </div>
   196→        </DialogHeader>
   197→
   198→        {isLoading || !rental ? (
   199→          <div className="p-6 space-y-4">
   200→            <Skeleton className="h-6 w-48" />
   201→            <Skeleton className="h-4 w-full" />
   202→            <Skeleton className="h-4 w-3/4" />
   203→            <Skeleton className="h-32 w-full" />
   204→            <Skeleton className="h-24 w-full" />
   205→          </div>
   206→        ) : (
   207→          <div className="p-6 space-y-6">
   208→            {/* Status Badge */}
   209→            <div className="flex items-center justify-between">
   210→              <span className={`inline-flex items-center px-3 py-1 rounded-full text-xs font-semibold border ${statusColors[rental.status]}`}>
   211→                {statusLabels[rental.status]}
   212→              </span>
   213→              <span className="text-xs text-slate-400">ID: {rental.id.slice(0, 8)}</span>
   214→            </div>
   215→
   216→            {/* Product Info */}
   217→            <div
   218→              className="flex gap-4 p-4 bg-slate-50 rounded-lg border border-slate-200 cursor-pointer hover:bg-slate-100 transition-colors"
   219→              onClick={() => { onClose(); navigate('product', { productId: rental.productId }); }}
   220→            >
   221→              <div className="h-16 w-16 rounded-lg bg-gradient-to-br from-slate-400 to-gray-300 flex items-center justify-center shrink-0">
   222→                <Package className="h-8 w-8 text-white/80" strokeWidth={1.5} />
   223→              </div>
   224→              <div className="flex-1 min-w-0">
   225→                <h3 className="font-semibold text-slate-900 text-sm">{rental.product.title}</h3>
   226→                <p className="text-xs text-slate-500 mt-0.5">Condition: {rental.product.condition.replace('_', ' ')}</p>
   227→                {rental.product.pickupAddress && (
   228→                  <div className="flex items-center gap-1 text-xs text-slate-400 mt-1">
   229→                    <MapPin className="h-3 w-3" />{rental.product.pickupAddress}
   230→                  </div>
   231→                )}
   232→              </div>
   233→              <div className="text-right shrink-0">
   234→                <p className="text-sm font-bold text-emerald-600">₹{rental.dailyRate.toLocaleString('en-IN')}</p>
   235→                <p className="text-[11px] text-slate-400">/day</p>
   236→              </div>
   237→            </div>
   238→
   239→            {/* People */}
   240→            <div className="grid grid-cols-2 gap-4">
   241→              <div className="flex items-center gap-3 p-3 bg-slate-50 rounded-lg">
   242→                <Avatar className="h-9 w-9">
   243→                  <AvatarFallback className="bg-emerald-100 text-emerald-700 text-xs font-semibold">
   244→                    {rental.customer.name.split(' ').map((n) => n[0]).join('').slice(0, 2).toUpperCase()}
   245→                  </AvatarFallback>
   246→                </Avatar>
   247→                <div className="min-w-0">
   248→                  <p className="text-xs text-slate-400">Renter</p>
   249→                  <p className="text-sm font-medium text-slate-900 truncate">{rental.customer.name}</p>
   250→                </div>
   251→              </div>
   252→              <div className="flex items-center gap-3 p-3 bg-slate-50 rounded-lg">
   253→                <Avatar className="h-9 w-9">
   254→                  <AvatarFallback className="bg-sky-100 text-sky-700 text-xs font-semibold">
   255→                    {rental.owner.name.split(' ').map((n) => n[0]).join('').slice(0, 2).toUpperCase()}
   256→                  </AvatarFallback>
   257→                </Avatar>
   258→                <div className="min-w-0">
   259→                  <p className="text-xs text-slate-400">Owner</p>
   260→                  <p className="text-sm font-medium text-slate-900 truncate">{rental.owner.name}</p>
   261→                </div>
   262→              </div>
   263→            </div>
   264→
   265→            {/* Dates & Period */}
   266→            <div className="grid grid-cols-2 gap-4">
   267→              <div className="p-3 bg-slate-50 rounded-lg">
   268→                <div className="flex items-center gap-1.5 text-xs text-slate-400 mb-1">
   269→                  <CalendarRange className="h-3 w-3" />Start Date
   270→                </div>
   271→                <p className="text-sm font-medium text-slate-900">{format(new Date(rental.startDate), 'dd MMM yyyy')}</p>
   272→              </div>
   273→              <div className="p-3 bg-slate-50 rounded-lg">
   274→                <div className="flex items-center gap-1.5 text-xs text-slate-400 mb-1">
   275→                  <CalendarRange className="h-3 w-3" />End Date
   276→                </div>
   277→                <p className="text-sm font-medium text-slate-900">{format(new Date(rental.endDate), 'dd MMM yyyy')}</p>
   278→              </div>
   279→            </div>
   280→            <div className="flex items-center justify-center gap-2 text-xs text-slate-500">
   281→              <Clock className="h-3 w-3" />
   282→              <span>{rental.rentalDays} days rental period</span>
   283→            </div>
   284→
   285→            <Separator />
   286→
   287→            {/* Pricing Breakdown */}
   288→            <div>
   289→              <h4 className="text-sm font-semibold text-slate-900 mb-3">Pricing Breakdown</h4>
   290→              <div className="space-y-2 text-sm">
   291→                <div className="flex justify-between"><span className="text-slate-500">Rental ({rental.rentalDays} days × ₹{rental.dailyRate.toLocaleString('en-IN')})</span><span className="text-slate-900">₹{rental.rentalAmount.toLocaleString('en-IN')}</span></div>
   292→                <div className="flex justify-between"><span className="text-slate-500">Platform Fee (10%)</span><span className="text-slate-900">₹{rental.platformFee.toLocaleString('en-IN')}</span></div>
   293→                <div className="flex justify-between"><span className="text-slate-500">GST (18%)</span><span className="text-slate-900">₹{rental.tax.toLocaleString('en-IN')}</span></div>
   294→                {rental.deliveryFee > 0 && (
   295→                  <div className="flex justify-between"><span className="text-slate-500">Delivery Fee</span><span className="text-slate-900">₹{rental.deliveryFee.toLocaleString('en-IN')}</span></div>
   296→                )}
   297→                {rental.discount > 0 && (
   298→                  <div className="flex justify-between text-emerald-600"><span>Discount</span><span>-₹{rental.discount.toLocaleString('en-IN')}</span></div>
   299→                )}
   300→                <div className="flex justify-between"><span className="text-slate-500">Security Deposit</span><span className="text-slate-900">₹{rental.securityDeposit.toLocaleString('en-IN')}</span></div>
   301→                {rental.totalLateFee > 0 && (
   302→                  <div className="flex justify-between text-red-600"><span>Late Fee</span><span>₹{rental.totalLateFee.toLocaleString('en-IN')}</span></div>
   303→                )}
   304→                <Separator />
   305→                <div className="flex justify-between font-bold text-base"><span className="text-slate-900">Total</span><span className="text-emerald-600">₹{rental.totalAmount.toLocaleString('en-IN')}</span></div>
   306→              </div>
   307→            </div>
   308→
   309→            <Separator />
   310→
   311→            {/* Status Timeline */}
   312→            <div>
   313→              <h4 className="text-sm font-semibold text-slate-900 mb-3">Status Timeline</h4>
   314→              <div className="relative pl-6">
   315→                {statusFlow.map((step, idx) => {
   316→                  const isReached = idx <= currentStep;
   317→                  const isCurrent = idx === currentStep;
   318→                  const isRejected = rental.status === 'OWNER_REJECTED' && step === 'OWNER_ACCEPTED';
   319→                  const isCancelled = rental.status === 'CANCELLED';
   320→                  const showTerminal = (isRejected || isCancelled || rental.status === 'OVERDUE' || rental.status === 'DISPUTED') && idx === currentStep;
   321→
   322→                  return (
   323→                    <div key={step} className="relative pb-4 last:pb-0">
   324→                      {/* Vertical line */}
   325→                      {idx < statusFlow.length - 1 && (
   326→                        <div className={`absolute left-[-18px] top-5 w-0.5 h-full ${isReached && !isRejected ? 'bg-emerald-400' : 'bg-slate-200'}`} />
   327→                      )}
   328→                      {/* Circle */}
   329→                      <div className={`absolute left-[-24px] top-0.5 h-4 w-4 rounded-full border-2 flex items-center justify-center ${
   330→                        isRejected || showTerminal
   331→                          ? 'border-red-400 bg-red-100'
   332→                          : isCurrent
   333→                            ? 'border-emerald-500 bg-emerald-500'
   334→                            : isReached
   335→                              ? 'border-emerald-400 bg-emerald-400'
   336→                              : 'border-slate-300 bg-white'
   337→                      }`}>
   338→                        {(isCurrent || isReached) && !isRejected && !showTerminal && (
   339→                          <CheckCircle2 className="h-2.5 w-2.5 text-white" strokeWidth={3} />
   340→                        )}
   341→                        {isRejected && <X className="h-2.5 w-2.5 text-red-500" />}
   342→                      </div>
   343→                      <div className="flex items-center gap-2">
   344→                        <span className={`text-xs font-medium ${isCurrent ? 'text-emerald-700' : isReached && !isRejected ? 'text-slate-700' : 'text-slate-400'}`}>
   345→                          {statusLabels[step]}
   346→                        </span>
   347→                        {isCurrent && (
   348→                          <span className="text-[10px] bg-emerald-100 text-emerald-700 px-1.5 py-0.5 rounded-full">Current</span>
   349→                        )}
   350→                      </div>
   351→                      {/* Show terminal status label */}
   352→                      {showTerminal && (
   353→                        <div className="flex items-center gap-1 mt-0.5">
   354→                          <ArrowRight className="h-3 w-3 text-red-400" />
   355→                          <span className={`text-xs font-semibold ${statusColors[rental.status]}`}>{statusLabels[rental.status]}</span>
   356→                        </div>
   357→                      )}
   358→                    </div>
   359→                  );
   360→                })}
   361→              </div>
   362→            </div>
   363→
   364→            <Separator />
   365→
   366→            {/* Payment History */}
   367→            <div>
   368→              <h4 className="text-sm font-semibold text-slate-900 mb-3">Payment History</h4>
   369→              {rental.payments && rental.payments.length > 0 ? (
   370→                <div className="space-y-2">
   371→                  {(rental.payments as Payment[]).map((payment) => (
   372→                    <div key={payment.id} className="flex items-center justify-between p-3 bg-slate-50 rounded-lg">
   373→                      <div className="flex items-center gap-3">
   374→                        <div className={`h-8 w-8 rounded-full flex items-center justify-center ${
   375→                          payment.type === 'REFUND' ? 'bg-sky-100' : 'bg-emerald-100'
   376→                        }`}>
   377→                          {payment.type === 'REFUND' ? (
   378→                            <RotateCcw className="h-4 w-4 text-sky-600" />
   379→                          ) : (
   380→                            <CreditCard className="h-4 w-4 text-emerald-600" />
   381→                          )}
   382→                        </div>
   383→                        <div>
   384→                          <p className="text-sm font-medium text-slate-900">{paymentTypeLabels[payment.type] || payment.type}</p>
   385→                          <p className="text-[11px] text-slate-400">{format(new Date(payment.createdAt), 'dd MMM yyyy, hh:mm a')}</p>
   386→                        </div>
   387→                      </div>
   388→                      <div className="text-right">
   389→                        <p className={`text-sm font-semibold ${payment.type === 'REFUND' ? 'text-sky-600' : 'text-slate-900'}`}>
   390→                          {payment.type === 'REFUND' ? '+' : '-'}₹{payment.amount.toLocaleString('en-IN')}
   391→                        </p>
   392→                        <span className={`text-[10px] px-1.5 py-0.5 rounded-full ${paymentStatusColors[payment.status]}`}>
   393→                          {payment.status}
   394→                        </span>
   395→                      </div>
   396→                    </div>
   397→                  ))}
   398→                </div>
   399→              ) : (
   400→                <p className="text-sm text-slate-400">No payments yet.</p>
   401→              )}
   402→            </div>
   403→
   404→            {/* Extension Requests */}
   405→            {rental.extensionRequests && rental.extensionRequests.length > 0 && (
   406→              <>
   407→                <Separator />
   408→                <div>
   409→                  <h4 className="text-sm font-semibold text-slate-900 mb-3">Extension Requests</h4>
   410→                  <div className="space-y-2">
   411→                    {(rental.extensionRequests as ExtensionRequest[]).map((ext) => (
   412→                      <div key={ext.id} className="p-3 bg-slate-50 rounded-lg">
   413→                        <div className="flex items-center justify-between mb-1">
   414→                          <span className="text-sm font-medium text-slate-900">+{ext.requestedDays} days</span>
   415→                          <span className={`text-[10px] px-1.5 py-0.5 rounded-full ${extensionStatusColors[ext.status]}`}>
   416→                            {ext.status}
   417→                          </span>
   418→                        </div>
   419→                        <p className="text-xs text-slate-500">New end: {format(new Date(ext.newEndDate), 'dd MMM yyyy')} • Fee: ₹{ext.additionalFee.toLocaleString('en-IN')}</p>
   420→                        {ext.reason && <p className="text-xs text-slate-400 mt-1 italic">&quot;{ext.reason}&quot;</p>}
   421→                        {ext.status === 'PENDING' && isOwner && (
   422→                          <div className="flex gap-2 mt-2">
   423→                            <Button
   424→                              size="sm"
   425→                              className="h-7 px-3 text-[11px] bg-emerald-600 hover:bg-emerald-700 text-white"
