   143→export default function PaymentCheckoutModal({
   144→  open,
   145→  onClose,
   146→  rentalId,
   147→  rentalData,
   148→  onSuccess,
   149→}: PaymentCheckoutModalProps) {
   150→  const isMobile = useIsMobile();
   151→  const queryClient = useQueryClient();
   152→  const [enabledMethods, setEnabledMethods] = useState<Record<string, boolean> | null>(null);
   153→  const [selectedMethod, setSelectedMethod] = useState('razorpay');
   154→  const [upiId, setUpiId] = useState('');
   155→  const [cardNumber, setCardNumber] = useState('');
   156→  const [cardExpiry, setCardExpiry] = useState('');
   157→  const [cardCvv, setCardCvv] = useState('');
   158→  const [cardName, setCardName] = useState('');
   159→  const [selectedBank, setSelectedBank] = useState('');
   160→  const [selectedWallet, setSelectedWallet] = useState('');
   161→  const [processing, setProcessing] = useState(false);
   162→  const [step, setStep] = useState<'checkout' | 'processing' | 'success'>('checkout');
   163→  const [txnId, setTxnId] = useState('');
   164→  const [summaryExpanded, setSummaryExpanded] = useState(false);
   165→
   166→  // Fetch enabled payment methods from backend
   167→  useEffect(() => {
   168→    if (!open) return;

   170→    api.getPaymentSettings().then((res) => {
   171→      if (cancelled) return;

   174→      if (!res.enabledMethods[selectedMethod]) {
   175→        const firstEnabled = Object.entries(res.enabledMethods).find(([, v]) => v)?.[0];
   176→        if (firstEnabled) setSelectedMethod(firstEnabled);
   177→      }
   178→    }).catch(() => {
   179→      // Fallback: enable all if settings fetch fails
   180→      if (!cancelled) {
   181→        setEnabledMethods({
   182→          razorpay: true, upi: true, card: true, netbanking: true, wallet: true, cash: true,
   183→        });
   184→      }
   185→    });
   186→    return () => { cancelled = true; };
   187→  }, [open]);
   188→

   190→  const paymentMethods = useMemo(() => {
   191→    if (!enabledMethods) return allPaymentMethods; // show all while loading
   192→    return allPaymentMethods.filter((m) => enabledMethods[m.id]);
   193→  }, [enabledMethods]);
   194→
   195→  const dateRange = useMemo(() => {
   196→    if (!rentalData) return '';

   199→
   200→  const formatCardNumber = (value: string) => {
   201→    const v = value.replace(/\D/g, '').slice(0, 16);
   202→    return v.replace(/(.{4})/g, '$1 ').trim();
   203→  };
   204→
   205→  const formatExpiry = (value: string) => {
   206→    const v = value.replace(/\D/g, '').slice(0, 4);
   207→    if (v.length >= 3) return v.slice(0, 2) + '/' + v.slice(2);
   208→    return v;
   209→  };

   211→  const validateUPI = (id: string) => {
   212→    return id.includes('@') && id.length >= 3;
   213→  };
   214→
   215→  const validateCard = () => {
   216→    const num = cardNumber.replace(/\s/g, '');
   217→    return num.length === 16 && cardExpiry.length === 5 && cardCvv.length === 3 && cardName.trim().length > 0;
   218→  };

   223→      case 'razorpay': return `Pay ${amount} with Razorpay`;
   224→      case 'upi': return `Pay ${amount} via UPI`;
   225→      case 'card': return `Pay ${amount} via Card`;
   226→      case 'netbanking': return `Pay ${amount} via Net Banking`;
   227→      case 'wallet': return `Pay ${amount} via Wallet`;
   228→      case 'cash': return 'Confirm Cash on Pickup';
   229→      default: return `Pay ${amount}`;
   230→    }
   231→  };
   232→
   233→  const isPayDisabled = () => {
   234→    if (processing) return true;
   235→    switch (selectedMethod) {
   236→      case 'upi': return !validateUPI(upiId);
   237→      case 'card': return !validateCard();
   238→      case 'netbanking': return !selectedBank;
   239→      case 'wallet': return !selectedWallet;
   240→      default: return false;
   241→    }
   242→  };
   243→

   248→    setCardExpiry('');
   249→    setCardCvv('');
   250→    setCardName('');
   251→    setSelectedBank('');
   252→    setSelectedWallet('');
   253→    setProcessing(false);
   254→    setStep('checkout');
   255→    setTxnId('');
   256→    setSummaryExpanded(false);
   257→  }, []);
   258→

   261→    setStep('success');

   263→    queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   264→    queryClient.invalidateQueries({ queryKey: ['my-rentals'] });
   265→  }, [queryClient]);

   271→  }, []);
   272→
   273→  const handlePayment = async () => {
   274→    if (!rentalData) return;
   275→
   276→    setProcessing(true);
   277→    setStep('processing');
   278→
   279→    try {
   280→      if (selectedMethod === 'cash') {
   281→        await api.verifyPayment({

   286→        handlePaymentSuccess(txId);
   287→        return;
   288→      }
   289→
   290→      // Create order
   291→      const orderData = await api.createPaymentOrder(rentalId, selectedMethod.toUpperCase()) as Record<string, unknown>;
   292→
   293→      if (orderData.method === 'razorpay' && typeof window !== 'undefined' && window.Razorpay) {
   294→        // Real Razorpay mode
   295→        const rzpOptions: Record<string, unknown> = {
   296→          key: orderData.key,
   297→          amount: Math.round((orderData.amount as number) * 100),
   298→          currency: orderData.currency,
   299→          name: 'RentCart',
   300→          description: `Rental: ${rentalData.productTitle || 'Item'}`,
   301→          order_id: orderData.orderId,
   302→          prefill: {
   303→            name: (orderData.customer as Record<string, string>)?.name || '',

   305→            contact: (orderData.customer as Record<string, string>)?.contact || '',
   306→          },
   307→          theme: { color: '#059669' },
   308→          modal: !isMobile,
   309→          handler: async (response: Record<string, string>) => {
   310→            try {
   311→              await api.verifyPayment({
   312→                rentalId,
   313→                razorpayPaymentId: response.razorpay_payment_id,
   314→                razorpayOrderId: response.razorpay_order_id,
   315→                razorpaySignature: response.razorpay_signature,
   316→              });
   317→              handlePaymentSuccess(response.razorpay_payment_id || (orderData.orderId as string));
   318→            } catch {
   319→              handlePaymentError();
   320→            }
   321→          },
   322→        };
   323→
   324→        const rzp = new window.Razorpay(rzpOptions);
   325→
   326→        rzp.on('payment.failed', () => {
   327→          handlePaymentError();
   328→        });
   329→
   330→        rzp.open();
   331→        // Don't reset processing here — the handler callbacks will do it
   332→        return;
   333→      }
   334→
   335→      // Simulated/demo mode — add a short delay to mimic real payment
   336→      await new Promise((resolve) => setTimeout(resolve, 2000));
   337→      await api.verifyPayment({
   338→        rentalId,
   339→        paymentMethod: selectedMethod.toUpperCase(),
   340→      });
   341→      const txId = (orderData.orderId as string) || `TXN-${Date.now()}`;
   342→      handlePaymentSuccess(txId);
   343→    } catch {
   344→      handlePaymentError();
   345→    }
   346→  };
   347→
   348→  const handleClose = () => {
   349→    if (processing) return;
   350→    if (step === 'success') {
   351→      onSuccess?.();
   352→    }
   353→    resetState();
   354→    onClose();
   355→  };
   356→
   357→  const handleSuccess = () => {
   358→    onSuccess?.();
   359→    resetState();
   360→    onClose();
   361→  };
   362→
   363→  const renderOrderSummary = () => (
   364→    <div className="rounded-xl border border-gray-200 bg-gray-50/50 overflow-hidden">
   365→      {/* Compact header — always visible on mobile */}
   366→      <button
   367→        type="button"
   368→        onClick={() => setSummaryExpanded(!summaryExpanded)}
   369→        className="w-full flex items-center justify-between p-3 sm:p-4 hover:bg-gray-100/50 transition-colors"
   370→      >
   371→        <div className="flex items-center gap-2.5 min-w-0">
   372→          <h3 className="text-sm font-semibold text-gray-900">Order Summary</h3>
   373→          <Badge variant="secondary" className="text-[10px] font-medium bg-emerald-100 text-emerald-700 shrink-0">
   374→            {rentalData?.rentalDays} days
   375→          </Badge>
   376→        </div>
   377→        <div className="flex items-center gap-2 shrink-0">
   378→          <span className="text-sm font-bold text-gray-900">₹{formatCurrency(rentalData?.totalAmount ?? 0)}</span>
   379→          {isMobile && (
   380→            summaryExpanded ? <ChevronUp className="size-4 text-gray-400" /> : <ChevronDown className="size-4 text-gray-400" />

   382→        </div>
   383→      </button>
   384→
   385→      {/* Expandable details — always shown on desktop, toggleable on mobile */}
   386→      <AnimatePresence>
   387→        {(!isMobile || summaryExpanded) && (
   388→          <motion.div
   389→            initial={isMobile ? { height: 0, opacity: 0 } : false}

   392→            transition={{ duration: 0.2 }}
   393→            className="overflow-hidden"
   394→          >
   395→            <Separator className="bg-gray-200" />
   396→            <div className="p-3 sm:p-4 space-y-2">
   397→              <p className="text-sm font-medium text-gray-800 truncate">
   398→                {rentalData?.productTitle || 'Rental Item'}
   399→              </p>
   400→              <p className="text-xs text-gray-500">{dateRange}</p>
   401→
   402→              <Separator className="my-2.5 bg-gray-200" />
   403→
   404→              <div className="space-y-1.5 text-xs">
   405→                <div className="flex items-center justify-between">
   406→                  <span className="text-gray-500">Rental Amount</span>
   407→                  <span className="font-medium text-gray-700">
   408→                    ₹{rentalData?.dailyRate.toLocaleString('en-IN')} × {rentalData?.rentalDays} days
   409→                  </span>
   410→                </div>
   411→                <div className="flex items-center justify-between">
   412→                  <span className="text-gray-500">Platform Fee (10%)</span>
   413→                  <span className="font-medium text-gray-700">₹{formatCurrency(rentalData?.platformFee ?? 0)}</span>
   414→                </div>
   415→                <div className="flex items-center justify-between">
   416→                  <span className="text-gray-500">GST (18%)</span>
   417→                  <span className="font-medium text-gray-700">₹{formatCurrency(rentalData?.tax ?? 0)}</span>
   418→                </div>
   419→                {rentalData && rentalData.deliveryFee > 0 && (
   420→                  <div className="flex items-center justify-between">
   421→                    <span className="text-gray-500">Delivery Fee</span>
   422→                    <span className="font-medium text-gray-700">₹{formatCurrency(rentalData.deliveryFee)}</span>
   423→                  </div>
   424→                )}
   425→                <div className="flex items-center justify-between">
   426→                  <span className="text-gray-500">Security Deposit</span>
   427→                  <div className="flex items-center gap-1.5">
   428→                    <span className="font-medium text-gray-700">₹{formatCurrency(rentalData?.securityDeposit ?? 0)}</span>
   429→                    <span className="text-[10px] text-emerald-600 font-medium">Refundable</span>
   430→                  </div>
   431→                </div>

   438→              </div>
   439→
   440→              <Separator className="my-2.5 bg-gray-200" />
   441→
   442→              <div className="flex items-center justify-between">
   443→                <span className="text-sm font-bold text-gray-900">Total Payable</span>
   444→                <span className="text-base font-bold text-gray-900">₹{formatCurrency(rentalData?.totalAmount ?? 0)}</span>
   445→              </div>
   446→            </div>
   447→          </motion.div>
   448→        )}
   449→      </AnimatePresence>
   450→    </div>
   451→  );
   452→
   453→  const renderMethodCards = () => (
   454→    <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   455→      {paymentMethods.map((method) => {
   456→        const isSelected = selectedMethod === method.id;
   457→        return (
   458→          <motion.button
   459→            key={method.id}
   460→            type="button"
   461→            whileTap={{ scale: 0.97 }}
   462→            onClick={() => {
   463→              setSelectedMethod(method.id);
   464→              setSelectedBank('');
   465→              setSelectedWallet('');
   466→            }}
   467→            className={`
   468→              relative flex items-center gap-3 rounded-xl border-2 p-3 cursor-pointer transition-all text-left w-full min-h-[56px]
   469→              ${isSelected
   470→                ? 'border-emerald-500 bg-emerald-50 shadow-sm'
   471→                : 'border-gray-200 hover:border-gray-300 bg-white'
   472→              }
   473→            `}
   474→          >
   475→            <div
   476→              className={`
   477→                flex items-center justify-center size-9 rounded-lg shrink-0
   478→                ${isSelected ? 'bg-emerald-100 text-emerald-700' : 'bg-gray-100 text-gray-500'}
   479→              `}
   480→            >
   481→              {method.icon}
   482→            </div>
   483→            <div className="min-w-0 flex-1">

   486→                  {method.label}
   487→                </span>
   488→                {method.tag && method.tagColor && (
   489→                  <Badge

   492→                    {method.tag}
   493→                  </Badge>
   494→                )}
   495→              </div>

   498→            {isSelected && (

   507→  );
   508→
   509→  const renderUPIForm = () => (

   517→    >
   518→      <div className="rounded-xl border border-gray-200 bg-white p-3 sm:p-4 space-y-3">
   519→        <div>
