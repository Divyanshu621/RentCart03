    34→// ─── Animated Counter ──────────────────────────────────────
    35→function AnimatedCounter({ target, prefix = '', suffix = '', decimals = 0 }: {
    36→  target: number; prefix?: string; suffix?: string; decimals?: number;
    37→}) {
    38→  const [count, setCount] = useState(0);
    39→  const ref = useRef<HTMLSpanElement>(null);
    40→  const started = useRef(false);
    41→
    42→  useEffect(() => {
    43→    const el = ref.current;
    44→    if (!el) return;
    45→    const observer = new IntersectionObserver(
    46→      ([entry]) => {
    47→        if (entry.isIntersecting && !started.current) {
    48→          started.current = true;
    49→          const duration = 1200;
    50→          const startTime = performance.now();
    51→          const animate = (now: number) => {
    52→            const elapsed = now - startTime;
    53→            const progress = Math.min(elapsed / duration, 1);
    54→            // ease-out cubic
    55→            const eased = 1 - Math.pow(1 - progress, 3);
    56→            setCount(eased * target);
    57→            if (progress < 1) requestAnimationFrame(animate);
    58→          };
    59→          requestAnimationFrame(animate);
    60→        }
    61→      },
    62→      { threshold: 0.1 }
    63→    );
    64→    observer.observe(el);
    65→    return () => observer.disconnect();
    66→  }, [target]);
    67→
    68→  const formatted = decimals > 0
    69→    ? count.toLocaleString('en-IN', { minimumFractionDigits: decimals, maximumFractionDigits: decimals })
    70→    : Math.round(count).toLocaleString('en-IN');
    71→
    72→  return <span ref={ref}>{prefix}{formatted}{suffix}</span>;
    73→}
